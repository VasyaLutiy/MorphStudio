```go
package api

import (
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"
	"time"

	"morphstudio/auth"
	"morphstudio/internal/testhelp"
	"morphstudio/project"
	"morphstudio/store"
)

var testNow = time.Date(2026, time.October, 8, 9, 0, 0, 0, time.UTC)

func exampleRouter() (http.Handler, *store.Mem) {
	st := store.NewMem()
	users := auth.Users{"ann": auth.HashPassword("pw1", "x")}
	h := Handlers{
		Store: st,
		Now:   func() time.Time { return testNow },
		NewID: func() string { return "r-1" },
	}
	return NewRouter(h, users), st
}

func TestRouterExample1(t *testing.T) {
	r, _ := exampleRouter()

	rec := httptest.NewRecorder()
	req := httptest.NewRequest("GET", "/healthz", nil)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "status", rec.Code, http.StatusOK)
	testhelp.Equal(t, "Content-Type", rec.Header().Get("Content-Type"), "application/json")
	testhelp.Equal(t, "body", rec.Body.String(), "{\"status\":\"ok\"}\n")
}

func TestRouterExample2(t *testing.T) {
	r, st := exampleRouter()

	rec := httptest.NewRecorder()
	req := httptest.NewRequest("GET", "/v1/projects", nil)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "GET status", rec.Code, http.StatusUnauthorized)
	testhelp.Equal(t, "GET WWW-Authenticate", rec.Header().Get("WWW-Authenticate"), `Basic realm="morph-studio"`)
	testhelp.Equal(t, "GET body", rec.Body.String(), "{\"error\":{\"code\":\"unauthorized\"}}\n")

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("POST", "/v1/projects", strings.NewReader(`{"name":"A","slug":"a"}`))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Basic YW5uOndyb25n")
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "POST status", rec.Code, http.StatusUnauthorized)
	testhelp.Equal(t, "POST WWW-Authenticate", rec.Header().Get("WWW-Authenticate"), `Basic realm="morph-studio"`)
	testhelp.Equal(t, "POST body", rec.Body.String(), "{\"error\":{\"code\":\"unauthorized\"}}\n")

	items, err := st.List()
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "store", items, []project.Project{})
}

func TestRouterExample3(t *testing.T) {
	r, _ := exampleRouter()
	authHeader := "Basic YW5uOnB3MQ=="

	rec := httptest.NewRecorder()
	req := httptest.NewRequest("POST", "/v1/projects", strings.NewReader(`{"name":"A","slug":"a"}`))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "POST status", rec.Code, http.StatusCreated)
	testhelp.Equal(t, "POST Location", rec.Header().Get("Location"), "/v1/projects/r-1")
	createdJSON := `{"id":"r-1","name":"A","slug":"a","description":"","status":"draft","createdAt":"2026-10-08T09:00:00Z","updatedAt":"2026-10-08T09:00:00Z"}`
	testhelp.Equal(t, "POST body", rec.Body.String(), createdJSON+"\n")

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("GET", "/v1/projects", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "GET list status", rec.Code, http.StatusOK)
	testhelp.Equal(t, "GET list body", rec.Body.String(), `{"items":[`+createdJSON+`]}`+"\n")

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("GET", "/v1/projects/r-1", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "GET status", rec.Code, http.StatusOK)
	testhelp.Equal(t, "GET body", rec.Body.String(), createdJSON+"\n")

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("PUT", "/v1/projects/r-1", strings.NewReader(`{"name":"B","slug":"b"}`))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "PUT status", rec.Code, http.StatusOK)
	updatedJSON := `{"id":"r-1","name":"B","slug":"b","description":"","status":"draft","createdAt":"2026-10-08T09:00:00Z","updatedAt":"2026-10-08T09:00:00Z"}`
	testhelp.Equal(t, "PUT body", rec.Body.String(), updatedJSON+"\n")

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("DELETE", "/v1/projects/r-1", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "DELETE status", rec.Code, http.StatusNoContent)

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("GET", "/v1/projects/r-1", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)

	testhelp.Equal(t, "GET after delete status", rec.Code, http.StatusNotFound)
}

func TestRouterExample4(t *testing.T) {
	r, _ := exampleRouter()
	authHeader := "Basic YW5uOnB3MQ=="

	rec := httptest.NewRecorder()
	req := httptest.NewRequest("PATCH", "/v1/projects/r-1", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)
	testhelp.Equal(t, "PATCH status", rec.Code, http.StatusMethodNotAllowed)

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("DELETE", "/v1/projects", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)
	testhelp.Equal(t, "DELETE collection status", rec.Code, http.StatusMethodNotAllowed)

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("POST", "/healthz", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)
	testhelp.Equal(t, "POST healthz status", rec.Code, http.StatusMethodNotAllowed)

	rec = httptest.NewRecorder()
	req = httptest.NewRequest("GET", "/nowhere", nil)
	req.Header.Set("Authorization", authHeader)
	r.ServeHTTP(rec, req)
	testhelp.Equal(t, "GET nowhere status", rec.Code, http.StatusNotFound)
}
```
