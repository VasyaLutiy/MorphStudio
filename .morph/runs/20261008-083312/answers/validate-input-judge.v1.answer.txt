package project

import (
	"strings"
	"testing"

	"morphstudio/internal/testhelp"
)

func TestValidateExample1(t *testing.T) {
	in := Input{
		Name:        " Morph Studio ",
		Slug:        " morph-studio ",
		Description: "\tThe backend\n",
		Status:      "",
	}
	got, errs := Validate(in)
	testhelp.Equal(t, "input", got, Input{
		Name:        "Morph Studio",
		Slug:        "morph-studio",
		Description: "The backend",
		Status:      "draft",
	})
	testhelp.Equal(t, "errors", errs, map[string]string(nil))
}

func TestValidateExample2(t *testing.T) {
	got, errs := Validate(Input{})
	testhelp.Equal(t, "input", got, Input{Status: "draft"})
	testhelp.Equal(t, "errors", errs, map[string]string{
		"name": "required",
		"slug": "required",
	})
}

func TestValidateExample3(t *testing.T) {
	in := Input{
		Name:        " " + strings.Repeat("é", 64) + " ",
		Slug:        strings.Repeat("a", 40),
		Description: strings.Repeat("ж", 500),
		Status:      "archived",
	}
	got, errs := Validate(in)
	testhelp.Equal(t, "input", got, Input{
		Name:        strings.Repeat("é", 64),
		Slug:        strings.Repeat("a", 40),
		Description: strings.Repeat("ж", 500),
		Status:      "archived",
	})
	testhelp.Equal(t, "errors", errs, map[string]string(nil))
}

func TestValidateExample4(t *testing.T) {
	in := Input{
		Name:        strings.Repeat("é", 65),
		Slug:        strings.Repeat("a", 41),
		Description: strings.Repeat("ж", 501),
		Status:      "Active",
	}
	got, errs := Validate(in)
	testhelp.Equal(t, "input", got, in)
	testhelp.Equal(t, "errors", errs, map[string]string{
		"name":        "too long",
		"slug":        "too long",
		"description": "too long",
		"status":      "invalid",
	})
}

func TestValidateExample5(t *testing.T) {
	cases := []struct {
		slug string
		errs map[string]string
	}{
		{"a1-b2-c3", nil},
		{"0", nil},
		{"Morph", map[string]string{"slug": "invalid"}},
		{"morph--studio", map[string]string{"slug": "invalid"}},
		{"-morph", map[string]string{"slug": "invalid"}},
		{"morph-", map[string]string{"slug": "invalid"}},
		{"morph_studio", map[string]string{"slug": "invalid"}},
		{"morph studio", map[string]string{"slug": "invalid"}},
		{strings.Repeat("A", 41), map[string]string{"slug": "too long"}},
	}
	for _, c := range cases {
		_, errs := Validate(Input{Name: "x", Slug: c.slug, Status: "active"})
		testhelp.Equal(t, "slug "+c.slug, errs, c.errs)
	}
}

func TestValidateExample6(t *testing.T) {
	got, errs := Validate(Input{
		Name:        " ",
		Slug:        "ok",
		Description: " ",
		Status:      " active ",
	})
	testhelp.Equal(t, "input", got, Input{
		Name:        "",
		Slug:        "ok",
		Description: "",
		Status:      " active ",
	})
	testhelp.Equal(t, "errors", errs, map[string]string{
		"name":   "required",
		"status": "invalid",
	})
}
