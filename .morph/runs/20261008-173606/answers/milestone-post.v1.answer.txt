package telegram

import (
	"context"
	"io"
	"net/http"
	"net/url"
	"strings"
)

var Icons = map[string]string{
	"start":    "🚀",
	"gate":     "🚦",
	"run":      "🏁",
	"fail":     "❌",
	"merge":    "🔀",
	"smoke":    "🧪",
	"stop":     "🛑",
	"debt":     "💸",
	"end":      "🎉",
	"info":     "💬",
	"watchdog": "🐕",
	"idle":     "💤",
	"ask":      "🔔",
}

type Status struct {
	Sent       bool `json:"sent"`
	Skipped    bool `json:"skipped"`
	HTTPStatus int  `json:"http_status"`
}

type Client struct {
	Token  string
	ChatID string
	Do     func(*http.Request) (*http.Response, error)
}

func Escape(s string) string {
	s = strings.ReplaceAll(s, "&", "&amp;")
	s = strings.ReplaceAll(s, "<", "&lt;")
	s = strings.ReplaceAll(s, ">", "&gt;")
	return s
}

func firstRunes(s string, n int) string {
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n])
}

func Text(project, kind, headline, numbers string) string {
	icon, ok := Icons[kind]
	if !ok {
		icon = "💬"
	}
	out := "[" + project + "] " + icon + " <b>" + Escape(firstRunes(headline, 300)) + "</b>"
	if numbers != "" {
		out += "\n" + Escape(firstRunes(numbers, 3500))
	}
	return out
}

func (c Client) Post(ctx context.Context, project, kind, headline, numbers string) (Status, error) {
	if c.Token == "" || c.ChatID == "" {
		return Status{Skipped: true}, nil
	}
	form := url.Values{
		"chat_id":                  {c.ChatID},
		"text":                     {Text(project, kind, headline, numbers)},
		"parse_mode":               {"HTML"},
		"disable_web_page_preview": {"true"},
	}
	endpoint := "https://api.telegram.org/bot" + c.Token + "/sendMessage"
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, endpoint, strings.NewReader(form.Encode()))
	if err != nil {
		return Status{}, err
	}
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	resp, err := c.Do(req)
	if err != nil {
		return Status{}, errFrom("telegram: post failed: "+err.Error(), c.Token)
	}
	defer resp.Body.Close()
	body, _ := io.ReadAll(resp.Body)
	if resp.StatusCode == 200 {
		return Status{Sent: true, HTTPStatus: 200}, nil
	}
	return Status{HTTPStatus: resp.StatusCode}, errFrom("telegram: status "+itoa(resp.StatusCode)+": "+firstBytes(body, 200), c.Token)
}

func firstBytes(b []byte, n int) string {
	if len(b) > n {
		b = b[:n]
	}
	return string(b)
}

func itoa(n int) string {
	if n == 0 {
		return "0"
	}
	neg := n < 0
	if neg {
		n = -n
	}
	var buf [20]byte
	i := len(buf)
	for n > 0 {
		i--
		buf[i] = byte('0' + n%10)
		n /= 10
	}
	if neg {
		i--
		buf[i] = '-'
	}
	return string(buf[i:])
}

func errFrom(msg, token string) error {
	if token != "" {
		msg = strings.ReplaceAll(msg, token, "***")
	}
	return &postError{msg}
}

type postError struct{ s string }

func (e *postError) Error() string { return e.s }
