// scKeys returns the runner.Key of each recorded call in order.
func scKeys(calls []runner.Call) []string {
	keys := make([]string, 0, len(calls))
	for _, c := range calls {
		keys = append(keys, runner.Key(c.Name, c.Args...))
	}
	return keys
}

// scDirs returns the Dir of each recorded call in order.
func scDirs(calls []runner.Call) []string {
	dirs := make([]string, 0, len(calls))
	for _, c := range calls {
		dirs = append(dirs, c.Dir)
	}
	return dirs
}

// scWantDirs returns n copies of dir for comparing against scDirs.
func scWantDirs(n int, dir string) []string {
	out := make([]string, n)
	for i := range out {
		out[i] = dir
	}
	return out
}

func TestPhaseStartCheckExample1(t *testing.T) {
	f := &runner.Fake{Script: map[string]runner.Result{
		"git rev-parse --abbrev-ref HEAD": {Stdout: "main\n"},
		"git rev-parse HEAD":              {Stdout: "aaa111\n"},
		"git rev-parse origin/main":       {Stdout: "aaa111\n"},
		"git status --porcelain":          {Stdout: ""},
	}}
	got := StartCheck(context.Background(), f, "/p")
	testhelp.Equal(t, "StartCheck", got, Start{Mode: "fresh", Branch: "main", Local: "aaa111", Remote: "aaa111", Reason: ""})
	testhelp.Equal(t, "call keys", scKeys(f.Calls), []string{
		"git fetch -q origin",
		"git rev-parse --abbrev-ref HEAD",
		"git rev-parse HEAD",
		"git rev-parse origin/main",
		"git status --porcelain",
	})
	testhelp.Equal(t, "call dirs", scDirs(f.Calls), scWantDirs(5, "/p"))
}
