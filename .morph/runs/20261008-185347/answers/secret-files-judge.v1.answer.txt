package registry

import (
	"os"
	"path/filepath"
	"testing"

	"morphstudio/internal/testhelp"
)

func sfDemo(t *testing.T) (*Registry, string) {
	t.Helper()
	base := filepath.Join(t.TempDir(), "state")
	r, err := Open(base)
	if err != nil {
		t.Fatal(err)
	}
	if err := r.Add(Project{Name: "demo"}); err != nil {
		t.Fatal(err)
	}
	return r, base
}

func sfDemoWithToken(t *testing.T) (*Registry, string) {
	t.Helper()
	r, base := sfDemo(t)
	if err := r.PutSecret("demo", "github-token", "ghp_abc"); err != nil {
		t.Fatal(err)
	}
	return r, base
}

func sfErrText(err error) string {
	if err == nil {
		return ""
	}
	return err.Error()
}

func sfNames(t *testing.T, dir string) []string {
	t.Helper()
	entries, err := os.ReadDir(dir)
	if err != nil {
		t.Fatal(err)
	}
	names := []string{}
	for _, e := range entries {
		names = append(names, e.Name())
	}
	return names
}

func TestSecretFilesExample1(t *testing.T) {
	r, base := sfDemo(t)

	err := r.PutSecret("demo", "github-token", "ghp_abc")
	testhelp.Equal(t, "PutSecret", err, nil)

	testhelp.Equal(t, "HasSecret", r.HasSecret("demo", "github-token"), true)

	got, err := r.ReadSecret("demo", "github-token")
	testhelp.Equal(t, "ReadSecret", got, "ghp_abc")
	testhelp.Equal(t, "ReadSecret err", err, nil)

	info, err := os.Stat(r.SecretPath("demo", "github-token"))
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "secret mode", info.Mode()&0o777, os.FileMode(0o600))

	dirInfo, err := os.Stat(filepath.Join(base, "demo"))
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "project dir mode", dirInfo.Mode()&0o777, os.FileMode(0o700))

	testhelp.Equal(t, "project dir holds", sfNames(t, filepath.Join(base, "demo")), []string{"github-token"})
}

func TestSecretFilesExample2(t *testing.T) {
	r, _ := sfDemoWithToken(t)

	v := "https://x-access-token:ghp_abc@github.com\n"
	err := r.PutSecret("demo", "git-credentials", v)
	testhelp.Equal(t, "PutSecret", err, nil)

	got, err := r.ReadSecret("demo", "git-credentials")
	testhelp.Equal(t, "ReadSecret", got, v)
	testhelp.Equal(t, "ReadSecret err", err, nil)
}

func TestSecretFilesExample3(t *testing.T) {
	r, base := sfDemoWithToken(t)

	err := r.PutSecret("nope", "github-token", "x")
	testhelp.Equal(t, "PutSecret unknown project", err, ErrNotFound)

	err = r.PutSecret("demo", "password", "x")
	testhelp.Equal(t, "PutSecret unknown kind", sfErrText(err), `registry: unknown secret kind "password"`)

	err = r.PutSecret("demo", "session-token", "")
	testhelp.Equal(t, "PutSecret empty value", err, ErrEmptySecret)

	testhelp.Equal(t, "project dir holds", sfNames(t, filepath.Join(base, "demo")), []string{"github-token"})
}

func TestSecretFilesExample4(t *testing.T) {
	r, _ := sfDemoWithToken(t)

	path := r.SecretPath("demo", "github-token")
	if err := os.Chmod(path, 0o644); err != nil {
		t.Fatal(err)
	}

	err := r.PutSecret("demo", "github-token", "ghp_new")
	testhelp.Equal(t, "PutSecret", err, nil)

	info, err := os.Stat(path)
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "secret mode", info.Mode()&0o777, os.FileMode(0o600))

	got, err := r.ReadSecret("demo", "github-token")
	testhelp.Equal(t, "ReadSecret", got, "ghp_new")
	testhelp.Equal(t, "ReadSecret err", err, nil)
}

func TestSecretFilesExample5(t *testing.T) {
	r, _ := sfDemoWithToken(t)

	testhelp.Equal(t, "HasSecret session-token", r.HasSecret("demo", "session-token"), false)

	got, err := r.ReadSecret("demo", "session-token")
	testhelp.Equal(t, "ReadSecret session-token", got, "")
	testhelp.Equal(t, "ReadSecret session-token err", err, ErrNoSecret)

	got, err = r.ReadSecret("nope", "github-token")
	testhelp.Equal(t, "ReadSecret unknown project", got, "")
	testhelp.Equal(t, "ReadSecret unknown project err", err, ErrNotFound)

	err = r.DeleteSecret("demo", "session-token")
	testhelp.Equal(t, "DeleteSecret missing", err, nil)
}

func TestSecretFilesExample6(t *testing.T) {
	r, _ := sfDemoWithToken(t)

	if err := r.PutSecret("demo", "session-token", "tok-9"); err != nil {
		t.Fatal(err)
	}

	err := r.DeleteSecret("demo", "session-token")
	testhelp.Equal(t, "DeleteSecret", err, nil)

	testhelp.Equal(t, "HasSecret session-token", r.HasSecret("demo", "session-token"), false)
}
