// ptConnect opens the in-memory transports and connects a client to s.
func ptConnect(t *testing.T, s *mcp.Server) *mcp.ClientSession {
	t.Helper()
	ctx := context.Background()
	ct, st := mcp.NewInMemoryTransports()
	ss, err := s.Connect(ctx, st, nil)
	if err != nil {
		t.Fatalf("server connect: %v", err)
	}
	cs, err := mcp.NewClient(&mcp.Implementation{Name: "t", Version: "0"}, nil).Connect(ctx, ct, nil)
	if err != nil {
		t.Fatalf("client connect: %v", err)
	}
	t.Cleanup(func() {
		cs.Close()
		ss.Wait()
	})
	return cs
}

// ptNames returns the tool names of a ListTools result.
func ptNames(r *mcp.ListToolsResult) []string {
	names := make([]string, 0, len(r.Tools))
	for _, tool := range r.Tools {
		names = append(names, tool.Name)
	}
	return names
}

// ptText returns the text of the first content of a CallTool result, or "" when there is none.
func ptText(res *mcp.CallToolResult) string {
	if res == nil || len(res.Content) == 0 {
		return ""
	}
	tc, ok := res.Content[0].(*mcp.TextContent)
	if !ok {
		return ""
	}
	return tc.Text
}

// TestPMToolsExample1 lists the per-project tools the PM server exposes.
func TestPMToolsExample1(t *testing.T) {
	ctx := context.Background()

		"plan_load", "restart", "status", "stop_check", "usage",
	})
}

// TestPMToolsExample3 sends an order, answers a question and loads the approved plan.
func TestPMToolsExample3(t *testing.T) {
	ctx := context.Background()
	c := &ptFake{}
	c.order = session.OrderResult{Queued: 1}
	c.planLoad = control.QueueView{State: "queued", Approved: "7b31dfe", Index: 0, Phases: 3, Current: "P17"}
	s := PMServer(c, "demo")
	cs := ptConnect(t, s)

	res, err := cs.CallTool(ctx, &mcp.CallToolParams{
		Name:      "order",
		Arguments: map[string]any{"text": "smoke checked green. Resume with P12b"},
	})
	if err != nil {
		t.Fatalf("CallTool order: %v", err)
	}
	testhelp.Equal(t, "order text", ptText(res), `{"queued":1}`)
	testhelp.Equal(t, "calls after order", c.calls, []ptCall{
		{Method: "Order", Args: []any{"demo", "smoke checked green. Resume with P12b"}},
	})

	res, err = cs.CallTool(ctx, &mcp.CallToolParams{Name: "answer", Arguments: map[string]any{"option": 1}})
	if err != nil {
		t.Fatalf("CallTool answer: %v", err)
	}
	testhelp.Equal(t, "answer text", ptText(res), `{"ok":true}`)
	testhelp.Equal(t, "calls after answer", c.calls, []ptCall{
		{Method: "Order", Args: []any{"demo", "smoke checked green. Resume with P12b"}},
		{Method: "Answer", Args: []any{"demo", 1}},
	})

	raw, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatalf("read plan: %v", err)
	}
	var args map[string]any
	if err := json.Unmarshal(raw, &args); err != nil {
		t.Fatalf("unmarshal plan: %v", err)
	}
	res, err = cs.CallTool(ctx, &mcp.CallToolParams{Name: "plan_load", Arguments: args})
	if err != nil {
		t.Fatalf("CallTool plan_load: %v", err)
	}
	testhelp.Equal(t, "plan_load text", ptText(res), `{"state":"queued","approved":"7b31dfe","index":0,"phases":3,"current":"P17","reason":""}`)
	testhelp.Equal(t, "calls after plan_load", c.calls, []ptCall{
		{Method: "Order", Args: []any{"demo", "smoke checked green. Resume with P12b"}},
		{Method: "Answer", Args: []any{"demo", 1}},
		{Method: "PlanLoad", Args: []any{"demo", queue.Plan{
			Approved: "7b31dfe",
			Phases: []queue.Phase{
				{ID: "P17", StopAfter: "none"},
				{ID: "P18", StopAfter: "smoke", Caps: queue.Caps{ClaudeUSD: 12}},
				{ID: "P19", StopAfter: "none"},
			},
		}}},
	})
}

		{Method: "Status", Args: []any{"beta"}},
	})
}

// TestPMToolsExample7 rejects a wrongly typed option before touching Control.
func TestPMToolsExample7(t *testing.T) {
	ctx := context.Background()
	c := &ptFake{}
	s := PMServer(c, "demo")
	cs := ptConnect(t, s)

	res, err := cs.CallTool(ctx, &mcp.CallToolParams{Name: "answer", Arguments: map[string]any{"option": "one"}})
	if err == nil && (res == nil || !res.IsError) {
		t.Fatalf("answer with a string option: err=%v res=%v", err, res)
	}
	testhelp.Equal(t, "calls after bad option", c.calls, []ptCall(nil))

	res, err = cs.CallTool(ctx, &mcp.CallToolParams{Name: "answer", Arguments: map[string]any{"option": 2}})
	if err != nil {
		t.Fatalf("CallTool answer: %v", err)
	}
	testhelp.Equal(t, "answer IsError", res.IsError, false)
	testhelp.Equal(t, "answer text", ptText(res), `{"ok":true}`)
	testhelp.Equal(t, "calls after answer", c.calls, []ptCall{{Method: "Answer", Args: []any{"demo", 2}}})
}
