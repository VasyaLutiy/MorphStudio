type dcHarness struct {
	t   *testing.T
	tmp string

	reg *registry.Registry
	run *runner.Fake

	mu       sync.Mutex
	launches []claude.Launch
	spawns   []*claude.Fake
	posts    [][4]string
	clock    time.Time
	nextID   int
	do       func(*http.Request) (*http.Response, error)

func dcNew(t *testing.T) *dcHarness {
	t.Helper()
	tmp := t.TempDir()
	reg, err := registry.Open(filepath.Join(tmp, "state"))
	if err != nil {
		t.Fatal(err)
	}
	h := &dcHarness{
		t:      t,
		tmp:    tmp,
		reg:    reg,
		clock:  time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC),
		nextID: 1,
	}
	h.do = func(r *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: 200,
			Body:       io.NopCloser(strings.NewReader(`{"permissions":{"push":true}}`)),
			Header:     http.Header{},
		}, nil
	}
	h.run = &runner.Fake{Script: map[string]runner.Result{
		"git rev-parse --abbrev-ref HEAD": {Stdout: "main\n"},
		"git rev-parse HEAD":              {Stdout: "aaa111\n"},
		"git rev-parse origin/main":       {Stdout: "aaa111\n"},
		"git status --porcelain":          {Stdout: ""},
	}}
	h.addProject("demo", "go", "https://github.com/acme/demo", queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5})
	return h
}

func (h *dcHarness) addProject(name, lang, repo string, caps queue.Caps) {
	h.t.Helper()
	dir := filepath.Join(h.tmp, "projects", name)
	if err := h.reg.Add(registry.Project{
		Name:      name,
		Language:  lang,
		RepoURL:   repo,
		Dir:       dir,
		Caps:      caps,
		CreatedAt: time.Date(2026, 10, 8, 14, 0, 0, 0, time.UTC),
	}); err != nil {
		h.t.Fatal(err)
	}
	b, err := os.ReadFile("../tests/fixtures/git/measure.md")
	if err != nil {
		h.t.Fatal(err)
	}
	docs := filepath.Join(dir, "docs")
	if err := os.MkdirAll(docs, 0o755); err != nil {
		h.t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(docs, "MEASURE.md"), b, 0o644); err != nil {
		h.t.Fatal(err)
	}
}

func (h *dcHarness) open(ctx context.Context) *Daemon {
	h.t.Helper()
	deps := Deps{
		Registry: h.reg,
		Runner:   h.run,
		Spawn: func(ctx context.Context, l claude.Launch) (claude.Process, error) {
			h.mu.Lock()
			defer h.mu.Unlock()
			h.launches = append(h.launches, l)
			f := claude.NewFake()
			h.spawns = append(h.spawns, f)
			return f, nil
		},
		Post: func(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
			h.mu.Lock()
			defer h.mu.Unlock()
			h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
			return telegram.Status{}, nil
		},
		GitHubDo: func(r *http.Request) (*http.Response, error) {
			h.mu.Lock()
			do := h.do
			h.mu.Unlock()
			return do(r)
		},
		Now: func() time.Time {
			h.mu.Lock()
			defer h.mu.Unlock()
			now := h.clock
			h.clock = h.clock.Add(time.Second)
			return now
		},
		NewID: func() string {
			h.mu.Lock()
			defer h.mu.Unlock()
			id := fmt.Sprintf("id-%d", h.nextID)
			h.nextID++
			return id
		},
		Config: Config{
			ClaudeBin:   "/opt/claude/bin/claude",
			MorphBin:    "/usr/local/bin/morph",
			ProjectsDir: filepath.Join(h.tmp, "projects"),
			MCPBaseURL:  "http://127.0.0.1:7181",
			Defaults:    queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4},
			MaxParallel: 1,
			Loop: supervisor.Config{
				MaxResumesPerHour: 3,
				StallMinutes:      30,
				StretchUSD:        30,
				UsageAlertPercent: 50,
			},
		},
	}
	d, err := Open(ctx, deps)
	if err != nil {
		h.t.Fatal(err)
	}
	return d
}

func (h *dcHarness) plan() queue.Plan {
	h.t.Helper()
	b, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		h.t.Fatal(err)
	}
	var p queue.Plan
	if err := json.Unmarshal(b, &p); err != nil {
		h.t.Fatal(err)
	}
	return p
}

func (h *dcHarness) launchCount() int {
	h.mu.Lock()
	defer h.mu.Unlock()
	return len(h.launches)
}

func (h *dcHarness) launch(i int) claude.Launch {
	h.mu.Lock()
	defer h.mu.Unlock()
	return h.launches[i]
}

func (h *dcHarness) proc(i int) *claude.Fake {
	h.mu.Lock()
	defer h.mu.Unlock()
	return h.spawns[i]
}

func (h *dcHarness) postList() [][4]string {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([][4]string, len(h.posts))
	copy(out, h.posts)
	return out
}

func dcSetupExample2(h *dcHarness, d *Daemon) *claude.Fake {
	h.t.Helper()
	if _, err := d.PlanLoad("demo", h.plan()); err != nil {
		h.t.Fatal(err)
	}
	if h.launchCount() != 1 {
		h.t.Fatalf("launches = %d, want 1", h.launchCount())
	}
	return h.proc(0)
}

func TestDaemonCoreExample6(t *testing.T) {

func TestDaemonCoreExample8(t *testing.T) {
	h := dcNew(t)
	h.run.Script["git rev-parse --verify -q origin/main"] = runner.Result{Code: 1}
	h.run.Script["git push -q -u origin main"] = runner.Result{Code: 0}
	d := h.open(context.Background())
	ctx := context.Background()

	res, err := d.PutGithubToken(ctx, "demo", "ghp_abc")
	if err != nil {
		t.Fatal(err)
	}
	if res.Access.CheckedAt.IsZero() {
		t.Error("CheckedAt is zero")
	}
	acc := res.Access
	acc.CheckedAt = time.Time{}
	testhelp.Equal(t, "access", acc, github.Access{Checked: true, Reachable: true, Push: true})
	testhelp.Equal(t, "pushed", res.Pushed, true)
	testhelp.Equal(t, "push error", res.PushError, "")

	tok, err := h.reg.ReadSecret("demo", "github-token")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "github-token", tok, "ghp_abc")

	cred, err := h.reg.ReadSecret("demo", "git-credentials")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "git-credentials", cred, "https://x-access-token:ghp_abc@github.com\n")

	for _, kind := range []string{"github-token", "git-credentials"} {
		fi, err := os.Stat(h.reg.SecretPath("demo", kind))
		if err != nil {
			t.Fatal(err)
		}
		testhelp.Equal(t, "mode "+kind, fi.Mode().Perm(), os.FileMode(0o600))
	}

	st, err := d.Status("demo")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "status repo", st.Repo, res.Access)

	testhelp.Equal(t, "runner calls", h.run.Calls, []runner.Call{
		{Dir: filepath.Join(h.tmp, "projects", "demo"), Name: "git", Args: []string{"rev-parse", "--verify", "-q", "origin/main"}},
		{Dir: filepath.Join(h.tmp, "projects", "demo"), Name: "git", Args: []string{"push", "-q", "-u", "origin", "main"}},
	})

	h.mu.Lock()
	h.do = func(r *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: 401,
			Body:       io.NopCloser(strings.NewReader(`{"message":"Bad credentials"}`)),
			Header:     http.Header{},
		}, nil
	}
	h.mu.Unlock()

	res2, err := d.PutGithubToken(ctx, "demo", "ghp_abc")
	if err != nil {
		t.Fatal(err)
	}
	acc2 := res2.Access
	acc2.CheckedAt = time.Time{}
	testhelp.Equal(t, "access 401", acc2, github.Access{Checked: true, Reason: "401 bad credentials"})
	testhelp.Equal(t, "pushed 401", res2.Pushed, false)
	testhelp.Equal(t, "still two calls", len(h.run.Calls), 2)

	if _, err := d.PutGithubToken(ctx, "demo", ""); !errors.Is(err, control.ErrBadInput) {
		t.Errorf("empty token err = %v", err)
	}
	if _, err := d.PutGithubToken(ctx, "nope", "x"); !errors.Is(err, control.ErrUnknownProject) {
		t.Errorf("unknown project err = %v", err)
	}
}

	ctx := context.Background()

	if _, err := d.CreateProject(ctx, "demo", "go", "https://github.com/acme/demo"); !errors.Is(err, control.ErrExists) {
		t.Errorf("CreateProject demo err = %v", err)
	}
	testhelp.Equal(t, "no runner call for existing", len(h.run.Calls), 0)

	gv, err := d.CreateProject(ctx, "gamma", "python", "https://github.com/acme/gamma")
	if err != nil {
		t.Fatal(err)
	}
	if gv.CreatedAt.IsZero() {
		t.Error("gamma CreatedAt is zero")
	}
	gv.CreatedAt = time.Time{}
	testhelp.Equal(t, "gamma view", gv, control.ProjectView{
		Name:     "gamma",
		Language: "python",
		RepoURL:  "https://github.com/acme/gamma",
		State:    "idle",
	})

	gp, err := h.reg.Get("gamma")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "gamma dir", gp.Dir, filepath.Join(h.tmp, "projects", "gamma"))
	testhelp.Equal(t, "gamma caps", gp.Caps, queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4})

	testhelp.Equal(t, "first runner call", h.run.Calls[0], runner.Call{
		Dir:  filepath.Join(h.tmp, "projects"),
		Name: "/usr/local/bin/morph",
		Args: []string{
			"init",
			"--root", filepath.Join(h.tmp, "projects", "gamma"),
			"--name", "gamma",
			"--language", "python",
		},
	})

	ps := d.Projects()
	names := make([]string, 0, len(ps))
	states := make([]string, 0, len(ps))
	for _, p := range ps {
		names = append(names, p.Name)
		states = append(states, p.State)
	}
	testhelp.Equal(t, "names", names, []string{"demo", "gamma"})
	testhelp.Equal(t, "states", states, []string{"idle", "idle"})

	_, err = d.CreateProject(ctx, "delta", "cobol", "https://github.com/acme/delta")
	if err == nil {
		t.Fatal("CreateProject delta: expected error")
	}
	if !errors.Is(err, control.ErrBadInput) {
		t.Errorf("delta err = %v", err)
	}
	testhelp.Equal(t, "delta text", err.Error(), `bad input: bootstrap: unknown language "cobol"`)
}
