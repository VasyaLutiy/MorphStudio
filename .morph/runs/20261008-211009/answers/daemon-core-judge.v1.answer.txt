Looking at the examples, I need to build a test file with a `dcHarness` helper and 10 test functions. Let me trace through each example carefully to get the exact call sequences and timestamp consumption.

Key observations:
- `Open` doesn't consume `Now` for idle projects
- `PlanLoad` consumes `Now` (queuedAt), then `Begin` consumes another (arg evaluation), then `StartChecked` consumes another
- SessionID and token IDs alternate through `NewID`

func (h *dcHarness) open() *Daemon {
	h.t.Helper()
	deps := Deps{
		Registry: h.reg,
		Runner:   h.run,
		Spawn: func(ctx context.Context, l claude.Launch) (claude.Process, error) {
			h.mu.Lock()
			defer h.mu.Unlock()
			h.launches = append(h.launches, l)
			f := claude.NewFake()
			h.spawns = append(h.spawns, f)
			return f, nil
		},
		Post: func(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
			h.mu.Lock()
			defer h.mu.Unlock()
			h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
			return telegram.Status{}, nil
		},
		GitHubDo: func(r *http.Request) (*http.Response, error) {
			return h.doFn(r)
		},
		Now: func() time.Time {
			h.mu.Lock()
			defer h.mu.Unlock()
			now := h.clock
			h.clock = h.clock.Add(time.Second)
			return now
		},
		NewID: func() string {
			h.mu.Lock()
			defer h.mu.Unlock()
			id := fmt.Sprintf("id-%d", h.nextID)
			h.nextID++
			return id
		},
		Config: Config{
			ClaudeBin:   "/opt/claude/bin/claude",
			MorphBin:    "/usr/local/bin/morph",
			ProjectsDir: h.projDir,
			MCPBaseURL:  "http://127.0.0.1:7181",
			Defaults:    queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4},
			MaxParallel: 1,
			Loop: supervisor.Config{
				MaxResumesPerHour: 3,
				StallMinutes:      30,
				StretchUSD:        30,
				UsageAlertPercent: 50,
			},
		},
	}
	d, err := Open(context.Background(), deps)
	if err != nil {
		h.t.Fatal(err)
	}
	return d
}

func dcPlan(t *testing.T) queue.Plan {

	}
	var p queue.Plan
	if err := json.Unmarshal(b, &p); err != nil {
		t.Fatal(err)
	}
	return p
}

func TestDaemonCoreExample8(t *testing.T) {
	h := dcNew(t)
	h.mu.Lock()
	h.run.Script["git rev-parse --verify -q origin/main"] = runner.Result{Code: 1}
	h.run.Script["git push -q -u origin main"] = runner.Result{Code: 0}
	h.mu.Unlock()
	h.doFn = func(r *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: 200,
			Body:       io.NopCloser(strings.NewReader(`{"permissions":{"push":true}}`)),
			Header:     http.Header{},
		}, nil
	}
	d := h.open()
	ctx := context.Background()

	res, err := d.PutGithubToken(ctx, "demo", "ghp_abc")
	if err != nil {
		t.Fatalf("PutGithubToken 200: %v", err)
	}
	testhelp.Equal(t, "Access checked", res.Access.Checked, true)
	testhelp.Equal(t, "Access reachable", res.Access.Reachable, true)
	testhelp.Equal(t, "Access push", res.Access.Push, true)
	testhelp.Equal(t, "Access reason", res.Access.Reason, "")
	if res.Access.CheckedAt.IsZero() {
		t.Error("Access CheckedAt is zero")
	}
	testhelp.Equal(t, "TokenResult pushed", res.Pushed, true)
	testhelp.Equal(t, "TokenResult push error", res.PushError, "")

	tok, err := h.reg.ReadSecret("demo", "github-token")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "github token", tok, "ghp_abc")

	cred, err := h.reg.ReadSecret("demo", "git-credentials")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "git credentials", cred, "https://x-access-token:ghp_abc@github.com\n")

	for _, kind := range []string{"github-token", "git-credentials"} {
		fi, err := os.Stat(h.reg.SecretPath("demo", kind))
		if err != nil {
			t.Fatalf("stat %s: %v", kind, err)
		}
		testhelp.Equal(t, "mode "+kind, fi.Mode().Perm(), os.FileMode(0o600))
	}

	st, err := d.Status("demo")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "Status repo", st.Repo, res.Access)

	calls := h.run.Calls
	if len(calls) != 2 {
		t.Fatalf("runner calls: %d", len(calls))
	}
	testhelp.Equal(t, "runner call 0", calls[0], runner.Call{
		Dir:  filepath.Join(h.tmp, "projects", "demo"),
		Name: "git",
		Args: []string{"rev-parse", "--verify", "-q", "origin/main"},
	})
	testhelp.Equal(t, "runner call 1", calls[1], runner.Call{
		Dir:  filepath.Join(h.tmp, "projects", "demo"),
		Name: "git",
		Args: []string{"push", "-q", "-u", "origin", "main"},
	})

	h.doFn = func(r *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: 401,
			Body:       io.NopCloser(strings.NewReader(`{"message":"Bad credentials"}`)),
			Header:     http.Header{},
		}, nil
	}

	res2, err := d.PutGithubToken(ctx, "demo", "ghp_abc")
	if err != nil {
		t.Fatalf("PutGithubToken 401: %v", err)
	}
	testhelp.Equal(t, "Access checked 2", res2.Access.Checked, true)
	testhelp.Equal(t, "Access reachable 2", res2.Access.Reachable, false)
	testhelp.Equal(t, "Access push 2", res2.Access.Push, false)
	testhelp.Equal(t, "Access reason 2", res2.Access.Reason, "401 bad credentials")
	if res2.Access.CheckedAt.IsZero() {
		t.Error("Access CheckedAt 2 is zero")
	}
	testhelp.Equal(t, "TokenResult pushed 2", res2.Pushed, false)

	if len(h.run.Calls) != 2 {
		t.Errorf("runner calls after 401: %d", len(h.run.Calls))
	}

	if _, err := d.PutGithubToken(ctx, "demo", ""); !errors.Is(err, control.ErrBadInput) {
		t.Errorf("PutGithubToken empty err = %v", err)
	}
	if _, err := d.PutGithubToken(ctx, "nope", "x"); !errors.Is(err, control.ErrUnknownProject) {
		t.Errorf("PutGithubToken unknown err = %v", err)
	}
}
