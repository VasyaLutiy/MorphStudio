func TestConfigAndMainExample4(t *testing.T) {
	cases := []struct {
		name string
		env  map[string]string
		want string
	}{
		{
			name: "no token",
			env:  map[string]string{"HOME": "/h"},
			want: "config: MORPHD_TOKEN is required",
		},
		{
			name: "bad port",
			env:  map[string]string{"HOME": "/h", "MORPHD_TOKEN": "x", "MORPHD_PORT": "80a"},
			want: "config: MORPHD_PORT: not a number: 80a",
		},
		{
			name: "bad stall",
			env:  map[string]string{"HOME": "/h", "MORPHD_TOKEN": "x", "MORPHD_STALL_MINUTES": "soon"},
			want: "config: MORPHD_STALL_MINUTES: not a number: soon",
		},
	}
	for _, tc := range cases {
		env := tc.env
		getenv := func(k string) string { return env[k] }
		_, err := Load(getenv, map[string]string{})
		if err == nil {
			t.Fatalf("%s: Load returned nil error", tc.name)
		}
		testhelp.Equal(t, tc.name+" error", err.Error(), tc.want)
	}
}

func cmServe(h http.Handler, req *http.Request) *httptest.ResponseRecorder {
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	return rec

func TestConfigAndMainExample5(t *testing.T) {
	f := &cmFake{status: control.Status{
		State:     "busy",
		Phase:     "P18",
		Minutes:   37,
		CostUSD:   1.92,
		FiveHour:  22,
		SevenDay:  60,
		SessionID: "s-1",
		Queue: control.QueueView{
			State:    "running",
			Approved: "7b31dfe",
			Index:    1,
			Phases:   3,
			Current:  "P18",
		},
		Repo: github.Access{
			Checked:   true,
			Reachable: true,
			Push:      true,
			CheckedAt: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC),
		},
	}}
	cmKnown := func(p string) bool { return p == "demo" }
	cmSession := func(p string) (string, bool) { return "s", p == "demo" }
	h := Build(f, cmKnown, cmSession, "tok-7")

	rec := cmServe(h, httptest.NewRequest("GET", "/healthz", nil))
	testhelp.Equal(t, "healthz code", rec.Code, 200)
	testhelp.Equal(t, "healthz body", rec.Body.String(), `{"status":"ok"}`+"\n")

	req := httptest.NewRequest("GET", "/projects/demo/status", nil)
	req.Header.Set("Authorization", "Bearer tok-7")
	rec = cmServe(h, req)
	testhelp.Equal(t, "status code", rec.Code, 200)
	testhelp.Equal(t, "status body", rec.Body.String(), `{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}`+"\n")

	mcpReq := httptest.NewRequest("POST", "/mcp/demo", strings.NewReader(`{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}`))
	mcpReq.Header.Set("Authorization", "Bearer tok-7")
	mcpReq.Header.Set("Content-Type", "application/json")
	mcpReq.Header.Set("Accept", "application/json, text/event-stream")
	mcpRec := cmServe(h, mcpReq)
	testhelp.Equal(t, "mcp code", mcpRec.Code, 200)

	var mcpBody struct {
		Result struct {
			Tools []struct {
				Name string `json:"name"`
			} `json:"tools"`
		} `json:"result"`
	}
	if err := json.Unmarshal(mcpRec.Body.Bytes(), &mcpBody); err != nil {
		t.Fatalf("mcp body decode: %v", err)
	}
	testhelp.Equal(t, "mcp tools", len(mcpBody.Result.Tools), 10)
	found := false
	for _, tool := range mcpBody.Result.Tools {
		if tool.Name == "status" {
			found = true
		}
	}
	testhelp.Equal(t, "mcp status tool", found, true)

	rec = cmServe(h, httptest.NewRequest("GET", "/projects/demo/status", nil))
	testhelp.Equal(t, "unauthorized code", rec.Code, 401)
}

func TestConfigAndMainExample7(t *testing.T) {
	re := regexp.MustCompile(`^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$`)
	a := newID()
	b := newID()
	testhelp.Equal(t, "a matches", re.MatchString(a), true)
	testhelp.Equal(t, "b matches", re.MatchString(b), true)
	testhelp.Equal(t, "a len", len(a), 36)
	testhelp.Equal(t, "b len", len(b), 36)
	testhelp.Equal(t, "a != b", a != b, true)
	testhelp.Equal(t, "a lower", strings.ToLower(a) == a, true)
	testhelp.Equal(t, "b lower", strings.ToLower(b) == b, true)
}
