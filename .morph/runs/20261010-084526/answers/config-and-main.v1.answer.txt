package main

import (
	"fmt"
	"strconv"
	"strings"

	"morphstudio/queue"
)

// Config is the daemon's configuration, loaded from the environment and .env.
type Config struct {
	Token             string
	Port              int
	StateDir          string
	ProjectsDir       string
	ClaudeBin         string
	MorphBin          string
	TGToken           string
	TGChatID          string
	Model             string
	ExtraArgs         []string
	Defaults          queue.Caps
	StretchUSD        float64
	MaxParallel       int
	ResumesPerHour    int
	StallMinutes      int
	UsageAlertPercent int
}

// ParseDotenv parses .env text into a key value map. Empty lines and lines
// beginning with "#" are skipped, an "export " prefix is removed, the first
// "=" splits key and value, the value is trimmed and unquoted when wrapped in
// matching single or double quotes, and a line without "=" is skipped.
func ParseDotenv(text string) map[string]string {
	out := map[string]string{}
	for _, raw := range strings.Split(text, "\n") {
		line := strings.TrimSpace(raw)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		line = strings.TrimPrefix(line, "export ")
		eq := strings.Index(line, "=")
		if eq < 0 {
			continue
		}
		key := strings.TrimSpace(line[:eq])
		val := strings.TrimSpace(line[eq+1:])
		if len(val) >= 2 {
			if (val[0] == '"' && val[len(val)-1] == '"') || (val[0] == '\'' && val[len(val)-1] == '\'') {
				val = val[1 : len(val)-1]
			}
		}
		out[key] = val
	}
	return out
}

// Load reads each key as getenv(key), falling back to dotenv[key], then the
// default.
func Load(getenv func(string) string, dotenv map[string]string) (Config, error) {
	lookup := func(key string) string {
		if v := getenv(key); v != "" {
			return v
		}
		return dotenv[key]
	}
	mustInt := func(key string, def int) (int, error) {
		s := lookup(key)
		if s == "" {
			return def, nil
		}
		n, err := strconv.Atoi(s)
		if err != nil {
			return 0, fmt.Errorf("config: %s: not a number: %s", key, s)
		}
		return n, nil
	}
	mustFloat := func(key string, def float64) (float64, error) {
		s := lookup(key)
		if s == "" {
			return def, nil
		}
		f, err := strconv.ParseFloat(s, 64)
		if err != nil {
			return 0, fmt.Errorf("config: %s: not a number: %s", key, s)
		}
		return f, nil
	}

	var c Config

	c.Token = lookup("MORPHD_TOKEN")
	if c.Token == "" {
		return Config{}, fmt.Errorf("config: MORPHD_TOKEN is required")
	}

	home := getenv("HOME")

	c.StateDir = lookup("MORPHD_STATE_DIR")
	if c.StateDir == "" {
		c.StateDir = home + "/.local/state/morphd"
	}
	c.ProjectsDir = lookup("MORPHD_PROJECTS_DIR")
	if c.ProjectsDir == "" {
		c.ProjectsDir = home + "/projects"
	}
	c.ClaudeBin = lookup("MORPHD_CLAUDE_BIN")
	if c.ClaudeBin == "" {
		c.ClaudeBin = "claude"
	}
	c.MorphBin = lookup("MORPHD_MORPH_BIN")
	if c.MorphBin == "" {
		c.MorphBin = "morph"
	}
	c.TGToken = lookup("TG_BOT_TOKEN")
	c.TGChatID = lookup("TG_CHAT_ID")
	c.Model = lookup("MORPHD_MODEL")

	if extra := lookup("MORPHD_CLAUDE_EXTRA_ARGS"); extra != "" {
		c.ExtraArgs = strings.Fields(extra)
		if len(c.ExtraArgs) == 0 {
			c.ExtraArgs = nil
		}
	}

	var err error
	if c.Port, err = mustInt("MORPHD_PORT", 7080); err != nil {
		return Config{}, err
	}
	if c.Defaults.ClaudeUSD, err = mustFloat("MORPHD_CLAUDE_USD", 30); err != nil {
		return Config{}, err
	}
	if c.Defaults.Hours, err = mustFloat("MORPHD_HOURS", 3); err != nil {
		return Config{}, err
	}
	if c.Defaults.ExecutorUSD, err = mustFloat("MORPHD_EXECUTOR_USD", 5); err != nil {
		return Config{}, err
	}
	if c.StretchUSD, err = mustFloat("MORPHD_STRETCH_USD", 30); err != nil {
		return Config{}, err
	}
	if c.MaxParallel, err = mustInt("MORPHD_MAX_PARALLEL", 1); err != nil {
		return Config{}, err
	}
	if c.ResumesPerHour, err = mustInt("MORPHD_RESUMES_PER_HOUR", 3); err != nil {
		return Config{}, err
	}
	if c.StallMinutes, err = mustInt("MORPHD_STALL_MINUTES", 30); err != nil {
		return Config{}, err
	}
	if c.UsageAlertPercent, err = mustInt("MORPHD_USAGE_ALERT", 50); err != nil {
		return Config{}, err
	}

	return c, nil
}

// UUIDv4 writes the RFC 4122 version 4 form of b as lower-case hex in the
// groups 8-4-4-4-12 joined by "-".
func UUIDv4(b [16]byte) string {
	b[6] = 0x40 | (b[6] & 0x0f)
	b[8] = 0x80 | (b[8] & 0x3f)

	const hex = "0123456789abcdef"
	var s [36]byte
	j := 0
	for i := 0; i < 16; i++ {
		if i == 4 || i == 6 || i == 8 || i == 10 {
			s[j] = '-'
			j++
		}
		s[j] = hex[b[i]>>4]
		s[j+1] = hex[b[i]&0x0f]
		j += 2
	}
	return string(s[:])
}

package main

import (
	"context"
	"crypto/rand"
	"fmt"
	"net/http"
	"os"
	"time"

	"morphstudio/api"
	"morphstudio/claude"
	"morphstudio/control"
	"morphstudio/daemon"
	"morphstudio/mcpserver"
	"morphstudio/registry"
	"morphstudio/runner"
	"morphstudio/supervisor"
	"morphstudio/telegram"
)

// newID returns a version 4 UUID. A read error panics: the daemon cannot run
// without ids.
func newID() string {
	var b [16]byte
	if _, err := rand.Read(b[:]); err != nil {
		panic(err)
	}
	return UUIDv4(b)
}

// Build assembles the HTTP surface: the API behind its token and the MCP
// mount.
func Build(d control.Control, known func(string) bool, session func(string) (string, bool), token string) http.Handler {
	return api.NewRouter(
		api.Handlers{Control: d},
		token,
		mcpserver.Handler(d, known, mcpserver.Tokens{API: token, Session: session}),
	)
}

func main() {
	dotenv := map[string]string{}
	if data, err := os.ReadFile(".env"); err == nil {
		dotenv = ParseDotenv(string(data))
	}
	cfg, err := Load(os.Getenv, dotenv)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}

	reg, err := registry.Open(cfg.StateDir)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}

	ctx := context.Background()
	deps := daemon.Deps{
		Registry: reg,
		Runner:   runner.OS{},
		Spawn: func(ctx context.Context, l claude.Launch) (claude.Process, error) {
			return claude.Start(ctx, l.Bin, claude.Args(l), l.Dir, claude.Env(l))
		},
		Post:     telegram.Client{Token: cfg.TGToken, ChatID: cfg.TGChatID, Do: http.DefaultClient.Do}.Post,
		GitHubDo: http.DefaultClient.Do,
		Now:      time.Now,
		NewID:    newID,
		Config: daemon.Config{
			ClaudeBin:   cfg.ClaudeBin,
			MorphBin:    cfg.MorphBin,
			ProjectsDir: cfg.ProjectsDir,
			MCPBaseURL:  fmt.Sprintf("http://127.0.0.1:%d", cfg.Port),
			Model:       cfg.Model,
			ExtraArgs:   cfg.ExtraArgs,
			Defaults:    cfg.Defaults,
			MaxParallel: cfg.MaxParallel,
			Loop: supervisor.Config{
				MaxResumesPerHour: cfg.ResumesPerHour,
				StallMinutes:      cfg.StallMinutes,
				StretchUSD:        cfg.StretchUSD,
				UsageAlertPercent: cfg.UsageAlertPercent,
			},
		},
	}

	d, err := daemon.Open(ctx, deps)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}

	known := func(p string) bool {
		for _, v := range d.Projects() {
			if v.Name == p {
				return true
			}
		}
		return false
	}
	session := func(p string) (string, bool) {
		tok, err := reg.ReadSecret(p, "session-token")
		if err != nil {
			return "", false
		}
		return tok, true
	}

	handler := Build(d, known, session, cfg.Token)

	go func() {
		t := time.NewTicker(30 * time.Second)
		defer t.Stop()
		for range t.C {
			d.Tick(ctx)
		}
	}()

	addr := fmt.Sprintf("127.0.0.1:%d", cfg.Port)
	fmt.Println("morphd listening on " + addr)
	if err := http.ListenAndServe(addr, handler); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}
}
