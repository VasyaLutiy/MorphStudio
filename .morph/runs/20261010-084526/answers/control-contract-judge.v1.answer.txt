func ccJSON(t *testing.T, v any) string {
	t.Helper()
	b, err := json.Marshal(v)
	if err != nil {
		t.Fatal(err)
	}
	return string(b)
}

func ccCode(err error) [2]any {
	n, s := Code(err)
	return [2]any{n, s}
}

func TestControlContractExample1(t *testing.T) {
	s := Status{
		State:     "busy",
		Phase:     "P18",
		Minutes:   37,
		CostUSD:   1.92,
		FiveHour:  22,
		SevenDay:  60,
		SessionID: "s-1",
		Queue:     QueueView{State: "running", Approved: "7b31dfe", Index: 1, Phases: 3, Current: "P18"},
		Repo:      github.Access{Checked: true, Reachable: true, Push: true, CheckedAt: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC)},
	}
	testhelp.Equal(t, "Status json", ccJSON(t, s),
		`{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}`)
}

func TestControlContractExample2(t *testing.T) {
	s := Status{
		State: "waiting",
		Stop: &Stop{
			Kind:   "not_pushed",
			Phase:  "P17",
			Reason: "not pushed: HEAD bbb222, origin/main aaa111",
			At:     time.Date(2026, 10, 8, 16, 0, 0, 0, time.UTC),
		},
	}
	got := ccJSON(t, s)
	if !strings.HasSuffix(got, `,"stop":{"kind":"not_pushed","phase":"P17","reason":"not pushed: HEAD bbb222, origin/main aaa111","at":"2026-10-08T16:00:00Z"}}`) {
		t.Errorf("Status stop suffix:\ngot:  %q", got)
	}
	if strings.Contains(got, "issue_url") {
		t.Errorf("Status stop should not carry issue_url:\ngot:  %q", got)
	}
}

func TestControlContractExample5(t *testing.T) {
	testhelp.Equal(t, "Code(nil)", ccCode(nil), [2]any{200, ""})
	testhelp.Equal(t, "Code(ErrUnknownProject)", ccCode(ErrUnknownProject), [2]any{404, "unknown_project"})

	testhelp.Equal(t, "Code(ErrBadToken)", ccCode(ErrBadToken), [2]any{401, "bad_session_token"})
	testhelp.Equal(t, "Code(ErrExists)", ccCode(ErrExists), [2]any{409, "exists"})
	testhelp.Equal(t, "Code(ErrNoSession)", ccCode(ErrNoSession), [2]any{409, "no_session"})
	testhelp.Equal(t, "Code(ErrBusy)", ccCode(ErrBusy), [2]any{409, "busy"})
	testhelp.Equal(t, "Code(ErrNoQuestion)", ccCode(ErrNoQuestion), [2]any{409, "no_question"})
	testhelp.Equal(t, "Code(ErrNotBusy)", ccCode(ErrNotBusy), [2]any{409, "not_busy"})
	testhelp.Equal(t, "Code(ErrNotWaiting)", ccCode(ErrNotWaiting), [2]any{409, "not_waiting"})

func TestControlContractExample8(t *testing.T) {
	idle := Status{State: "idle", LimitsUnknown: true}
	testhelp.Equal(t, "Status idle json", ccJSON(t, idle),
		`{"state":"idle","phase":"","minutes":0,"cost_usd":0,"five_hour":null,"seven_day":null,"session_id":"","queue":{"state":"","approved":"","index":0,"phases":0,"current":"","reason":""},"repo":{"checked":false,"reachable":false,"push":false,"reason":"","checked_at":"0001-01-01T00:00:00Z"}}`)

	busy := Status{
		State:         "busy",
		Phase:         "S1",
		Minutes:       2,
		CostUSD:       0.03,
		FiveHour:      77,
		SevenDay:      72,
		LimitsUnknown: true,
		SessionID:     "s-9",
	}
	got := ccJSON(t, busy)
	if !strings.HasPrefix(got, `{"state":"busy","phase":"S1","minutes":2,"cost_usd":0.03,`) {
		t.Errorf("Status busy prefix:\ngot:  %q", got)
	}
	if !strings.Contains(got, `,"five_hour":null,"seven_day":null,"session_id":"s-9",`) {
		t.Errorf("Status busy limits:\ngot:  %q", got)
	}
	if strings.Contains(got, "77") || strings.Contains(got, "72") {
		t.Errorf("Status busy should hide the raw ints:\ngot:  %q", got)
	}
}

func TestControlContractExample10(t *testing.T) {
	exit := Exit{
		Code:     1,
		At:       time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
		Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
		Restarts: 3,
	}
	testhelp.Equal(t, "Exit json", ccJSON(t, exit),
		`{"code":1,"at":"2026-10-09T08:14:02Z","stderr":"Error: Invalid session ID. Must be a valid UUID.","restarts":3}`)

	s := Status{
		State:         "waiting",
		Phase:         "S1",
		LimitsUnknown: true,
		Stop: &Stop{
			Kind:   "crash",
			Phase:  "S1",
			Reason: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.",
			At:     time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
		},
		LastExit: &Exit{
			Code:     1,
			At:       time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
			Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
			Restarts: 3,
		},
	}
	got := ccJSON(t, s)
	if !strings.HasSuffix(got, `,"stop":{"kind":"crash","phase":"S1","reason":"exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.","at":"2026-10-09T08:14:02Z"},"last_exit":{"code":1,"at":"2026-10-09T08:14:02Z","stderr":"Error: Invalid session ID. Must be a valid UUID.","restarts":3}}`) {
		t.Errorf("Status stop/last_exit suffix:\ngot:  %q", got)
	}
	if !strings.Contains(got, `"five_hour":null`) {
		t.Errorf("Status should carry five_hour null:\ngot:  %q", got)
	}
}
