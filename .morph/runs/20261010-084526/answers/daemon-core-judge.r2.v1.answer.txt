type dcHarness struct {
	t    *testing.T
	tmp  string
	reg  *registry.Registry
	run  *runner.Fake
	deps Deps
	d    *Daemon

	mu       sync.Mutex
	now      time.Time
	n        int
	launches []claude.Launch
	procs    []*claude.Fake
	posts    [][4]string
	githubDo func(*http.Request) (*http.Response, error)
	spawnErr error
}

func newDCHarness(t *testing.T) *dcHarness {
	t.Helper()
	tmp := t.TempDir()
	reg, err := registry.Open(filepath.Join(tmp, "state"))
	if err != nil {
		t.Fatal(err)
	}
	h := &dcHarness{
		t:   t,
		tmp: tmp,
		reg: reg,
		now: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC),
	}
	h.run = &runner.Fake{Script: map[string]runner.Result{
		"git rev-parse --abbrev-ref HEAD": {Stdout: "main\n"},
		"git rev-parse HEAD":              {Stdout: "aaa111\n"},
		"git rev-parse origin/main":       {Stdout: "aaa111\n"},
		"git status --porcelain":          {Stdout: ""},
	}}
	h.githubDo = func(r *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: 200,
			Body:       io.NopCloser(strings.NewReader(`{"permissions":{"push":true}}`)),
			Header:     http.Header{},
		}, nil
	}
	h.deps = Deps{
		Registry: reg,
		Runner:   h.run,
		Spawn:    h.spawn,
		Post:     h.post,
		GitHubDo: h.github,
		Now:      h.nowTime,
		NewID:    h.newID,
		Config: Config{
			ClaudeBin:   "/opt/claude/bin/claude",
			MorphBin:    "/usr/local/bin/morph",
			ProjectsDir: filepath.Join(tmp, "projects"),
			MCPBaseURL:  "http://127.0.0.1:7181",
			Defaults:    queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4},
			MaxParallel: 1,
			Loop: supervisor.Config{
				MaxResumesPerHour: 3,
				StallMinutes:      30,
				StretchUSD:        30,
				UsageAlertPercent: 50,
			},
		},
	}
	return h
}

func (h *dcHarness) addProject(name, language, repoURL string) {
	h.t.Helper()
	dir := filepath.Join(h.tmp, "projects", name)
	if err := os.MkdirAll(dir, 0o755); err != nil {
		h.t.Fatal(err)
	}
	if err := h.reg.Add(registry.Project{
		Name:      name,
		Language:  language,
		RepoURL:   repoURL,
		Dir:       dir,
		Caps:      queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5},
		CreatedAt: time.Date(2026, 10, 8, 14, 0, 0, 0, time.UTC),
	}); err != nil {
		h.t.Fatal(err)
	}
	measure, err := os.ReadFile("../tests/fixtures/git/measure.md")
	if err != nil {
		h.t.Fatal(err)
	}
	measurePath := filepath.Join(dir, "docs", "MEASURE.md")
	if err := os.MkdirAll(filepath.Dir(measurePath), 0o755); err != nil {
		h.t.Fatal(err)
	}
	if err := os.WriteFile(measurePath, measure, 0o644); err != nil {
		h.t.Fatal(err)
	}
}

func (h *dcHarness) open() {
	h.t.Helper()
	d, err := Open(context.Background(), h.deps)
	if err != nil {
		h.t.Fatal(err)
	}
	h.d = d
}

func (h *dcHarness) spawn(ctx context.Context, l claude.Launch) (claude.Process, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.launches = append(h.launches, l)
	if h.spawnErr != nil {
		return nil, h.spawnErr
	}
	f := claude.NewFake()
	h.procs = append(h.procs, f)
	return f, nil
}

func (h *dcHarness) post(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
	return telegram.Status{}, nil
}

func (h *dcHarness) github(req *http.Request) (*http.Response, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	return h.githubDo(req)
}

func (h *dcHarness) nowTime() time.Time {
	h.mu.Lock()
	defer h.mu.Unlock()
	t := h.now
	h.now = h.now.Add(time.Second)
	return t
}

func (h *dcHarness) newID() string {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.n++
	return fmt.Sprintf("id-%d", h.n)
}

func (h *dcHarness) launchList() []claude.Launch {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([]claude.Launch, len(h.launches))
	copy(out, h.launches)
	return out
}

func (h *dcHarness) postList() [][4]string {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([][4]string, len(h.posts))
	copy(out, h.posts)
	return out
}

func (h *dcHarness) procList() []*claude.Fake {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([]*claude.Fake, len(h.procs))
	copy(out, h.procs)
	return out
}

func dcPlan(t *testing.T) queue.Plan {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var plan queue.Plan
	if err := json.Unmarshal(data, &plan); err != nil {
		t.Fatal(err)
	}
	return plan
}

func TestDaemonCoreExample8(t *testing.T) {
	h := newDCHarness(t)
	h.addProject("demo", "go", "https://github.com/acme/demo")
	h.run.Script["git rev-parse --verify -q origin/main"] = runner.Result{Code: 1}
	h.run.Script["git push -q -u origin main"] = runner.Result{Code: 0}
	h.open()
	d := h.d

	ctx := context.Background()

	h.githubDo = func(r *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: 200,
			Body:       io.NopCloser(strings.NewReader(`{"permissions":{"push":true}}`)),
			Header:     http.Header{},
		}, nil
	}

	res, err := d.PutGithubToken(ctx, "demo", "ghp_abc")
	testhelp.Equal(t, "put err", err, nil)
	acc := res.Access
	testhelp.Equal(t, "checked", acc.Checked, true)
	testhelp.Equal(t, "reachable", acc.Reachable, true)
	testhelp.Equal(t, "push", acc.Push, true)
	testhelp.Equal(t, "reason", acc.Reason, "")
	testhelp.Equal(t, "checked at zero", acc.CheckedAt.IsZero(), false)
	testhelp.Equal(t, "pushed", res.Pushed, true)
	testhelp.Equal(t, "push error", res.PushError, "")

	token, err := h.reg.ReadSecret("demo", "github-token")
	testhelp.Equal(t, "github token err", err, nil)
	testhelp.Equal(t, "github token", token, "ghp_abc")

	cred, err := h.reg.ReadSecret("demo", "git-credentials")
	testhelp.Equal(t, "git cred err", err, nil)
	testhelp.Equal(t, "git cred", cred, "https://x-access-token:ghp_abc@github.com\n")

	for _, kind := range []string{"github-token", "git-credentials"} {
		fi, err := os.Stat(h.reg.SecretPath("demo", kind))
		testhelp.Equal(t, "secret stat err "+kind, err, nil)
		testhelp.Equal(t, "secret mode "+kind, fi.Mode().Perm(), os.FileMode(0o600))
	}

	st, err := d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "repo", st.Repo, acc)

	calls := h.run.Calls
	wantCalls := []runner.Call{
		{Dir: filepath.Join(h.tmp, "projects", "demo"), Name: "git", Args: []string{"rev-parse", "--verify", "-q", "origin/main"}},
		{Dir: filepath.Join(h.tmp, "projects", "demo"), Name: "git", Args: []string{"push", "-q", "-u", "origin", "main"}},
	}
	testhelp.Equal(t, "calls", calls, wantCalls)

	h.githubDo = func(r *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: 401,
			Body:       io.NopCloser(strings.NewReader(`{"message":"Bad credentials"}`)),
			Header:     http.Header{},
		}, nil
	}

	res, err = d.PutGithubToken(ctx, "demo", "ghp_abc")
	testhelp.Equal(t, "put 401 err", err, nil)
	acc = res.Access
	testhelp.Equal(t, "401 checked", acc.Checked, true)
	testhelp.Equal(t, "401 reachable", acc.Reachable, false)
	testhelp.Equal(t, "401 push", acc.Push, false)
	testhelp.Equal(t, "401 reason", acc.Reason, "401 bad credentials")

	testhelp.Equal(t, "401 pushed", res.Pushed, false)

func TestDaemonCoreExample11(t *testing.T) {
	h := newDCHarness(t)
	h.addProject("demo", "go", "https://github.com/acme/demo")
	h.spawnErr = errors.New("claude: start fork/exec /opt/claude/bin/claude: no such file or directory")
	h.open()
	d := h.d
	plan := dcPlan(t)

	qv, err := d.PlanLoad("demo", plan)
	testhelp.Equal(t, "planload err", err, nil)
	testhelp.Equal(t, "planload qv", qv, control.QueueView{
		State:    "stopped",
		Approved: "7b31dfe",
		Index:    0,
		Phases:   3,
		Current:  "P17",
		Reason:   "crash",
	})

	launches := h.launchList()
	testhelp.Equal(t, "launch count", len(launches), 4)
	wantSessions := []string{"id-1", "id-3", "id-5", "id-7"}
	for i, l := range launches {
		testhelp.Equal(t, fmt.Sprintf("launch %d session", i), l.SessionID, wantSessions[i])
		testhelp.Equal(t, fmt.Sprintf("launch %d resume", i), l.Resume, false)
	}

	token, err := h.reg.ReadSecret("demo", "session-token")
	testhelp.Equal(t, "token err", err, nil)
	testhelp.Equal(t, "token", token, "id-8")

	posts := h.postList()
	wantPosts := [][4]string{
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 1 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 2 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 3 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "stop", "P17: session keeps exiting", "exited 4 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
	}
	testhelp.Equal(t, "posts", posts, wantPosts)

	st, err := d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "state", st.State, "waiting")
	testhelp.Equal(t, "phase", st.Phase, "P17")
	testhelp.Equal(t, "session", st.SessionID, "id-7")
	testhelp.Equal(t, "limits unknown", st.LimitsUnknown, true)
	if st.Stop == nil {
		t.Fatal("stop nil")
	}
	testhelp.Equal(t, "stop kind", st.Stop.Kind, "crash")
	testhelp.Equal(t, "stop reason", st.Stop.Reason, "exited 4 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory")

	if st.LastExit == nil {
		t.Fatal("last exit nil")
	}
	le := *st.LastExit
	testhelp.Equal(t, "last exit at zero", le.At.IsZero(), false)
	le.At = time.Time{}
	testhelp.Equal(t, "last exit", le, control.Exit{
		Code:     -3,
		Stderr:   "claude: start fork/exec /opt/claude/bin/claude: no such file or directory",
		Restarts: 3,
	})

	stop, err := d.StopCheck("demo")
	testhelp.Equal(t, "stopcheck err", err, nil)
	if stop == nil {
		t.Fatal("stopcheck nil")
	}
	cp := *stop
	testhelp.Equal(t, "stopcheck at zero", cp.At.IsZero(), false)
	cp.At = time.Time{}
	testhelp.Equal(t, "stopcheck", cp, control.Stop{
		Kind:   "crash",
		Phase:  "P17",
		Reason: st.Stop.Reason,
	})

	ev, err := d.Events("demo", 0, 10)
	testhelp.Equal(t, "events err", err, nil)
	testhelp.Equal(t, "events last", ev.Last, int64(0))
	testhelp.Equal(t, "events len", len(ev.Entries), 0)

	data, err := os.ReadFile(filepath.Join(h.tmp, "state", "demo", "state.json"))
	testhelp.Equal(t, "state read err", err, nil)
	var saved State
	err = json.Unmarshal(data, &saved)
	testhelp.Equal(t, "state unmarshal err", err, nil)
	testhelp.Equal(t, "saved loop state", saved.Loop.State, "waiting")
	testhelp.Equal(t, "saved restarts len", len(saved.Loop.Restarts), 3)

	err = d.Continue("demo")
	testhelp.Equal(t, "continue err", err, nil)

	launches2 := h.launchList()
	testhelp.Equal(t, "launch count after continue", len(launches2), 8)
	wantSessions2 := []string{"id-9", "id-11", "id-13", "id-15"}
	for i, l := range launches2[4:] {
		testhelp.Equal(t, fmt.Sprintf("launch2 %d session", i), l.SessionID, wantSessions2[i])
	}

	st, err = d.Status("demo")
	testhelp.Equal(t, "status err 2", err, nil)
	testhelp.Equal(t, "state after", st.State, "waiting")
	if st.Stop == nil {
		t.Fatal("stop nil after")
	}
	testhelp.Equal(t, "stop kind after", st.Stop.Kind, "crash")

	posts2 := h.postList()
	testhelp.Equal(t, "posts total", len(posts2), 8)
}
