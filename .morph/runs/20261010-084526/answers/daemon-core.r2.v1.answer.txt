```go
// Package daemon is the core of morphd: it owns the registry, the project
// loops, the claude processes and implements control.Control.
package daemon

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"net/http"
	"path/filepath"
	"sync"
	"time"

	"morphstudio/bootstrap"
	"morphstudio/claude"
	"morphstudio/control"
	"morphstudio/eventlog"
	"morphstudio/github"
	"morphstudio/gitrules"
	"morphstudio/queue"
	"morphstudio/registry"
	"morphstudio/runner"
	"morphstudio/session"
	"morphstudio/supervisor"
	"morphstudio/telegram"
)

// Deps are the daemon's dependencies: the registry, the runner, the claude
// spawn function, the milestone poster, the GitHub Do, the clock and the id
// source.
type Deps struct {
	Registry *registry.Registry
	Runner   runner.Runner
	Spawn    func(ctx context.Context, l claude.Launch) (claude.Process, error)
	Post     func(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error)
	GitHubDo func(*http.Request) (*http.Response, error)
	Now      func() time.Time
	NewID    func() string
	Config   Config
}

// Config bounds the daemon's behaviour.
type Config struct {
	ClaudeBin   string
	MorphBin    string
	ProjectsDir string
	MCPBaseURL  string
	Model       string
	ExtraArgs   []string
	Defaults    queue.Caps
	MaxParallel int
	Loop        supervisor.Config
}

// State is one project's persisted state.
type State struct {
	Loop supervisor.Loop `json:"loop"`
	Repo github.Access   `json:"repo"`
}

// proj is one project's runtime.
type proj struct {
	reg      registry.Project
	state    State
	machine  *session.Machine
	proc     claude.Process
	log      *eventlog.Log
	queuedAt time.Time
	pumps    map[claude.Process]chan struct{}
}

// Daemon is the core of morphd.
type Daemon struct {
	deps     Deps
	ctx      context.Context
	mu       sync.Mutex
	projects map[string]*proj
}

// startPump is set by pump.go.
var startPump func(d *Daemon, ctx context.Context, project string, p claude.Process, l *eventlog.Log)

var _ control.Control = (*Daemon)(nil)

// Open builds a daemon around deps, loading every registered project.
func Open(ctx context.Context, deps Deps) (*Daemon, error) {
	d := &Daemon{deps: deps, ctx: ctx, projects: map[string]*proj{}}
	d.mu.Lock()
	defer d.mu.Unlock()

	for _, rp := range deps.Registry.List() {
		p := &proj{reg: rp, pumps: map[claude.Process]chan struct{}{}}
		var st State
		err := deps.Registry.LoadState(rp.Name, &st)
		if err != nil {
			if !errors.Is(err, registry.ErrNoState) {
				return nil, err
			}
			st = State{Loop: *supervisor.New(rp.Name, queue.Queue{})}
		}
		p.state = st
		d.projects[rp.Name] = p
	}

	for _, rp := range deps.Registry.List() {
		p := d.projects[rp.Name]
		switch p.state.Loop.State {
		case "running", "starting", "paused":
			d.execute(ctx, rp.Name, p.state.Loop.Exited(-2, "", deps.Now(), deps.Config.Loop))
		}
	}
	return d, nil
}

// runningCount returns the number of projects with an active loop; d.mu must
// be held.
func (d *Daemon) runningCount() int {
	n := 0
	for _, p := range d.projects {
		switch p.state.Loop.State {
		case "running", "starting", "paused":
			n++
		}
	}
	return n
}

// Projects lists every registered project in name order.
func (d *Daemon) Projects() []control.ProjectView {
	d.mu.Lock()
	defer d.mu.Unlock()
	out := []control.ProjectView{}
	for _, rp := range d.deps.Registry.List() {
		p := d.projects[rp.Name]
		if p == nil {
			continue
		}
		out = append(out, control.ProjectView{
			Name:      p.reg.Name,
			Language:  p.reg.Language,
			RepoURL:   p.reg.RepoURL,
			State:     p.state.Loop.State,
			CreatedAt: p.reg.CreatedAt,
		})
	}
	return out
}

// CreateProject validates and runs the project bootstrap sequence.
func (d *Daemon) CreateProject(ctx context.Context, name, language, repoURL string) (control.ProjectView, error) {
	d.mu.Lock()
	defer d.mu.Unlock()

	if _, ok := d.projects[name]; ok {
		return control.ProjectView{}, control.ErrExists
	}

	spec := bootstrap.Spec{
		Name:            name,
		Language:        language,
		RepoURL:         repoURL,
		ProjectsDir:     d.deps.Config.ProjectsDir,
		CredentialsFile: d.deps.Registry.SecretPath(name, "git-credentials"),
		MorphBin:        d.deps.Config.MorphBin,
	}
	rep, err := bootstrap.Create(ctx, d.deps.Runner, spec)
	if err != nil {
		if len(rep.Steps) == 0 {
			return control.ProjectView{}, fmt.Errorf("%w: %v", control.ErrBadInput, err)
		}
		return control.ProjectView{}, err
	}

	rp := registry.Project{
		Name:      name,
		Language:  language,
		RepoURL:   repoURL,
		Dir:       rep.Dir,
		Caps:      d.deps.Config.Defaults,
		CreatedAt: d.deps.Now(),
	}
	if err := d.deps.Registry.Add(rp); err != nil {
		switch {
		case errors.Is(err, registry.ErrExists):
			return control.ProjectView{}, control.ErrExists
		case errors.Is(err, registry.ErrBadName):
			return control.ProjectView{}, fmt.Errorf("%w: %v", control.ErrBadInput, err)
		default:
			return control.ProjectView{}, err
		}
	}

	d.projects[name] = &proj{
		reg:   rp,
		state: State{Loop: *supervisor.New(name, queue.Queue{})},
		pumps: map[claude.Process]chan struct{}{},
	}
	return control.ProjectView{
		Name:      name,
		Language:  language,
		RepoURL:   repoURL,
		State:     "idle",
		CreatedAt: rp.CreatedAt,
	}, nil
}

// Status reports a project's current state.
func (d *Daemon) Status(project string) (control.Status, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.Status{}, control.ErrUnknownProject
	}
	loop := p.state.Loop
	var out control.Status
	out.State = loop.State
	if loop.State == "running" && p.proc != nil && p.machine != nil {
		out.State = p.machine.State
	}
	out.Phase = loop.Phase
	if loop.State == "running" || loop.State == "paused" {
		out.Minutes = int(d.deps.Now().Sub(loop.PhaseStartedAt).Minutes())
	}
	if p.machine != nil {
		out.CostUSD = p.machine.CostUSD
		if p.machine.Limits != nil {
			out.FiveHour = control.Percent(p.machine.Limits.FiveHour.Utilization)
			out.SevenDay = control.Percent(p.machine.Limits.SevenDay.Utilization)
		} else {
			out.LimitsUnknown = true
		}
		if p.machine.SessionID != "" {
			out.SessionID = p.machine.SessionID
		}
	} else {
		out.LimitsUnknown = true
	}
	if out.SessionID == "" {
		out.SessionID = loop.SessionID
	}
	out.Queue = queueView(loop.Queue)
	out.Repo = p.state.Repo
	out.Stop = loop.Stop
	out.LastExit = loop.LastExit
	return out, nil
}

// Events returns a page of the project's event log.
func (d *Daemon) Events(project string, since int64, max int) (control.Events, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.Events{}, control.ErrUnknownProject
	}
	if p.log == nil {
		return control.Events{Entries: []eventlog.Entry{}, Last: 0}, nil
	}
	return control.Events{Entries: p.log.Since(since, max), Last: p.log.Last()}, nil
}

// Order routes text to the project's machine.
func (d *Daemon) Order(project, text string) (session.OrderResult, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return session.OrderResult{}, control.ErrUnknownProject
	}
	if p.proc == nil || p.machine == nil {
		return session.OrderResult{}, control.ErrNoSession
	}
	res, acts := p.machine.Order(text, d.deps.Now())
	for _, a := range acts {
		if a.Kind == "write" {
			d.writeLine(project, a.Line)
		}
	}
	return res, nil
}

// Pending returns the pending AskUserQuestion, if any.
func (d *Daemon) Pending(project string) (*control.Question, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return nil, control.ErrUnknownProject
	}
	if p.machine == nil || p.machine.Pending == nil {
		return nil, nil
	}
	pend := p.machine.Pending
	return &control.Question{
		RequestID: pend.RequestID,
		Text:      pend.Question.Text,
		Header:    pend.Question.Header,
		Options:   append([]string(nil), pend.Question.Options...),
		AskedAt:   pend.AskedAt,
	}, nil
}

// Answer grants the pending AskUserQuestion with option.
func (d *Daemon) Answer(project string, option int) error {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.ErrUnknownProject
	}
	if p.proc == nil || p.machine == nil {
		return control.ErrNoSession
	}
	acts, err := p.machine.Answer(option, d.deps.Now())
	if err != nil {
		return err
	}
	for _, a := range acts {
		if a.Kind == "write" {
			d.writeLine(project, a.Line)
		}
	}
	return nil
}

// Interrupt asks the running session to stop its current turn.
func (d *Daemon) Interrupt(project string) error {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.ErrUnknownProject
	}
	if p.proc == nil || p.machine == nil {
		return control.ErrNoSession
	}
	acts, err := p.machine.Interrupt("int-" + d.deps.NewID())
	if err != nil {
		return err
	}
	for _, a := range acts {
		if a.Kind == "write" {
			d.writeLine(project, a.Line)
		}
	}
	return nil
}

// Usage reports the project's rate limit windows and costs.
func (d *Daemon) Usage(project string) (control.Usage, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.Usage{}, control.ErrUnknownProject
	}
	var out control.Usage
	if p.machine != nil {
		if p.machine.Limits != nil {
			out.FiveHour = control.Percent(p.machine.Limits.FiveHour.Utilization)
			out.SevenDay = control.Percent(p.machine.Limits.SevenDay.Utilization)
			out.FiveHourResetsAt = p.machine.Limits.FiveHour.ResetsAt
			out.SevenDayResetsAt = p.machine.Limits.SevenDay.ResetsAt
		}
		out.SessionCostUSD = p.machine.CostUSD
		out.LimitsAt = p.machine.LimitsAt
	}
	out.StretchCostUSD = p.state.Loop.StretchUSD + p.state.Loop.SessionUSD
	return out, nil
}

// Restart kills the running session and asks for a fresh start check.
func (d *Daemon) Restart(project string, force bool) error {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.ErrUnknownProject
	}
	if p.proc != nil && p.machine != nil && p.machine.State != "ready" && !force {
		return control.ErrBusy
	}
	acts, err := p.state.Loop.Restart(d.deps.Now())
	if err != nil {
		return err
	}
	d.execute(d.ctx, project, acts)
	return nil
}

// PlanLoad validates and installs a queue for the project.
func (d *Daemon) PlanLoad(project string, plan queue.Plan) (control.QueueView, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.QueueView{}, control.ErrUnknownProject
	}
	if p.state.Loop.State != "idle" && p.state.Loop.State != "waiting" {
		return control.QueueView{}, control.ErrBusy
	}
	q, err := queue.Load(plan, p.reg.Caps)
	if err != nil {
		return control.QueueView{}, fmt.Errorf("%w: %v", control.ErrBadInput, err)
	}
	p.state.Loop = *supervisor.New(project, q)
	p.queuedAt = d.deps.Now()
	if d.runningCount() < d.deps.Config.MaxParallel {
		acts, err := p.state.Loop.Begin(d.deps.Now())
		if err == nil {
			d.execute(d.ctx, project, acts)
		}
	} else {
		_ = d.deps.Registry.SaveState(project, p.state)
	}
	return queueView(p.state.Loop.Queue), nil
}

// Continue resumes a stopped loop.
func (d *Daemon) Continue(project string) error {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.ErrUnknownProject
	}
	acts, err := p.state.Loop.Continue(d.deps.Now())
	if err != nil {
		if errors.Is(err, control.ErrNotWaiting) {
			return err
		}
		return fmt.Errorf("%w: %v", control.ErrNotWaiting, err)
	}
	d.execute(d.ctx, project, acts)
	return nil
}

// StopCheck returns the loop's current stop, if any.
func (d *Daemon) StopCheck(project string) (*control.Stop, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return nil, control.ErrUnknownProject
	}
	if p.state.Loop.Stop == nil {
		return nil, nil
	}
	cp := *p.state.Loop.Stop
	return &cp, nil
}

// PutGithubToken stores a GitHub token and checks repository access.
func (d *Daemon) PutGithubToken(ctx context.Context, project, token string) (control.TokenResult, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, ok := d.projects[project]
	if !ok {
		return control.TokenResult{}, control.ErrUnknownProject
	}
	if token == "" {
		return control.TokenResult{}, control.ErrBadInput
	}
	if err := d.deps.Registry.PutSecret(project, "github-token", token); err != nil {
		return control.TokenResult{}, err
	}
	if err := d.deps.Registry.PutSecret(project, "git-credentials", github.CredentialLine(token)); err != nil {
		return control.TokenResult{}, err
	}
	access := github.Check(ctx, d.deps.GitHubDo, token, p.reg.RepoURL, d.deps.Now())
	p.state.Repo = access
	_ = d.deps.Registry.SaveState(project, p.state)
	res := control.TokenResult{Access: access}
	if access.Push {
		pushed, err := bootstrap.FirstPush(ctx, d.deps.Runner, p.reg.Dir)
		res.Pushed = pushed
		if err != nil {
			res.PushError = err.Error()
		}
	}
	return res, nil
}

// checkSession verifies a project session token; d.mu must be held.
func (d *Daemon) checkSession(project, sessionToken string) (*proj, error) {
	p, ok := d.projects[project]
	if !ok {
		return nil, control.ErrUnknownProject
	}
	stored, err := d.deps.Registry.ReadSecret(project, "session-token")
	if err != nil || stored != sessionToken {
		return nil, control.ErrBadToken
	}
	return p, nil
}

// PhaseDone records that a running phase finished and moves the queue.
func (d *Daemon) PhaseDone(project, sessionToken, phase, next string) error {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, err := d.checkSession(project, sessionToken)
	if err != nil {
		return err
	}
	acts, err := p.state.Loop.PhaseDone(phase, next)
	if err != nil {
		return err
	}
	d.execute(d.ctx, project, acts)
	return nil
}

// WaitOperator records that a session is waiting on the operator.
func (d *Daemon) WaitOperator(project, sessionToken, kind, reason, issueURL string) error {
	d.mu.Lock()
	defer d.mu.Unlock()
	p, err := d.checkSession(project, sessionToken)
	if err != nil {
		return err
	}
	acts := p.state.Loop.WaitOperator(kind, reason, issueURL, d.deps.Now())
	d.execute(d.ctx, project, acts)
	return nil
}

// Milestone posts a progress milestone for the project.
func (d *Daemon) Milestone(project, sessionToken string, m control.Milestone) error {
	d.mu.Lock()
	defer d.mu.Unlock()
	if _, err := d.checkSession(project, sessionToken); err != nil {
		return err
	}
	_, _ = d.deps.Post(d.ctx, project, m.Kind, m.Headline, m.Numbers)
	return nil
}

// Execute runs a batch of supervisor actions under the daemon lock.
func (d *Daemon) Execute(ctx context.Context, project string, acts []supervisor.Action) {
	d.mu.Lock()
	defer d.mu.Unlock()
	d.execute(ctx, project, acts)
}

// execute runs a batch of supervisor actions; d.mu must be held.
func (d *Daemon) execute(ctx context.Context, project string, acts []supervisor.Action) {
	p, ok := d.projects[project]
	if !ok {
		return
	}
	for _, a := range acts {
		switch a.Kind {
		case "start_check":
			start := gitrules.StartCheck(ctx, d.deps.Runner, p.reg.Dir)
			d.execute(ctx, project, p.state.Loop.StartChecked(start, d.deps.NewID, d.deps.Now()))
		case "spawn":
			if err := d.spawn(ctx, project, p, a); err != nil {
				d.execute(ctx, project, p.state.Loop.Exited(-3, err.Error(), d.deps.Now(), d.deps.Config.Loop))
				return
			}
		case "first_line", "write":
			if p.machine != nil {
				_, was := p.machine.Order(a.Line, d.deps.Now())
				for _, w := range was {
					if w.Kind == "write" {
						d.writeLine(project, w.Line)
					}
				}
			}
		case "end_check":
			end := gitrules.EndCheck(ctx, d.deps.Runner, p.reg.Dir, p.state.Loop.Phase)
			d.execute(ctx, project, p.state.Loop.EndChecked(end, d.deps.Now()))
		case "kill":
			if p.proc != nil {
				_ = p.proc.Kill()
			}
		case "post":
			_, _ = d.deps.Post(ctx, project, a.Milestone.Kind, a.Milestone.Headline, a.Milestone.Numbers)
		}
	}
	_ = d.deps.Registry.SaveState(project, p.state)
}

// spawn launches a new claude session for project; d.mu must be held.
func (d *Daemon) spawn(ctx context.Context, project string, p *proj, a supervisor.Action) error {
	if p.proc != nil {
		_ = p.proc.Kill()
		p.proc = nil
		p.machine = nil
		p.log = nil
	}
	token := d.deps.NewID()
	if err := d.deps.Registry.PutSecret(project, "session-token", token); err != nil {
		return err
	}
	mcpPath, err := claude.WriteMCPConfig(
		filepath.Join(d.deps.Registry.Base(), project),
		"morphd",
		d.deps.Config.MCPBaseURL+"/mcp/"+project+"/session",
		token,
	)
	if err != nil {
		return err
	}
	log, err := eventlog.Open(filepath.Join(d.deps.Registry.Base(), project, "sessions", a.SessionID+".jsonl"))
	if err != nil {
		return err
	}
	proc, err := d.deps.Spawn(ctx, claude.Launch{
		Bin:           d.deps.Config.ClaudeBin,
		Dir:           p.reg.Dir,
		SessionID:     a.SessionID,
		Resume:        a.Resume,
		BudgetUSD:     a.BudgetUSD,
		Model:         d.deps.Config.Model,
		MCPConfigPath: mcpPath,
		Extra:         d.deps.Config.ExtraArgs,
	})
	if err != nil {
		return err
	}
	p.machine = session.New()
	p.proc = proc
	p.log = log
	if startPump != nil {
		startPump(d, ctx, project, proc, log)
	}
	return nil
}

// writeLine sends a line to the current process and logs it; d.mu must be held.
func (d *Daemon) writeLine(project string, line []byte) {
	p, ok := d.projects[project]
	if !ok || p.proc == nil {
		return
	}
	_ = p.proc.Write(line)
	if p.log != nil {
		_, _ = p.log.Append("in", json.RawMessage(line), d.deps.Now())
	}
}

// Tick drives every project's timers and starts queued work when capacity
// permits.
func (d *Daemon) Tick(ctx context.Context) {
	d.mu.Lock()
	defer d.mu.Unlock()

	for _, rp := range d.deps.Registry.List() {
		p := d.projects[rp.Name]
		if p == nil {
			continue
		}
		acts := p.state.Loop.Tick(d.deps.Now(), d.deps.Config.Loop)
		if len(acts) > 0 {
			d.execute(ctx, rp.Name, acts)
		}
	}

	for d.runningCount() < d.deps.Config.MaxParallel {
		var best *proj
		var bestName string
		for _, rp := range d.deps.Registry.List() {
			p := d.projects[rp.Name]
			if p == nil || p.state.Loop.State != "idle" || p.state.Loop.Queue.State != "queued" {
				continue
			}
			if best == nil || p.queuedAt.Before(best.queuedAt) {
				best = p
				bestName = rp.Name
			}
		}
		if best == nil {
			break
		}
		acts, err := best.state.Loop.Begin(d.deps.Now())
		if err != nil {
			break
		}
		d.execute(ctx, bestName, acts)
	}
}

// queueView serializes a queue's walking state.
func queueView(q queue.Queue) control.QueueView {
	cur := ""
	if c, ok := q.Current(); ok {
		cur = c.ID
	}
	return control.QueueView{
		State:    q.State,
		Approved: q.Approved,
		Index:    q.Index,
		Phases:   len(q.Phases),
		Current:  cur,
		Reason:   q.Reason,
	}
}
```
