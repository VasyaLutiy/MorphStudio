// puHarness is the harness of example 1: a registry holding one project, a
// scripted git runner, recorded spawns and posts, and a clock ticking one
// second per call. The daemon calls Spawn, Post, Now and NewID from its own
// goroutines, so all of them are guarded by mu.
type puHarness struct {
	t   *testing.T
	tmp string
	reg *registry.Registry
	run *runner.Fake

	starter func(ctx context.Context, l claude.Launch) (claude.Process, error)

	mu     sync.Mutex
	spawns []claude.Launch
	fakes  []*claude.Fake
	posts  [][4]string
	clock  time.Time
	ids    int
}

// puNew builds the example-1 harness without opening the daemon.
func puNew(t *testing.T) *puHarness {
	t.Helper()
	tmp := t.TempDir()
	reg, err := registry.Open(filepath.Join(tmp, "state"))
	if err != nil {
		t.Fatal(err)
	}
	dir := filepath.Join(tmp, "projects", "demo")
	if err := os.MkdirAll(filepath.Join(dir, "docs"), 0o755); err != nil {
		t.Fatal(err)
	}
	measure, err := os.ReadFile("../tests/fixtures/git/measure.md")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(dir, "docs", "MEASURE.md"), measure, 0o644); err != nil {
		t.Fatal(err)
	}
	if err := reg.Add(registry.Project{
		Name:      "demo",
		Language:  "go",
		RepoURL:   "https://github.com/acme/demo",
		Dir:       dir,
		Caps:      queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5},
		CreatedAt: time.Date(2026, 10, 8, 14, 0, 0, 0, time.UTC),
	}); err != nil {
		t.Fatal(err)
	}
	return &puHarness{
		t:     t,
		tmp:   tmp,
		reg:   reg,
		clock: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC),
	}
}

// runner returns the harness's scripted git runner.
func (h *puHarness) runner() *runner.Fake {
	if h.run == nil {
		h.run = &runner.Fake{Script: map[string]runner.Result{
			"git rev-parse --abbrev-ref HEAD": {Stdout: "main\n"},
			"git rev-parse HEAD":              {Stdout: "aaa111\n"},
			"git rev-parse origin/main":       {Stdout: "aaa111\n"},
			"git status --porcelain":          {Stdout: ""},
		}}
	}
	return h.run
}

func (h *puHarness) now() time.Time {
	h.mu.Lock()
	defer h.mu.Unlock()
	now := h.clock
	h.clock = h.clock.Add(time.Second)
	return now
}

func (h *puHarness) newID() string {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.ids++
	return fmt.Sprintf("id-%d", h.ids)
}

func (h *puHarness) post(ctx context.Context, project, kind, headline, numbers string) (st telegram.Status, err error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
	return
}

func (h *puHarness) spawnFunc() func(context.Context, claude.Launch) (claude.Process, error) {
	return func(ctx context.Context, l claude.Launch) (claude.Process, error) {
		var f *claude.Fake
		if h.starter == nil {
			f = claude.NewFake()
		}
		h.mu.Lock()
		h.spawns = append(h.spawns, l)
		if f != nil {
			h.fakes = append(h.fakes, f)
		}
		h.mu.Unlock()
		if h.starter != nil {
			return h.starter(ctx, l)
		}
		return f, nil
	}
}

// open builds and opens the daemon over the harness.
func (h *puHarness) open() *Daemon {
	h.t.Helper()
	deps := Deps{
		Registry: h.reg,
		Runner:   h.runner(),
		Spawn:    h.spawnFunc(),
		Post:     h.post,
		GitHubDo: func(*http.Request) (*http.Response, error) {
			return nil, errors.New("github: disabled")
		},
		Now:   h.now,
		NewID: h.newID,
		Config: Config{
			ClaudeBin:   "/opt/claude/bin/claude",
			MorphBin:    "/usr/local/bin/morph",
			ProjectsDir: filepath.Join(h.tmp, "projects"),
			MCPBaseURL:  "http://127.0.0.1:7181",
			Defaults:    queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4},
			MaxParallel: 1,
			Loop: supervisor.Config{
				MaxResumesPerHour: 3,
				StallMinutes:      30,
				StretchUSD:        30,
				UsageAlertPercent: 50,
			},
		},
	}
	d, err := Open(context.Background(), deps)
	if err != nil {
		h.t.Fatal(err)
	}
	return d
}

// puPoll calls f every 10 ms until it reports true or limit elapses, then
// returns the last value it saw.
func puPoll[T any](t *testing.T, limit time.Duration, what string, f func() (T, bool)) T {
	t.Helper()
	deadline := time.Now().Add(limit)
	for {
		last, ok := f()
		if ok {
			return last
		}
		if time.Now().After(deadline) {
			t.Fatalf("poll %s: last value %#v", what, last)
		}
		time.Sleep(10 * time.Millisecond)
	}
}

// puWait calls d.Wait and fails if it does not return within five seconds.
func puWait(t *testing.T, d *Daemon, project string) {
	t.Helper()
	done := make(chan struct{})
	go func() {
		d.Wait(project)
		close(done)
	}()
	select {
	case <-done:
	case <-time.After(5 * time.Second):
		t.Fatalf("Wait(%s) did not return within 5s", project)
	}
}

// puEntry is the compared part of an eventlog entry.
type puEntry struct {
	Seq int
	Dir string
	Msg string
}

func puEntries(es []eventlog.Entry) []puEntry {
	out := make([]puEntry, 0, len(es))
	for _, e := range es {
		out = append(out, puEntry{Seq: int(e.Seq), Dir: e.Dir, Msg: string(e.Msg)})
	}
	return out
}

// puReadEntries reads a JSONL log file into entries.
func puReadEntries(t *testing.T, path string) []eventlog.Entry {
	t.Helper()
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	text := strings.TrimSuffix(string(data), "\n")
	if text == "" {
		return nil
	}
	lines := strings.Split(text, "\n")
	out := make([]eventlog.Entry, 0, len(lines))
	for i, line := range lines {
		var e eventlog.Entry
		if err := json.Unmarshal([]byte(line), &e); err != nil {
			t.Fatalf("%s line %d: %v", path, i+1, err)
		}
		out = append(out, e)
	}
	return out
}

// puCompact strips the insignificant whitespace of b.
func puCompact(t *testing.T, b []byte) []byte {
	t.Helper()
	var buf bytes.Buffer
	if err := json.Compact(&buf, b); err != nil {
		t.Fatalf("compact: %v", err)
	}
	return buf.Bytes()
}

		Text:      "Which option do you want: A or B?",
		Header:    "A or B",
		Options:   []string{"Option A", "Option B"},

	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "state", st.State, "question")

	testhelp.Equal(t, "last post", posts[len(posts)-1],
		[4]string{"demo", "ask", "Which option do you want: A or B?", "1 Option A · 2 Option B"})

	testhelp.Equal(t, "answer", d.Answer("demo", 2), nil)

	w := fp.Written()

	if err := json.Unmarshal(w[len(w)-1], &gotLine); err != nil {
		t.Fatal(err)
	}
	_, msg28 := testhelp.ProbeLine(t, puProbe, 28)
	if err := json.Unmarshal(msg28, &wantLine); err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "written", gotLine, wantLine)

	if err != nil {
		t.Fatal(err)

	testhelp.Equal(t, "last entry dir", entries[len(entries)-1].Dir, "in")

	st, err = d.Status("demo")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "state", st.State, "busy")

	_ = h
}

func TestPumpExample5(t *testing.T) {
	h, d, fp := puExample1(t)

	h.runner().Script["git status --porcelain"] = runner.Result{Stdout: " M notes.md\n"}

	fp.Finish(4)
	puPoll(t, 2*time.Second, "spawns", func() (int, bool) {
		n := h.spawnCount()
		return n, n == 2
	})
	puWait(t, d, "demo")

	testhelp.Equal(t, "posts", h.postsFrom(3), [][4]string{
		{"demo", "watchdog", "P17: session exited (code 4), restarting", "restart 1 of 3 this hour"},
		{"demo", "start", "P17 resumed", "session id-1 · dirty tree (1 paths)"},
	})

	l := h.launch(1)
	testhelp.Equal(t, "session id", l.SessionID, "id-1")
	testhelp.Equal(t, "resume", l.Resume, true)
	testhelp.Equal(t, "budget", l.BudgetUSD, float64(30))
	testhelp.Equal(t, "mcp config", l.MCPConfigPath, filepath.Join(h.tmp, "state", "demo", "morphd-mcp.json"))

	tok, err := h.reg.ReadSecret("demo", "session-token")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "token", tok, "id-3")

	state := h.state()
	testhelp.Equal(t, "restarts", len(state.Loop.Restarts), 1)
	testhelp.Equal(t, "loop state", state.Loop.State, "running")
	if state.Loop.LastExit == nil {
		t.Fatal("state: last exit is nil")
	}
	if state.Loop.LastExit.At.IsZero() {
		t.Error("state: last exit at is zero")
	}
	le := *state.Loop.LastExit
	le.At = time.Time{}
	testhelp.Equal(t, "last exit", le, control.Exit{Code: 4, Stderr: "", Restarts: 1})

	es := puReadEntries(t, filepath.Join(h.tmp, "state", "demo", "sessions", "id-1.jsonl"))
	testhelp.Equal(t, "session log lines", len(es), 6)
	if len(es) == 6 {
		testhelp.Equal(t, "session log last", puEntries(es)[5], puEntry{Seq: 6, Dir: "exit", Msg: `{"code":4,"stderr":""}`})
	}

	st, err := d.Status("demo")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "state", st.State, "ready")
	testhelp.Equal(t, "session id", st.SessionID, "id-1")
	if st.LastExit == nil {
		t.Fatal("status: last exit is nil")
	}
	if st.LastExit.At.IsZero() {
		t.Error("status: last exit at is zero")
	}
	sle := *st.LastExit
	sle.At = time.Time{}
	testhelp.Equal(t, "status last exit", sle, control.Exit{Code: 4, Stderr: "", Restarts: 1})
}

	})

		t.Fatal("status: stop is nil")

	testhelp.Equal(t, "stop reason", st.Stop.Reason,
		"exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.")
	if st.LastExit == nil {
		t.Fatal("status: last exit is nil")
	}
	if st.LastExit.At.IsZero() {
		t.Error("status: last exit at is zero")
	}
	le := *st.LastExit
	le.At = time.Time{}
	testhelp.Equal(t, "last exit", le, control.Exit{
		Code:     1,
		Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
		Restarts: 3,
	})

	ev, err := d.Events("demo", 0, 10)
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "log entries", puEntries(ev.Entries), []puEntry{
		{Seq: 1, Dir: "in", Msg: string(puCompact(t, stream.User("/morph-orchestrator P17")))},
		{Seq: 2, Dir: "exit", Msg: `{"code":1,"stderr":"Error: Invalid session ID. Must be a valid UUID.\nsecond line"}`},
	})

	for _, sid := range []string{"id-1", "id-3", "id-5"} {
		es := puReadEntries(t, filepath.Join(h.tmp, "state", "demo", "sessions", sid+".jsonl"))
		testhelp.Equal(t, sid+" lines", len(es), 2)
	}

	_, err = d.Order("demo", "x")
	testhelp.Equal(t, "order error", errors.Is(err, control.ErrNoSession), true)
}
