func TestConfigAndMainExample4(t *testing.T) {
	_, err := Load(func(string) string { return "" }, map[string]string{})
	testhelp.Equal(t, "missing token error", err.Error(), "config: MORPHD_TOKEN is required")

	portEnv := map[string]string{"MORPHD_TOKEN": "x", "MORPHD_PORT": "80a"}
	_, err = Load(func(k string) string { return portEnv[k] }, map[string]string{})
	testhelp.Equal(t, "bad port error", err.Error(), "config: MORPHD_PORT: not a number: 80a")

	stallEnv := map[string]string{"MORPHD_TOKEN": "x", "MORPHD_STALL_MINUTES": "soon"}
	_, err = Load(func(k string) string { return stallEnv[k] }, map[string]string{})
	testhelp.Equal(t, "bad stall error", err.Error(), "config: MORPHD_STALL_MINUTES: not a number: soon")
}

func TestConfigAndMainExample5(t *testing.T) {
	f := cmFake{}
	h := Build(f, func(p string) bool { return p == "demo" }, func(p string) (string, bool) { return "s", p == "demo" }, "tok-7")

	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, httptest.NewRequest(http.MethodGet, "/healthz", nil))
	testhelp.Equal(t, "healthz code", rec.Code, http.StatusOK)
	testhelp.Equal(t, "healthz body", rec.Body.String(), `{"status":"ok"}`+"\n")

	req := httptest.NewRequest(http.MethodGet, "/projects/demo/status", nil)
	req.Header.Set("Authorization", "Bearer tok-7")
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	testhelp.Equal(t, "status code", rec.Code, http.StatusOK)
	testhelp.Equal(t, "status body", rec.Body.String(), `{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}`+"\n")

	mcpReq := httptest.NewRequest(http.MethodPost, "/mcp/demo", strings.NewReader(`{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}`))
	mcpReq.Header.Set("Content-Type", "application/json")
	mcpReq.Header.Set("Accept", "application/json, text/event-stream")
	mcpReq.Header.Set("Authorization", "Bearer tok-7")
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, mcpReq)
	testhelp.Equal(t, "mcp code", rec.Code, http.StatusOK)
	var mcpPayload struct {
		Result struct {
			Tools []struct {
				Name string `json:"name"`
			} `json:"tools"`
		} `json:"result"`
	}
	if err := json.Unmarshal(rec.Body.Bytes(), &mcpPayload); err != nil {
		t.Fatalf("decode mcp body: %v (body %q)", err, rec.Body.String())
	}
	testhelp.Equal(t, "mcp tool count", len(mcpPayload.Result.Tools), 10)
	hasStatus := false
	for _, tool := range mcpPayload.Result.Tools {
		if tool.Name == "status" {
			hasStatus = true
		}
	}
	testhelp.Equal(t, "mcp has status tool", hasStatus, true)

	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, httptest.NewRequest(http.MethodGet, "/projects/demo/status", nil))
	testhelp.Equal(t, "unauthorized code", rec.Code, http.StatusUnauthorized)
}

func TestConfigAndMainExample6(t *testing.T) {
	b := [16]byte{0x96, 0x7e, 0x8f, 0x4d, 0xa2, 0xeb, 0x0a, 0xa6, 0x16, 0x8b, 0xb9, 0x05, 0x4a, 0x1c, 0x3d, 0x7e}
	cases := []struct {
		in   [16]byte
		want string
	}{
		{b, "967e8f4d-a2eb-4aa6-968b-b9054a1c3d7e"},
		{[16]byte{}, "00000000-0000-4000-8000-000000000000"},
		{[16]byte{0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff}, "ffffffff-ffff-4fff-bfff-ffffffffffff"},
		{[16]byte{0x00, 0x11, 0x22, 0x33, 0x44, 0x55, 0x66, 0x77, 0x88, 0x99, 0xaa, 0xbb, 0xcc, 0xdd, 0xee, 0xff}, "00112233-4455-4677-8899-aabbccddeeff"},
	}
	for _, c := range cases {
		got := UUIDv4(c.in)
		testhelp.Equal(t, "UUIDv4", got, c.want)
		testhelp.Equal(t, "UUIDv4 length", len(got), 36)
		testhelp.Equal(t, "UUIDv4 lower case", strings.ToLower(got), got)
	}
}

func TestConfigAndMainExample7(t *testing.T) {
	shape := regexp.MustCompile(`^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$`)
	a := newID()
	b := newID()
	testhelp.Equal(t, "a matches", shape.MatchString(a), true)
	testhelp.Equal(t, "b matches", shape.MatchString(b), true)
	testhelp.Equal(t, "a length", len(a), 36)
	testhelp.Equal(t, "b length", len(b), 36)
	testhelp.Equal(t, "a != b", a != b, true)
	testhelp.Equal(t, "a has no upper case", strings.ToLower(a), a)
	testhelp.Equal(t, "b has no upper case", strings.ToLower(b), b)
}
