package supervisor

import (
	"encoding/json"

	"morphstudio/gitrules"

func plQueue(t *testing.T) queue.Queue {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var p queue.Plan
	if err := json.Unmarshal(data, &p); err != nil {
		t.Fatal(err)
	}
	q, err := queue.Load(p, queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5})
	if err != nil {
		t.Fatal(err)
	}
	return q
}

func plIDs() func() string {
	i := 0
	return func() string {
		i++
		return "id-" + strconv.Itoa(i)
	}
}

func plFixedID(id string) func() string {
	return func() string { return id }
}

func plTime(h, m, s int) time.Time {
	return time.Date(2026, 10, 8, h, m, s, 0, time.UTC)
}

func plRunningP17(t *testing.T) *Loop {
	t.Helper()
	l := New("demo", plQueue(t))
	if _, err := l.Begin(plTime(15, 0, 0)); err != nil {
		t.Fatal(err)
	}
	l.StartChecked(gitrules.Start{Mode: "fresh", Local: "aaa111", Remote: "aaa111"}, plIDs(), plTime(15, 0, 1))
	return l
}

func plRunningP18(t *testing.T) *Loop {
	t.Helper()
	q := plQueue(t)
	q.Index = 1
	q.State = "running"
	l := New("demo", q)
	l.State = "running"
	l.Phase = "P18"
	l.SessionID = "id-x"
	l.SessionUSD = 0.5
	l.StretchUSD = 1.0
	return l
}

func plStartingP18(t *testing.T) *Loop {
	t.Helper()
	q := plQueue(t)
	q.Index = 1
	q.State = "running"
	l := New("demo", q)
	l.State = "starting"
	l.Phase = "P18"
	return l
}

func TestPhaseLoopExample1(t *testing.T) {
	l := New("demo", plQueue(t))
	l.Restarts = []time.Time{plTime(14, 0, 0)}
	l.LastExit = &control.Exit{Code: 7, At: plTime(14, 0, 0), Stderr: "old", Restarts: 1}

	actions, err := l.Begin(plTime(15, 0, 0))
	testhelp.Equal(t, "error", err, nil)
	testhelp.Equal(t, "actions", actions, []Action{{Kind: "start_check", Phase: "P17"}})
	testhelp.Equal(t, "state", l.State, "starting")
	testhelp.Equal(t, "phase", l.Phase, "P17")
	testhelp.Equal(t, "queue state", l.Queue.State, "running")
	testhelp.Equal(t, "restarts", l.Restarts, []time.Time(nil))
	testhelp.Equal(t, "last exit nil", l.LastExit == nil, true)

	actions2, err2 := l.Begin(plTime(15, 0, 0))
	testhelp.Equal(t, "actions2", actions2, []Action(nil))
	if err2 == nil {
		t.Fatal("second Begin: want error, got nil")
	}
	testhelp.Equal(t, "error text", err2.Error(), `queue: cannot start in state "running"`)
	testhelp.Equal(t, "state unchanged", l.State, "starting")
	testhelp.Equal(t, "phase unchanged", l.Phase, "P17")
}

func TestPhaseLoopExample2(t *testing.T) {
	l := New("demo", plQueue(t))
	if _, err := l.Begin(plTime(15, 0, 0)); err != nil {
		t.Fatal(err)
	}

		{Kind: "spawn", Phase: "P17", SessionID: "id-1", Resume: false, BudgetUSD: 30},
		{Kind: "first_line", Line: "/morph-orchestrator P17"},
		{Kind: "post", Milestone: control.Milestone{Kind: "start", Headline: "P17 started", Numbers: "session id-1 · cap $30 · 3 h"}},
	})
	testhelp.Equal(t, "state", l.State, "running")
	testhelp.Equal(t, "session", l.SessionID, "id-1")
	testhelp.Equal(t, "started at", l.PhaseStartedAt, plTime(15, 0, 1))
	testhelp.Equal(t, "restarts kept", l.Restarts, []time.Time{plTime(14, 50, 0)})
	if l.LastExit == nil {
		t.Fatal("last exit: want value, got nil")
	}
	testhelp.Equal(t, "last exit kept", *l.LastExit, exit)
}

func TestPhaseLoopExample6(t *testing.T) {
	l := plRunningP17(t)
	actions, err := l.PhaseDone("P16", "P17")
	testhelp.Equal(t, "actions", actions, []Action(nil))
	if !errors.Is(err, ErrPhaseMismatch) {
		t.Fatalf("want ErrPhaseMismatch, got %v", err)
	}
	testhelp.Equal(t, "text", err.Error(), `phase mismatch: phase_done "P16" while running "P17"`)

	l2 := plRunningP17(t)
	l2.State = "waiting"
	actions2, err2 := l2.PhaseDone("P17", "P18")
	testhelp.Equal(t, "actions2", actions2, []Action(nil))
	if !errors.Is(err2, ErrPhaseMismatch) {
		t.Fatalf("want ErrPhaseMismatch, got %v", err2)
	}
	testhelp.Equal(t, "text2", err2.Error(), `phase mismatch: phase_done "P17" while waiting "P17"`)
}

func TestPhaseLoopExample7(t *testing.T) {
	l := plRunningP17(t)
	l.SessionUSD = 1.92

	actions := l.EndChecked(
		gitrules.End{OK: false, Problems: []string{"not pushed: HEAD bbb222, origin/main aaa111"}},
		plTime(16, 0, 0),
	)
	testhelp.Equal(t, "actions", actions, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17 not pushed", Numbers: "not pushed: HEAD bbb222, origin/main aaa111"}},
	})
	testhelp.Equal(t, "state", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("stop: want value, got nil")
	}
	testhelp.Equal(t, "stop", *l.Stop, control.Stop{
		Kind:   "not_pushed",
		Phase:  "P17",
		Reason: "not pushed: HEAD bbb222, origin/main aaa111",
		At:     plTime(16, 0, 0),
	})
	testhelp.Equal(t, "stretch", l.StretchUSD, 0.0)
	testhelp.Equal(t, "queue state", l.Queue.State, "stopped")

	contActions, contErr := l.Continue(plTime(16, 5, 0))
	testhelp.Equal(t, "continue error", contErr, nil)
	testhelp.Equal(t, "continue actions", contActions, []Action{{Kind: "end_check", Phase: "P17"}})
	testhelp.Equal(t, "state after", l.State, "running")
	testhelp.Equal(t, "stop cleared", l.Stop == nil, true)
	testhelp.Equal(t, "queue state after", l.Queue.State, "running")
	testhelp.Equal(t, "index after", l.Queue.Index, 0)

	endActions := l.EndChecked(
		gitrules.End{OK: true, Problems: []string{}},
		plTime(16, 10, 0),
	)
	testhelp.Equal(t, "end actions", endActions, []Action{
		{Kind: "kill"},
		{Kind: "start_check", Phase: "P18"},
	})
	testhelp.Equal(t, "stretch after", l.StretchUSD, 1.92)
	testhelp.Equal(t, "session after", l.SessionUSD, 0.0)
}

func TestPhaseLoopExample8(t *testing.T) {
	l := plRunningP18(t)
	actions := l.EndChecked(
		gitrules.End{OK: true, Local: "eee", Remote: "eee", Problems: []string{}},
		plTime(17, 0, 0),
	)
	testhelp.Equal(t, "actions", actions, []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P18 done: stop after smoke", Numbers: "next P19 · waiting for continue"}},
	})
	testhelp.Equal(t, "state", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("stop: want value, got nil")
	}
	testhelp.Equal(t, "stop kind", l.Stop.Kind, "smoke")
	testhelp.Equal(t, "stretch", l.StretchUSD, 1.5)

	cont, contErr := l.Continue(plTime(17, 30, 0))
	testhelp.Equal(t, "continue error", contErr, nil)
	testhelp.Equal(t, "continue actions", cont, []Action{{Kind: "start_check", Phase: "P19"}})
	testhelp.Equal(t, "state after continue", l.State, "starting")
	testhelp.Equal(t, "phase after continue", l.Phase, "P19")
	testhelp.Equal(t, "stop after continue", l.Stop == nil, true)

	l.StartChecked(
		gitrules.Start{Mode: "fresh", Local: "eee", Remote: "eee"},
		plIDs(),
		plTime(17, 30, 1),
	)
	end2 := l.EndChecked(
		gitrules.End{OK: true, Problems: []string{}},
		plTime(18, 0, 0),
	)
	testhelp.Equal(t, "end2", end2, []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{Kind: "end", Headline: "P19 done: plan complete", Numbers: "stretch $1.5000"}},
	})
	testhelp.Equal(t, "state2", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("stop2: want value, got nil")
	}
	testhelp.Equal(t, "stop2 kind", l.Stop.Kind, "end")
	testhelp.Equal(t, "queue state2", l.Queue.State, "done")

	_, err2 := l.Continue(plTime(18, 5, 0))
	if err2 == nil {
		t.Fatal("want error, got nil")
	}
	testhelp.Equal(t, "err2 text", err2.Error(), `queue: nothing to continue (state "done")`)
}

func TestPhaseLoopExample9(t *testing.T) {
	l := plRunningP17(t)
	actions := l.WaitOperator("gate", "forecast $1.4 over the cap", "", plTime(15, 10, 0))
	testhelp.Equal(t, "actions", actions, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: wait_operator gate", Numbers: "forecast $1.4 over the cap"}},
	})
	testhelp.Equal(t, "state", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("stop: want value, got nil")
	}
	testhelp.Equal(t, "stop", *l.Stop, control.Stop{
		Kind:   "gate",
		Phase:  "P17",
		Reason: "forecast $1.4 over the cap",
		At:     plTime(15, 10, 0),
	})

	l2 := plRunningP17(t)
	actions2 := l2.WaitOperator("emergency", "card x red after fix", "https://github.com/o/r/issues/12", plTime(15, 20, 0))
	testhelp.Equal(t, "actions2", actions2, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: wait_operator emergency", Numbers: "card x red after fix · https://github.com/o/r/issues/12"}},
	})
	if l2.Stop == nil {
		t.Fatal("stop2: want value, got nil")
	}
	testhelp.Equal(t, "issue url", l2.Stop.IssueURL, "https://github.com/o/r/issues/12")

	restart, err := l.Restart(plTime(15, 30, 0))
	testhelp.Equal(t, "restart actions", restart, []Action(nil))
	if !errors.Is(err, control.ErrNoSession) {
		t.Fatalf("want ErrNoSession, got %v", err)
	}

	l3 := plRunningP17(t)
	exit := control.Exit{Code: 3, At: plTime(15, 5, 0), Stderr: "", Restarts: 1}
	l3.Restarts = []time.Time{plTime(15, 5, 0)}
	l3.LastExit = &exit
	restart3, err3 := l3.Restart(plTime(15, 40, 0))
	testhelp.Equal(t, "restart3 error", err3, nil)
	testhelp.Equal(t, "restart3 actions", restart3, []Action{
		{Kind: "kill"},
		{Kind: "start_check", Phase: "P17"},
	})
	testhelp.Equal(t, "state3", l3.State, "starting")
	testhelp.Equal(t, "session3", l3.SessionID, "")
	testhelp.Equal(t, "restarts3", l3.Restarts, []time.Time(nil))
	if l3.LastExit == nil {
		t.Fatal("last exit3: want value, got nil")
	}
	testhelp.Equal(t, "last exit3", *l3.LastExit, exit)
}
