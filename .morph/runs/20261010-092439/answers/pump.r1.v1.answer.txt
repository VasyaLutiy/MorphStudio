// Package daemon launches claude sessions, pumps their stream output through
// the session machine and the supervisor loop, and exposes control.Control.
package daemon

import (
	"context"
	"encoding/json"
	"strings"

	"morphstudio/claude"
	"morphstudio/eventlog"
	"morphstudio/stream"
	"morphstudio/supervisor"
)

func init() {
	startPump = func(d *Daemon, ctx context.Context, project string, p claude.Process, l *eventlog.Log) {
		pr := d.projects[project]
		if pr == nil {
			return
		}
		done := make(chan struct{})
		pr.pumps[p] = done
		go func() {
			d.pump(ctx, project, p, l)
			close(done)
			d.mu.Lock()
			if cur := d.projects[project]; cur != nil {
				delete(cur.pumps, p)
			}
			d.mu.Unlock()
		}()
	}
}

// pump reads every line of p, records it, folds it into its project's session
// machine and supervisor loop, and runs the actions they return. When the
// lines end it consumes the exit status: the exit goes into the process's log
// and, only while p is still the project's current process, into the loop.
func (d *Daemon) pump(ctx context.Context, project string, p claude.Process, l *eventlog.Log) {
	for line := range p.Lines() {
		d.pumpLine(ctx, project, p, l, line)
	}
	st := <-p.Exit()
	d.pumpExit(ctx, project, p, l, st)
}

// pumpLine folds one stdout line into the process's log, the session machine
// and the loop. The whole handling runs under d.mu so a caller that observes
// the line through Events also observes its effects.
func (d *Daemon) pumpLine(ctx context.Context, project string, p claude.Process, l *eventlog.Log, line []byte) {
	d.mu.Lock()
	defer d.mu.Unlock()

	pr := d.projects[project]
	if pr == nil {
		return
	}
	now := d.deps.Now()
	msg := json.RawMessage(line)
	if !json.Valid(line) {
		if b, err := json.Marshal(string(line)); err == nil {
			msg = b
		}
	}
	if l != nil {
		_, _ = l.Append("out", msg, now)
	}
	ev, err := stream.Parse(line)
	if err != nil {
		return
	}
	if pr.proc != p || pr.machine == nil {
		return
	}
	pr.state.Loop.Event(now)
	macts := pr.machine.Apply(ev, now)

	var lacts []supervisor.Action
	switch ev.Type {
	case "result":
		if ev.Result != nil {
			lacts = pr.state.Loop.Turn(*ev.Result, now, d.deps.Config.Loop)
		}
	case "rate_limit_event":
		if ev.Limits != nil {
			lacts = pr.state.Loop.Limits(*ev.Limits, now, d.deps.Config.Loop)
		}
	}

	for _, a := range macts {
		switch a.Kind {
		case "write":
			d.writeLine(project, a.Line)
		case "ask":
			if d.deps.Post != nil {
				_, _ = d.deps.Post(ctx, project, "ask", a.Headline, a.Numbers)
			}
		case "turn":
			headline := a.Headline
			if headline == "" {
				headline = "(no text)"
			} else {
				headline = truncateRunes(headline, 200)
			}
			if d.deps.Post != nil {
				_, _ = d.deps.Post(ctx, project, "idle", pr.state.Loop.Phase+" turn ended: "+headline, a.Numbers)
			}
		case "limit":
			// Nothing to do here; the loop has already recorded the limits.
		}
	}
	d.execute(ctx, project, lacts)
}

// pumpExit records the process's exit in its own log and, when p is still the
// project's current process, lets the loop decide what to do about it. A
// process replaced by a newer spawn is pumped to its end and its exit is
// logged but changes nothing else.
func (d *Daemon) pumpExit(ctx context.Context, project string, p claude.Process, l *eventlog.Log, st claude.ExitStatus) {
	d.mu.Lock()
	defer d.mu.Unlock()

	pr := d.projects[project]
	if pr == nil {
		return
	}
	now := d.deps.Now()
	payload, err := json.Marshal(struct {
		Code   int    `json:"code"`
		Stderr string `json:"stderr"`
	}{Code: st.Code, Stderr: stderrTail(st.Stderr)})
	if err == nil && l != nil {
		_, _ = l.Append("exit", json.RawMessage(payload), now)
	}
	if pr.proc != p {
		return
	}
	pr.proc = nil
	d.execute(ctx, project, pr.state.Loop.Exited(st.Code, st.Stderr, now, d.deps.Config.Loop))
}

// Wait blocks until every pump of project whose process is no longer the
// current one has returned. It returns immediately when there is none.
func (d *Daemon) Wait(project string) {
	d.mu.Lock()
	var dones []chan struct{}
	if pr := d.projects[project]; pr != nil {
		for p, done := range pr.pumps {
			if p != pr.proc {
				dones = append(dones, done)
			}
		}
	}
	d.mu.Unlock()
	for _, done := range dones {
		<-done
	}
}

// stderrTail returns the last 20 lines of s, after trimming one trailing
// newline, or "" when s is empty.
func stderrTail(s string) string {
	if s == "" {
		return ""
	}
	s = strings.TrimSuffix(s, "\n")
	lines := strings.Split(s, "\n")
	if len(lines) > 20 {
		lines = lines[len(lines)-20:]
	}
	return strings.Join(lines, "\n")
}

// truncateRunes returns s cut to at most n runes.
func truncateRunes(s string, n int) string {
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n])
}
