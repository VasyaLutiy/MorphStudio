package daemon

import (
	"context"
	"encoding/json"
	"strings"

	"morphstudio/claude"
	"morphstudio/eventlog"
	"morphstudio/stream"
	"morphstudio/supervisor"
)

func init() {
	startPump = func(d *Daemon, ctx context.Context, project string, p claude.Process, l *eventlog.Log) {
		pr := d.projects[project]
		if pr == nil {
			return
		}
		if pr.pumps == nil {
			pr.pumps = make(map[claude.Process]chan struct{})
		}
		done := make(chan struct{})
		pr.pumps[p] = done
		go func() {
			d.pump(ctx, project, p, l)
			close(done)
			d.mu.Lock()
			delete(pr.pumps, p)
			d.mu.Unlock()
		}()
	}
}

func (d *Daemon) pump(ctx context.Context, project string, p claude.Process, l *eventlog.Log) {
	for line := range p.Lines() {
		d.pumpLine(ctx, project, p, l, line)
	}
	st := <-p.Exit()
	d.pumpExit(ctx, project, p, l, st)
}

// pumpLine folds one stdout line into the process's log, the session machine
// and the loop. The whole handling runs under d.mu so a caller that observes
// the line through Events also observes its effects.
func (d *Daemon) pumpLine(ctx context.Context, project string, p claude.Process, l *eventlog.Log, line []byte) {
	d.mu.Lock()
	defer d.mu.Unlock()

	pr := d.projects[project]
	if pr == nil {
		return
	}
	now := d.deps.Now()

	msg := json.RawMessage(line)
	if !json.Valid(line) {
		b, err := json.Marshal(string(line))
		if err != nil {
			return
		}
		msg = json.RawMessage(b)
	}
	_, _ = l.Append("out", msg, now)

	ev, err := stream.Parse(line)
	if err != nil {
		return
	}
	if pr.proc != p {
		return
	}
	pr.state.Loop.Event(now)
	if pr.machine == nil {
		return
	}
	macts := pr.machine.Apply(ev, now)

	var lacts []supervisor.Action
	switch ev.Type {
	case "result":
		if ev.Result != nil {
			lacts = pr.state.Loop.Turn(*ev.Result, now, d.deps.Config.Loop)
		}
	case "rate_limit_event":
		if ev.Limits != nil {
			lacts = pr.state.Loop.Limits(*ev.Limits, now, d.deps.Config.Loop)
		}
	}

	for _, a := range macts {
		switch a.Kind {
		case "write":
			d.writeLine(project, a.Line)
		case "ask":
			if d.deps.Post != nil {
				_, _ = d.deps.Post(ctx, project, "ask", a.Headline, a.Numbers)
			}
		case "turn":
			headline := a.Headline
			if headline == "" {
				headline = "(no text)"
			} else {
				headline = truncateRunes(headline, 200)
			}
			if d.deps.Post != nil {
				_, _ = d.deps.Post(ctx, project, "idle", pr.state.Loop.Phase+" turn ended: "+headline, a.Numbers)
			}
		case "limit":
			// nothing
		}
	}
	d.execute(ctx, project, lacts)
}

// pumpExit records the process's exit in its own log and, when the process is
// still the project's current one, lets the loop decide what to do about it.
func (d *Daemon) pumpExit(ctx context.Context, project string, p claude.Process, l *eventlog.Log, st claude.ExitStatus) {
	d.mu.Lock()
	defer d.mu.Unlock()

	pr := d.projects[project]
	if pr == nil {
		return
	}
	now := d.deps.Now()
	payload, err := json.Marshal(struct {
		Code   int    `json:"code"`
		Stderr string `json:"stderr"`
	}{Code: st.Code, Stderr: stderrTail(st.Stderr)})
	if err == nil {
		_, _ = l.Append("exit", json.RawMessage(payload), now)
	}
	if pr.proc != p {
		return
	}
	pr.proc = nil
	d.execute(ctx, project, pr.state.Loop.Exited(st.Code, st.Stderr, now, d.deps.Config.Loop))
}

// Wait blocks until every pump of project whose process is no longer current
// has returned. It returns immediately when there is none.
func (d *Daemon) Wait(project string) {
	d.mu.Lock()
	pr := d.projects[project]
	var dones []chan struct{}
	if pr != nil {
		for proc, done := range pr.pumps {
			if proc != pr.proc {
				dones = append(dones, done)
			}
		}
	}
	d.mu.Unlock()
	for _, done := range dones {
		<-done
	}
}

// stderrTail returns the last 20 lines of stderr, after trimming one trailing
// newline, or "" when stderr is empty.
func stderrTail(stderr string) string {
	if stderr == "" {
		return ""
	}
	stderr = strings.TrimSuffix(stderr, "\n")
	lines := strings.Split(stderr, "\n")
	if len(lines) > 20 {
		lines = lines[len(lines)-20:]
	}
	return strings.Join(lines, "\n")
}

// truncateRunes returns s cut to at most n runes.
func truncateRunes(s string, n int) string {
	r := []rune(s)
	if len(r) <= n {
		return s
	}
	return string(r[:n])
}
