func TestConfigAndMainExample5(t *testing.T) {
	f := cmFake{}
	known := func(p string) bool { return p == "demo" }
	sessionFn := func(p string) (string, bool) { return "s", p == "demo" }
	h := Build(f, known, sessionFn, "tok-7")

	req := httptest.NewRequest(http.MethodGet, "/healthz", nil)
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	testhelp.Equal(t, "healthz code", rec.Code, http.StatusOK)
	testhelp.Equal(t, "healthz body", rec.Body.String(), `{"status":"ok"}`+"\n")

	req = httptest.NewRequest(http.MethodGet, "/projects/demo/status", nil)
	req.Header.Set("Authorization", "Bearer tok-7")
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	testhelp.Equal(t, "status code", rec.Code, http.StatusOK)
	wantStatus := `{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}` + "\n"
	testhelp.Equal(t, "status body", rec.Body.String(), wantStatus)

	body := `{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}`
	req = httptest.NewRequest(http.MethodPost, "/mcp/demo", strings.NewReader(body))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Accept", "application/json, text/event-stream")
	req.Header.Set("Authorization", "Bearer tok-7")
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	testhelp.Equal(t, "mcp code", rec.Code, http.StatusOK)
	var mcp struct {
		Result struct {
			Tools []struct {
				Name string `json:"name"`
			} `json:"tools"`
		} `json:"result"`
	}
	if err := json.Unmarshal(rec.Body.Bytes(), &mcp); err != nil {
		t.Fatalf("mcp decode: %v", err)
	}
	testhelp.Equal(t, "mcp tool count", len(mcp.Result.Tools), 10)
	found := false
	for _, tool := range mcp.Result.Tools {
		if tool.Name == "status" {
			found = true
		}
	}
	testhelp.Equal(t, "mcp status tool", found, true)

	req = httptest.NewRequest(http.MethodGet, "/projects/demo/status", nil)
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	testhelp.Equal(t, "unauthorized code", rec.Code, http.StatusUnauthorized)
}

func TestConfigAndMainExample6(t *testing.T) {
	got1 := UUIDv4([16]byte{0x96, 0x7e, 0x8f, 0x4d, 0xa2, 0xeb, 0x0a, 0xa6, 0x16, 0x8b, 0xb9, 0x05, 0x4a, 0x1c, 0x3d, 0x7e})
	testhelp.Equal(t, "UUIDv4 probe", got1, "967e8f4d-a2eb-4aa6-968b-b9054a1c3d7e")
	testhelp.Equal(t, "probe len", len(got1), 36)

	got2 := UUIDv4([16]byte{})
	testhelp.Equal(t, "UUIDv4 zero", got2, "00000000-0000-4000-8000-000000000000")
	testhelp.Equal(t, "zero len", len(got2), 36)

	got3 := UUIDv4([16]byte{0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff})
	testhelp.Equal(t, "UUIDv4 ff", got3, "ffffffff-ffff-4fff-bfff-ffffffffffff")
	testhelp.Equal(t, "ff len", len(got3), 36)

	got4 := UUIDv4([16]byte{0x00, 0x11, 0x22, 0x33, 0x44, 0x55, 0x66, 0x77, 0x88, 0x99, 0xaa, 0xbb, 0xcc, 0xdd, 0xee, 0xff})
	testhelp.Equal(t, "UUIDv4 sequence", got4, "00112233-4455-4677-8899-aabbccddeeff")
	testhelp.Equal(t, "sequence len", len(got4), 36)
}

func TestConfigAndMainExample7(t *testing.T) {
	re := regexp.MustCompile(`^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$`)
	a := newID()
	b := newID()
	testhelp.Equal(t, "newID a regexp", re.MatchString(a), true)
	testhelp.Equal(t, "newID a len", len(a), 36)
	testhelp.Equal(t, "newID b regexp", re.MatchString(b), true)
	testhelp.Equal(t, "newID b len", len(b), 36)
	testhelp.Equal(t, "newID distinct", a != b, true)
	testhelp.Equal(t, "newID a lower", strings.ContainsAny(a, "ABCDEFGHIJKLMNOPQRSTUVWXYZ"), false)
	testhelp.Equal(t, "newID b lower", strings.ContainsAny(b, "ABCDEFGHIJKLMNOPQRSTUVWXYZ"), false)
}
