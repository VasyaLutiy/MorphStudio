func ccMarshal(t *testing.T, v any) string {
	t.Helper()
	b, err := json.Marshal(v)
	if err != nil {
		t.Fatal(err)
	}
	return string(b)
}

func TestControlContractExample1(t *testing.T) {
	s := Status{
		State:     "busy",
		Phase:     "P18",
		Minutes:   37,
		CostUSD:   1.92,
		FiveHour:  22,
		SevenDay:  60,
		SessionID: "s-1",
		Queue: QueueView{
			State:    "running",
			Approved: "7b31dfe",
			Index:    1,
			Phases:   3,
			Current:  "P18",
		},
		Repo: github.Access{
			Checked:   true,
			Reachable: true,
			Push:      true,
			CheckedAt: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC),
		},
	}
	got := ccMarshal(t, s)
	want := `{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}`
	testhelp.Equal(t, "example 1", got, want)
	if strings.Contains(got, "stop") {
		t.Errorf("example 1: unexpected stop key in %s", got)
	}
}

func TestControlContractExample2(t *testing.T) {
	s := Status{
		State: "waiting",
		Stop: &Stop{
			Kind:   "not_pushed",
			Phase:  "P17",
			Reason: "not pushed: HEAD bbb222, origin/main aaa111",
			At:     time.Date(2026, 10, 8, 16, 0, 0, 0, time.UTC),
		},
	}
	got := ccMarshal(t, s)
	suffix := `,"stop":{"kind":"not_pushed","phase":"P17","reason":"not pushed: HEAD bbb222, origin/main aaa111","at":"2026-10-08T16:00:00Z"}}`
	if !strings.HasSuffix(got, suffix) {
		t.Errorf("example 2: does not end with %s:\ngot:  %#v\nwant: %#v", suffix, got, got[:len(got)-len(suffix)]+suffix)
	}
	if strings.Contains(got, "issue_url") {
		t.Errorf("example 2: unexpected issue_url in %s", got)
	}
}

func TestControlContractExample5(t *testing.T) {
	type code struct {
		Status int
		Name   string
	}
	got := []code{
		ccCode(Code(nil)),
		ccCode(Code(ErrUnknownProject)),
		ccCode(Code(fmt.Errorf("wrap: %w", ErrBadOption))),
		ccCode(Code(ErrBadToken)),
		ccCode(Code(ErrExists)),
		ccCode(Code(ErrNoSession)),
		ccCode(Code(ErrBusy)),
		ccCode(Code(ErrNoQuestion)),
		ccCode(Code(ErrNotBusy)),
		ccCode(Code(ErrNotWaiting)),
		ccCode(Code(ErrBadInput)),
		ccCode(Code(errors.New("disk full"))),
	}
	want := []code{
		{200, ""},
		{404, "unknown_project"},
		{400, "bad_option"},
		{401, "bad_session_token"},
		{409, "exists"},
		{409, "no_session"},
		{409, "busy"},
		{409, "no_question"},
		{409, "not_busy"},
		{409, "not_waiting"},
		{400, "bad_input"},
		{500, "internal"},
	}
	testhelp.Equal(t, "example 5", got, want)
}

func ccCode(status int, name string) struct {
	Status int
	Name   string
} {
	return struct {
		Status int
		Name   string
	}{status, name}
}

func TestControlContractExample8(t *testing.T) {
	idle := Status{State: "idle", LimitsUnknown: true}
	got := ccMarshal(t, idle)
	want := `{"state":"idle","phase":"","minutes":0,"cost_usd":0,"five_hour":null,"seven_day":null,"session_id":"","queue":{"state":"","approved":"","index":0,"phases":0,"current":"","reason":""},"repo":{"checked":false,"reachable":false,"push":false,"reason":"","checked_at":"0001-01-01T00:00:00Z"}}`
	testhelp.Equal(t, "example 8 idle", got, want)

	busy := Status{
		State:         "busy",
		Phase:         "S1",
		Minutes:       2,
		CostUSD:       0.03,
		FiveHour:      77,
		SevenDay:      72,
		LimitsUnknown: true,
		SessionID:     "s-9",
	}
	got2 := ccMarshal(t, busy)
	hold := `,"five_hour":null,"seven_day":null,"session_id":"s-9",`
	if !strings.Contains(got2, hold) {
		t.Errorf("example 8 busy: missing %s in %s", hold, got2)
	}
	begin := `{"state":"busy","phase":"S1","minutes":2,"cost_usd":0.03,`
	if !strings.HasPrefix(got2, begin) {
		t.Errorf("example 8 busy: does not begin with %s:\ngot: %#v", begin, got2)
	}
	if strings.Contains(got2, "77") || strings.Contains(got2, "72") {
		t.Errorf("example 8 busy: leaked ints 77/72 in %s", got2)
	}
}

func TestControlContractExample10(t *testing.T) {
	e := Exit{
		Code:     1,
		At:       time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
		Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
		Restarts: 3,
	}
	got := ccMarshal(t, e)
	want := `{"code":1,"at":"2026-10-09T08:14:02Z","stderr":"Error: Invalid session ID. Must be a valid UUID.","restarts":3}`
	testhelp.Equal(t, "example 10 exit", got, want)

	s := Status{
		State:         "waiting",
		Phase:         "S1",
		LimitsUnknown: true,
		Stop: &Stop{
			Kind:   "crash",
			Phase:  "S1",
			Reason: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.",
			At:     time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
		},
		LastExit: &Exit{
			Code:     1,
			At:       time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
			Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
			Restarts: 3,
		},
	}
	got2 := ccMarshal(t, s)
	suffix := `,"stop":{"kind":"crash","phase":"S1","reason":"exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.","at":"2026-10-09T08:14:02Z"},"last_exit":{"code":1,"at":"2026-10-09T08:14:02Z","stderr":"Error: Invalid session ID. Must be a valid UUID.","restarts":3}}`
	if !strings.HasSuffix(got2, suffix) {
		t.Errorf("example 10 status: does not end with %s:\ngot:  %#v", suffix, got2)
	}
	if !strings.Contains(got2, `"five_hour":null`) {
		t.Errorf("example 10 status: missing null five_hour in %s", got2)
	}
}
