package supervisor

import (
	"encoding/json"
	"os"
	"strings"
	"testing"
	"time"

	"morphstudio/control"
	"morphstudio/gitrules"
	"morphstudio/internal/testhelp"
	"morphstudio/queue"
	"morphstudio/stream"
)

// gdConfig is the config the runtime guard examples use.
func gdConfig() Config {
	return Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}
}

// gdTime is a time on 2026-10-08.
func gdTime(h, m, s int) time.Time {
	return time.Date(2026, 10, 8, h, m, s, 0, time.UTC)
}

// gdNewID is the session id source the examples use.
func gdNewID() string { return "id-1" }

// gdRunning builds the running loop example 1 describes.
func gdRunning(t *testing.T) *Loop {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var plan queue.Plan
	if err := json.Unmarshal(data, &plan); err != nil {
		t.Fatal(err)
	}
	q, err := queue.Load(plan, queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5})
	if err != nil {
		t.Fatal(err)
	}
	l := New("demo", q)
	if _, err := l.Begin(gdTime(15, 0, 0)); err != nil {
		t.Fatal(err)
	}
	l.StartChecked(gitrules.Start{Mode: "fresh"}, gdNewID, gdTime(15, 0, 1))
	return l
}

func TestRuntimeGuardExample1(t *testing.T) {
	cfg := gdConfig()
	l := gdRunning(t)

	got := l.Exited(4, "", gdTime(15, 20, 0), cfg)
	want := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "watchdog",
			Headline: "P17: session exited (code 4), restarting",
			Numbers:  "restart 1 of 3 this hour",
		}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "exited actions", got, want)
	testhelp.Equal(t, "state", l.State, "starting")
	testhelp.Equal(t, "restarts", l.Restarts, []time.Time{gdTime(15, 20, 0)})
	testhelp.Equal(t, "session id", l.SessionID, "id-1")
	if l.LastExit == nil {
		t.Fatal("LastExit nil")
	}
	testhelp.Equal(t, "last exit", *l.LastExit, control.Exit{Code: 4, At: gdTime(15, 20, 0), Stderr: "", Restarts: 1})
}

func TestRuntimeGuardExample2(t *testing.T) {
	cfg := gdConfig()

	// Main sequence: four exits within the hour.
	l := gdRunning(t)
	l.Exited(4, "", gdTime(15, 20, 0), cfg)
	l.StartChecked(gitrules.Start{Mode: "fresh"}, gdNewID, gdTime(15, 20, 5))

	got2 := l.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", gdTime(15, 30, 0), cfg)
	want2 := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "watchdog",
			Headline: "P17: session exited (code 1), restarting",
			Numbers:  "restart 2 of 3 this hour · Error: Invalid session ID. Must be a valid UUID.",
		}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "second exit", got2, want2)
	l.StartChecked(gitrules.Start{Mode: "resume", Reason: "dirty tree (1 paths)"}, gdNewID, gdTime(15, 30, 5))

	got3 := l.Exited(137, "", gdTime(15, 40, 0), cfg)
	want3 := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "watchdog",
			Headline: "P17: session exited (code 137), restarting",
			Numbers:  "restart 3 of 3 this hour",
		}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "third exit", got3, want3)
	l.StartChecked(gitrules.Start{Mode: "fresh"}, gdNewID, gdTime(15, 40, 5))

	got4 := l.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", gdTime(15, 50, 0), cfg)
	want4 := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "stop",
			Headline: "P17: session keeps exiting",
			Numbers:  "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.",
		}},
	}
	testhelp.Equal(t, "fourth exit", got4, want4)
	testhelp.Equal(t, "state", l.State, "waiting")
	wantStop := &control.Stop{
		Kind:   "crash",
		Phase:  "P17",
		Reason: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.",
		At:     gdTime(15, 50, 0),
	}
	testhelp.Equal(t, "stop", l.Stop, wantStop)
	testhelp.Equal(t, "queue state", l.Queue.State, "stopped")
	testhelp.Equal(t, "restarts", l.Restarts, []time.Time{gdTime(15, 20, 0), gdTime(15, 30, 0), gdTime(15, 40, 0)})
	if l.LastExit == nil {
		t.Fatal("LastExit nil")
	}
	testhelp.Equal(t, "last exit", *l.LastExit, control.Exit{Code: 1, At: gdTime(15, 50, 0), Stderr: "Error: Invalid session ID. Must be a valid UUID.", Restarts: 3})

	// Variant: the fourth exit falls outside the hour of the first restart.
	l2 := gdRunning(t)
	l2.Exited(4, "", gdTime(15, 20, 0), cfg)
	l2.StartChecked(gitrules.Start{Mode: "fresh"}, gdNewID, gdTime(15, 20, 5))
	l2.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", gdTime(15, 30, 0), cfg)
	l2.StartChecked(gitrules.Start{Mode: "resume", Reason: "dirty tree (1 paths)"}, gdNewID, gdTime(15, 30, 5))
	l2.Exited(137, "", gdTime(15, 40, 0), cfg)
	l2.StartChecked(gitrules.Start{Mode: "fresh"}, gdNewID, gdTime(15, 40, 5))

	gotV := l2.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", gdTime(16, 25, 0), cfg)
	wantV := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "watchdog",
			Headline: "P17: session exited (code 1), restarting",
			Numbers:  "restart 3 of 3 this hour · Error: Invalid session ID. Must be a valid UUID.",
		}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "variant exit", gotV, wantV)
	testhelp.Equal(t, "variant restarts", l2.Restarts, []time.Time{gdTime(15, 30, 0), gdTime(15, 40, 0), gdTime(16, 25, 0)})
	testhelp.Equal(t, "variant state", l2.State, "starting")
}

func TestRuntimeGuardExample3(t *testing.T) {
	cfg := gdConfig()
	l := gdRunning(t)
	l.StretchUSD = 28
	l.SessionUSD = 0

	testhelp.Equal(t, "first turn", l.Turn(stream.Result{TotalCostUSD: 1.5}, gdTime(15, 30, 0), cfg), []Action(nil))
	testhelp.Equal(t, "session usd", l.SessionUSD, 1.5)

	got := l.Turn(stream.Result{TotalCostUSD: 2.25}, gdTime(16, 0, 0), cfg)
	want := []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "stop",
			Headline: "P17: stretch cap",
			Numbers:  "stretch cap $30.00 reached ($30.2500)",
		}},
	}
	testhelp.Equal(t, "second turn", got, want)
	testhelp.Equal(t, "state", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("Stop nil")
	}
	testhelp.Equal(t, "stop kind", l.Stop.Kind, "cap")

	// With StretchUSD 0 the cap is off and the turn is inert.
	cfg0 := gdConfig()
	cfg0.StretchUSD = 0
	l0 := gdRunning(t)
	l0.StretchUSD = 28
	l0.SessionUSD = 0
	testhelp.Equal(t, "zero first turn", l0.Turn(stream.Result{TotalCostUSD: 1.5}, gdTime(15, 30, 0), cfg0), []Action(nil))
	testhelp.Equal(t, "zero second turn", l0.Turn(stream.Result{TotalCostUSD: 2.25}, gdTime(16, 0, 0), cfg0), []Action(nil))
	testhelp.Equal(t, "zero state", l0.State, "running")
}

func TestRuntimeGuardExample4(t *testing.T) {
	cfg := gdConfig()
	l := gdRunning(t)

	lim := stream.Limits{
		FiveHour: stream.Window{Utilization: 0.22, ResetsAt: 1791469200},
		SevenDay: stream.Window{Utilization: 0.6, ResetsAt: 1791853200},
	}
	got := l.Limits(lim, gdTime(15, 30, 0), cfg)
	want := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "info",
			Headline: "P17: weekly usage 60%",
			Numbers:  "resets 2026-10-13T01:00:00Z",
		}},
	}
	testhelp.Equal(t, "limits", got, want)
	testhelp.Equal(t, "alerted", l.Alerted, true)

	testhelp.Equal(t, "limits again", l.Limits(lim, gdTime(15, 31, 0), cfg), []Action(nil))

	cfg0 := gdConfig()
	cfg0.UsageAlertPercent = 0
	l0 := gdRunning(t)
	testhelp.Equal(t, "alert off", l0.Limits(lim, gdTime(15, 30, 0), cfg0), []Action(nil))
	testhelp.Equal(t, "alert off flag", l0.Alerted, false)
}

func TestRuntimeGuardExample5(t *testing.T) {
	cfg := gdConfig()
	l := gdRunning(t)

	lim := stream.Limits{
		FiveHour: stream.Window{Utilization: 1, ResetsAt: 1791469200},
		SevenDay: stream.Window{Utilization: 0.6, ResetsAt: 1791853200},
	}
	got := l.Limits(lim, gdTime(13, 9, 3), cfg)
	want := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "watchdog",
			Headline: "P17: usage limit, paused",
			Numbers:  "until 2026-10-08T14:20:00Z",
		}},
	}
	testhelp.Equal(t, "limits", got, want)
	testhelp.Equal(t, "state", l.State, "paused")
	testhelp.Equal(t, "paused until", l.PausedUntil, gdTime(14, 20, 0))

	// When both windows are exhausted the pause runs to the later reset.
	l2 := gdRunning(t)
	lim2 := stream.Limits{
		FiveHour: stream.Window{Utilization: 1, ResetsAt: 1791469200},
		SevenDay: stream.Window{Utilization: 1, ResetsAt: 1791853200},
	}
	l2.Limits(lim2, gdTime(13, 9, 3), cfg)
	testhelp.Equal(t, "both paused until", l2.PausedUntil, time.Date(2026, 10, 13, 1, 0, 0, 0, time.UTC))
}

func TestRuntimeGuardExample6(t *testing.T) {

	}

	testhelp.Equal(t, "state", l.State, "running")
	testhelp.Equal(t, "last event at", l.LastEventAt, gdTime(14, 20, 0))

func TestRuntimeGuardExample9(t *testing.T) {
	cfg := gdConfig()

	// First: the first non-blank line, trimmed.
	l1 := gdRunning(t)
	got1 := l1.Exited(2, "\n\n node: bad option --foo \n(use node --help for usage)\n", gdTime(15, 5, 0), cfg)
	want1 := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "watchdog",
			Headline: "P17: session exited (code 2), restarting",
			Numbers:  "restart 1 of 3 this hour · node: bad option --foo",
		}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "first exit", got1, want1)
	if l1.LastExit == nil {
		t.Fatal("LastExit nil")
	}
	testhelp.Equal(t, "first stderr", l1.LastExit.Stderr, "node: bad option --foo")

	// Second: a first line longer than 200 runes is cut to 200.
	l2 := gdRunning(t)
	got2 := l2.Exited(3, strings.Repeat("x", 250)+"\nsecond\n", gdTime(15, 6, 0), cfg)
	want2 := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "watchdog",
			Headline: "P17: session exited (code 3), restarting",
			Numbers:  "restart 1 of 3 this hour · " + strings.Repeat("x", 200),
		}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "second exit", got2, want2)

	// Third: no restarts allowed, so the very first exit stops the walk.
	cfgZero := Config{MaxResumesPerHour: 0, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}
	l3 := gdRunning(t)
	got3 := l3.Exited(-3, "claude: start fork/exec /opt/claude/bin/claude: no such file or directory", gdTime(15, 7, 0), cfgZero)
	want3 := []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "stop",
			Headline: "P17: session keeps exiting",
			Numbers:  "exited 1 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory",
		}},
	}
	testhelp.Equal(t, "third exit", got3, want3)
	testhelp.Equal(t, "state", l3.State, "waiting")
	if l3.Stop == nil {
		t.Fatal("Stop nil")
	}
	testhelp.Equal(t, "stop kind", l3.Stop.Kind, "crash")
	testhelp.Equal(t, "restarts", l3.Restarts, []time.Time(nil))
	if l3.LastExit == nil {
		t.Fatal("LastExit nil")
	}
	testhelp.Equal(t, "last exit", *l3.LastExit, control.Exit{
		Code:     -3,
		At:       gdTime(15, 7, 0),
		Stderr:   "claude: start fork/exec /opt/claude/bin/claude: no such file or directory",
		Restarts: 0,
	})
}
