package supervisor

import (
	"fmt"
	"strings"
	"time"

	"morphstudio/control"
	"morphstudio/stream"
)

// Event records that the session produced a line.
func (l *Loop) Event(now time.Time) {
	l.LastEventAt = now
}

// Exited handles an exit of the session process while the loop still wanted
// the session. Anything else is not a crash and is left untouched.
func (l *Loop) Exited(code int, stderr string, now time.Time, cfg Config) []Action {
	if l.State != "running" && l.State != "paused" && l.State != "starting" {
		return nil
	}
	line := firstLine(stderr)

	var kept []time.Time
	for _, t := range l.Restarts {
		if now.Sub(t) < time.Hour {
			kept = append(kept, t)
		}
	}
	n := len(kept)

	if n < cfg.MaxResumesPerHour {
		kept = append(kept, now)
		l.Restarts = kept
		l.LastExit = &control.Exit{Code: code, At: now, Stderr: line, Restarts: n + 1}
		l.State = "starting"
		numbers := fmt.Sprintf("restart %d of %d this hour", n+1, cfg.MaxResumesPerHour)
		if line != "" {
			numbers += " · " + line
		}
		return []Action{
			{Kind: "post", Milestone: control.Milestone{
				Kind:     "watchdog",
				Headline: fmt.Sprintf("%s: session exited (code %d), restarting", l.Phase, code),
				Numbers:  numbers,
			}},
			{Kind: "start_check", Phase: l.Phase},
		}
	}

	l.Restarts = kept
	l.LastExit = &control.Exit{Code: code, At: now, Stderr: line, Restarts: n}
	reason := fmt.Sprintf("exited %d times within an hour (last code %d)", n+1, code)
	if line != "" {
		reason += ": " + line
	}
	l.State = "waiting"
	l.Stop = &control.Stop{Kind: "crash", Phase: l.Phase, Reason: reason, At: now}
	l.Queue.Stop("crash")
	return []Action{
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "stop",
			Headline: l.Phase + ": session keeps exiting",
			Numbers:  reason,
		}},
	}
}

// Turn records a result record's cost and stops the walk when the stretch
// budget is reached.
func (l *Loop) Turn(r stream.Result, now time.Time, cfg Config) []Action {
	l.SessionUSD = r.TotalCostUSD
	if cfg.StretchUSD <= 0 {
		return nil
	}
	if l.State != "running" && l.State != "paused" {
		return nil
	}
	sum := l.StretchUSD + l.SessionUSD
	if sum < cfg.StretchUSD {
		return nil
	}
	reason := fmt.Sprintf("stretch cap $%.2f reached ($%.4f)", cfg.StretchUSD, sum)
	l.State = "waiting"
	l.Stop = &control.Stop{Kind: "cap", Phase: l.Phase, Reason: reason, At: now}
	l.Queue.Stop("cap")
	return []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{
			Kind:     "stop",
			Headline: l.Phase + ": stretch cap",
			Numbers:  reason,
		}},
	}
}

// Limits pauses a running walk on an exhausted window and raises the weekly
// usage alert once.
func (l *Loop) Limits(lim stream.Limits, now time.Time, cfg Config) []Action {
	if l.State == "running" {
		var until time.Time
		if lim.FiveHour.Utilization >= 1 {
			until = time.Unix(lim.FiveHour.ResetsAt, 0).UTC()
		}
		if lim.SevenDay.Utilization >= 1 {
			t := time.Unix(lim.SevenDay.ResetsAt, 0).UTC()
			if t.After(until) {
				until = t
			}
		}
		if !until.IsZero() {
			l.State = "paused"
			l.PausedUntil = until
			return []Action{
				{Kind: "post", Milestone: control.Milestone{
					Kind:     "watchdog",
					Headline: l.Phase + ": usage limit, paused",
					Numbers:  "until " + until.Format(time.RFC3339),
				}},
			}
		}
	}
	if cfg.UsageAlertPercent > 0 && !l.Alerted &&
		lim.SevenDay.Utilization*100 >= float64(cfg.UsageAlertPercent) {
		l.Alerted = true
		resets := time.Unix(lim.SevenDay.ResetsAt, 0).UTC().Format(time.RFC3339)
		return []Action{
			{Kind: "post", Milestone: control.Milestone{
				Kind:     "info",
				Headline: fmt.Sprintf("%s: weekly usage %d%%", l.Phase, control.Percent(lim.SevenDay.Utilization)),
				Numbers:  "resets " + resets,
			}},
		}
	}
	return nil
}

// Tick advances the wall clock: it resumes a paused walk, stops one that hit
// the phase's hour cap and nudges one that went quiet.
func (l *Loop) Tick(now time.Time, cfg Config) []Action {
	if l.State == "paused" && !now.Before(l.PausedUntil) {
		l.State = "running"
		l.LastEventAt = now
		return []Action{
			{Kind: "write", Line: "Continue by docs/AUTONOMY.md from where you stopped; the usage window has reset."},
			{Kind: "post", Milestone: control.Milestone{
				Kind:     "watchdog",
				Headline: l.Phase + ": resumed after the limit",
				Numbers:  "",
			}},
		}
	}
	if l.State == "running" {
		caps := l.currentCaps()
		if caps.Hours > 0 && now.Sub(l.PhaseStartedAt) >= time.Duration(caps.Hours*float64(time.Hour)) {
			reason := fmt.Sprintf("wall clock cap %s h reached", formatFloat(caps.Hours, -1))
			l.State = "waiting"
			l.Stop = &control.Stop{Kind: "cap", Phase: l.Phase, Reason: reason, At: now}
			l.Queue.Stop("cap")
			return []Action{
				{Kind: "kill"},
				{Kind: "post", Milestone: control.Milestone{
					Kind:     "stop",
					Headline: l.Phase + ": wall clock cap",
					Numbers:  reason,
				}},
			}
		}
		if cfg.StallMinutes > 0 && !l.Nudged &&
			now.Sub(l.LastEventAt) >= time.Duration(cfg.StallMinutes)*time.Minute {
			l.Nudged = true
			return []Action{
				{Kind: "post", Milestone: control.Milestone{
					Kind:     "watchdog",
					Headline: fmt.Sprintf("%s: no events for %d min", l.Phase, cfg.StallMinutes),
					Numbers:  "one nudge sent",
				}},
				{Kind: "write", Line: fmt.Sprintf("Nothing has happened for %d minutes. Check the state of your agents and the run branch, then continue by docs/AUTONOMY.md; post the current state first.", cfg.StallMinutes)},
			}
		}
	}
	return nil
}

// firstLine returns the first non-blank line of s, trimmed and cut to 200
// runes, or "" when there is none.
func firstLine(s string) string {
	for _, raw := range strings.Split(s, "\n") {
		line := strings.TrimSpace(raw)
		if line == "" {
			continue
		}
		runes := []rune(line)
		if len(runes) > 200 {
			runes = runes[:200]
		}
		return string(runes)
	}
	return ""
}
