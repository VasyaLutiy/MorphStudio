func TestConfigAndMainExample4(t *testing.T) {
	_, err := Load(func(string) string { return "" }, map[string]string{})
	if err == nil {
		t.Fatal("want error")
	}
	testhelp.Equal(t, "err", err.Error(), "config: MORPHD_TOKEN is required")

	getenv := func(m map[string]string) func(string) string {
		return func(k string) string { return m[k] }
	}

	_, err = Load(getenv(map[string]string{"MORPHD_TOKEN": "x", "MORPHD_PORT": "80a"}), map[string]string{})
	if err == nil {
		t.Fatal("want error")
	}
	testhelp.Equal(t, "err", err.Error(), "config: MORPHD_PORT: not a number: 80a")

	_, err = Load(getenv(map[string]string{"MORPHD_TOKEN": "x", "MORPHD_STALL_MINUTES": "soon"}), map[string]string{})
	if err == nil {
		t.Fatal("want error")
	}
	testhelp.Equal(t, "err", err.Error(), "config: MORPHD_STALL_MINUTES: not a number: soon")
}

func TestConfigAndMainExample5(t *testing.T) {
	f := cmFake{}
	known := func(p string) bool { return p == "demo" }
	sess := func(p string) (string, bool) { return "s", p == "demo" }
	h := Build(f, known, sess, "tok-7")

	// GET /healthz
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, httptest.NewRequest(http.MethodGet, "/healthz", nil))
	testhelp.Equal(t, "healthz code", rec.Code, 200)
	testhelp.Equal(t, "healthz body", rec.Body.String(), `{"status":"ok"}`+"\n")

	// GET /projects/demo/status
	req := httptest.NewRequest(http.MethodGet, "/projects/demo/status", nil)
	req.Header.Set("Authorization", "Bearer tok-7")
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	testhelp.Equal(t, "status code", rec.Code, 200)
	testhelp.Equal(t, "status body", rec.Body.String(), `{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}`+"\n")

	// POST /mcp/demo
	mcpBody := `{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}`
	req = httptest.NewRequest(http.MethodPost, "/mcp/demo", strings.NewReader(mcpBody))
	req.Header.Set("Authorization", "Bearer tok-7")
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Accept", "application/json, text/event-stream")
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	testhelp.Equal(t, "mcp code", rec.Code, 200)
	var payload struct {
		Result struct {
			Tools []struct {
				Name string `json:"name"`
			} `json:"tools"`
		} `json:"result"`
	}
	if err := json.Unmarshal(rec.Body.Bytes(), &payload); err != nil {
		t.Fatalf("mcp decode: %v", err)
	}
	testhelp.Equal(t, "mcp tools count", len(payload.Result.Tools), 10)
	found := false
	for _, tool := range payload.Result.Tools {
		if tool.Name == "status" {
			found = true
		}
	}
	testhelp.Equal(t, "mcp status tool", found, true)

	// GET /projects/demo/status without Authorization
	rec = httptest.NewRecorder()
	h.ServeHTTP(rec, httptest.NewRequest(http.MethodGet, "/projects/demo/status", nil))
	testhelp.Equal(t, "unauthorized code", rec.Code, 401)
}

func TestConfigAndMainExample7(t *testing.T) {
	re := regexp.MustCompile(`^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$`)
	a := newID()
	b := newID()

	testhelp.Equal(t, "a matches", re.MatchString(a), true)
	testhelp.Equal(t, "a len", len(a), 36)
	testhelp.Equal(t, "a lower", strings.ToLower(a) == a, true)

	testhelp.Equal(t, "b matches", re.MatchString(b), true)
	testhelp.Equal(t, "b len", len(b), 36)
	testhelp.Equal(t, "b lower", strings.ToLower(b) == b, true)

	testhelp.Equal(t, "a != b", a != b, true)
}
