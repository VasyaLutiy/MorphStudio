// ccMarshal marshals v and returns the JSON text, failing the test on error.
func ccMarshal(t *testing.T, v any) string {
	t.Helper()
	b, err := json.Marshal(v)
	if err != nil {
		t.Fatalf("ccMarshal: %v", err)
	}
	return string(b)
}

func TestControlContractExample1(t *testing.T) {
	s := Status{
		State:     "busy",
		Phase:     "P18",
		Minutes:   37,
		CostUSD:   1.92,
		FiveHour:  22,
		SevenDay:  60,
		SessionID: "s-1",
		Queue: QueueView{
			State:    "running",
			Approved: "7b31dfe",
			Index:    1,
			Phases:   3,
			Current:  "P18",
		},
		Repo: github.Access{
			Checked:   true,
			Reachable: true,
			Push:      true,
			CheckedAt: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC),
		},
	}

	got := ccMarshal(t, s)
	testhelp.Equal(t, "status json", got, `{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}`)
	testhelp.Equal(t, "no stop key", strings.Contains(got, `"stop"`), false)
}

func TestControlContractExample5(t *testing.T) {
	cases := []struct {
		name   string
		err    error
		status int
		code   string
	}{
		{"nil", nil, 200, ""},
		{"ErrUnknownProject", ErrUnknownProject, 404, "unknown_project"},
		{"wrapped ErrBadOption", fmt.Errorf("wrap: %w", ErrBadOption), 400, "bad_option"},
		{"ErrBadToken", ErrBadToken, 401, "bad_session_token"},
		{"ErrExists", ErrExists, 409, "exists"},
		{"ErrNoSession", ErrNoSession, 409, "no_session"},
		{"ErrBusy", ErrBusy, 409, "busy"},
		{"ErrNoQuestion", ErrNoQuestion, 409, "no_question"},
		{"ErrNotBusy", ErrNotBusy, 409, "not_busy"},
		{"ErrNotWaiting", ErrNotWaiting, 409, "not_waiting"},
		{"ErrBadInput", ErrBadInput, 400, "bad_input"},
		{"disk full", errors.New("disk full"), 500, "internal"},
	}
	for _, c := range cases {
		status, code := Code(c.err)
		testhelp.Equal(t, "Code status "+c.name, status, c.status)
		testhelp.Equal(t, "Code code "+c.name, code, c.code)
	}
}

func TestControlContractExample8(t *testing.T) {
	idle := Status{State: "idle", LimitsUnknown: true}
	got := ccMarshal(t, idle)
	testhelp.Equal(t, "idle status json", got, `{"state":"idle","phase":"","minutes":0,"cost_usd":0,"five_hour":null,"seven_day":null,"session_id":"","queue":{"state":"","approved":"","index":0,"phases":0,"current":"","reason":""},"repo":{"checked":false,"reachable":false,"push":false,"reason":"","checked_at":"0001-01-01T00:00:00Z"}}`)
	testhelp.Equal(t, "no stop key", strings.Contains(got, `"stop"`), false)
	testhelp.Equal(t, "no last_exit key", strings.Contains(got, `"last_exit"`), false)
	testhelp.Equal(t, "no limits_unknown key", strings.Contains(got, `"limits_unknown"`), false)

	busy := Status{
		State:         "busy",
		Phase:         "S1",
		Minutes:       2,
		CostUSD:       0.03,
		FiveHour:      77,
		SevenDay:      72,
		LimitsUnknown: true,
		SessionID:     "s-9",
	}
	got = ccMarshal(t, busy)
	testhelp.Equal(t, "busy status has null limits", strings.Contains(got, `,"five_hour":null,"seven_day":null,"session_id":"s-9",`), true)
	testhelp.Equal(t, "no five_hour value", strings.Contains(got, "77"), false)
	testhelp.Equal(t, "no seven_day value", strings.Contains(got, "72"), false)
	testhelp.Equal(t, "busy status begins", strings.HasPrefix(got, `{"state":"busy","phase":"S1","minutes":2,"cost_usd":0.03,`), true)
}

func TestControlContractExample10(t *testing.T) {
	exit := Exit{
		Code:     1,
		At:       time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
		Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
		Restarts: 3,
	}
	got := ccMarshal(t, exit)
	testhelp.Equal(t, "exit json", got, `{"code":1,"at":"2026-10-09T08:14:02Z","stderr":"Error: Invalid session ID. Must be a valid UUID.","restarts":3}`)

	st := Status{
		State:         "waiting",
		Phase:         "S1",
		LimitsUnknown: true,
		Stop: &Stop{
			Kind:   "crash",
			Phase:  "S1",
			Reason: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.",
			At:     time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
		},
		LastExit: &Exit{
			Code:     1,
			At:       time.Date(2026, 10, 9, 8, 14, 2, 0, time.UTC),
			Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
			Restarts: 3,
		},
	}
	got = ccMarshal(t, st)
	testhelp.Equal(t, "status json ends with stop and last_exit", strings.HasSuffix(got, `,"stop":{"kind":"crash","phase":"S1","reason":"exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.","at":"2026-10-09T08:14:02Z"},"last_exit":{"code":1,"at":"2026-10-09T08:14:02Z","stderr":"Error: Invalid session ID. Must be a valid UUID.","restarts":3}}`), true)
	testhelp.Equal(t, "status json holds null limits", strings.Contains(got, `"five_hour":null`), true)
}
