	t      *testing.T
	tmp    string
	reg    *registry.Registry
	cfg    Config
	runner *runner.Fake

	spawnFn func(context.Context, claude.Launch) (claude.Process, error)

func dcNew(t *testing.T) *dcHarness {
	t.Helper()
	h := &dcHarness{t: t, tmp: t.TempDir()}
	reg, err := registry.Open(filepath.Join(h.tmp, "state"))
	if err != nil {
		t.Fatal(err)
	}
	h.reg = reg
	h.now = time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC)
	h.ghStatus = 200
	h.ghBody = "{}"
	h.cfg = Config{
		ClaudeBin:   "/opt/claude/bin/claude",
		MorphBin:    "/usr/local/bin/morph",
		ProjectsDir: filepath.Join(h.tmp, "projects"),
		MCPBaseURL:  "http://127.0.0.1:7181",
		Defaults:    queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4},
		MaxParallel: 1,
		Loop: supervisor.Config{
			MaxResumesPerHour: 3,
			StallMinutes:      30,
			StretchUSD:        30,
			UsageAlertPercent: 50,
		},
	}
	h.spawnFn = func(ctx context.Context, l claude.Launch) (claude.Process, error) {
		h.mu.Lock()
		defer h.mu.Unlock()
		h.spawns = append(h.spawns, l)
		f := claude.NewFake()
		h.procs = append(h.procs, f)
		return f, nil
	}
	return h
}

func dcAddProject(h *dcHarness, name string, caps queue.Caps, createdAt time.Time) {
	h.t.Helper()
	dir := filepath.Join(h.tmp, "projects", name)
	measurePath := filepath.Join(dir, "docs", "MEASURE.md")
	if err := os.MkdirAll(filepath.Dir(measurePath), 0o755); err != nil {
		h.t.Fatal(err)
	}
	data, err := os.ReadFile("../tests/fixtures/git/measure.md")
	if err != nil {
		h.t.Fatal(err)
	}
	if err := os.WriteFile(measurePath, data, 0o644); err != nil {
		h.t.Fatal(err)
	}
	if err := h.reg.Add(registry.Project{
		Name:      name,
		Language:  "go",
		RepoURL:   "https://github.com/acme/" + name,
		Dir:       dir,
		Caps:      caps,
		CreatedAt: createdAt,
	}); err != nil {
		h.t.Fatal(err)
	}
}

func dcOpen(h *dcHarness, script map[string]runner.Result) {
	h.t.Helper()
	h.runner = &runner.Fake{Script: script}
	deps := Deps{
		Registry: h.reg,
		Runner:   h.runner,
		Spawn:    h.spawnFn,
		Post: func(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
			h.mu.Lock()
			defer h.mu.Unlock()
			h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
			return telegram.Status{}, nil
		},
		GitHubDo: func(req *http.Request) (*http.Response, error) {
			h.mu.Lock()
			defer h.mu.Unlock()
			return &http.Response{
				StatusCode: h.ghStatus,
				Body:       io.NopCloser(strings.NewReader(h.ghBody)),
				Header:     http.Header{},
			}, nil
		},
		Now: func() time.Time {
			h.mu.Lock()
			defer h.mu.Unlock()
			v := h.now
			h.now = h.now.Add(time.Second)
			return v
		},
		NewID: func() string {
			h.mu.Lock()
			defer h.mu.Unlock()
			h.idN++
			return fmt.Sprintf("id-%d", h.idN)
		},
		Config: h.cfg,
	}
	d, err := Open(context.Background(), deps)
	if err != nil {
		h.t.Fatal(err)
	}
	h.d = d
}

func dcSetScript(h *dcHarness, key string, res runner.Result) {

		t.Fatal(err)
	}

	if !errors.Is(err, control.ErrNoQuestion) {

	if err := h.d.Interrupt("demo"); err != nil {
		t.Fatal(err)
	}
	written := fp.Written()

		t.FailNow()

	testhelp.Equal(t, "interrupt line", written[1], stream.Interrupt("int-id-3"))

	if err != nil {
		t.Fatal(err)
	}

		t.Fatal(err)
	}
	var raw struct {
		Entries []struct {
			Seq int64  `json:"seq"`
			Dir string `json:"dir"`
		} `json:"entries"`
		Last int64 `json:"last"`
	}
	if err := json.Unmarshal(b, &raw); err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "events last", raw.Last, int64(2))
	if !testhelp.Equal(t, "events len", len(raw.Entries), 2) {
		t.FailNow()
	}
	testhelp.Equal(t, "e0 seq", raw.Entries[0].Seq, int64(1))
	testhelp.Equal(t, "e0 dir", raw.Entries[0].Dir, "in")
	testhelp.Equal(t, "e1 seq", raw.Entries[1].Seq, int64(2))
	testhelp.Equal(t, "e1 dir", raw.Entries[1].Dir, "in")

		t.Errorf("events missing interrupt: %s", b)
	}

func TestDaemonCoreExample11(t *testing.T) {
	errMsg := "claude: start fork/exec /opt/claude/bin/claude: no such file or directory"
	h := dcNew(t)
	dcAddProject(h, "demo", queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5}, time.Date(2026, 10, 8, 14, 0, 0, 0, time.UTC))
	h.spawnFn = func(ctx context.Context, l claude.Launch) (claude.Process, error) {
		h.mu.Lock()
		defer h.mu.Unlock()
		h.spawns = append(h.spawns, l)
		return nil, errors.New(errMsg)
	}
	dcOpen(h, dcFreshScript())
	plan := dcLoadPlan(t)

	qv, err := h.d.PlanLoad("demo", plan)
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "queue view", qv, control.QueueView{State: "stopped", Approved: "7b31dfe", Index: 0, Phases: 3, Current: "P17", Reason: "crash"})

	spawns := dcSpawns(h)
	if !testhelp.Equal(t, "spawn count", len(spawns), 4) {
		t.FailNow()
	}
	testhelp.Equal(t, "session ids", []string{
		spawns[0].SessionID, spawns[1].SessionID, spawns[2].SessionID, spawns[3].SessionID,
	}, []string{"id-1", "id-3", "id-5", "id-7"})
	for i, l := range spawns {
		if l.Resume {
			t.Errorf("spawn %d: Resume true", i)
		}
	}
	tok, err := h.reg.ReadSecret("demo", "session-token")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "session token", tok, "id-8")

	testhelp.Equal(t, "posts", dcPosts(h), [][4]string{
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 1 of 3 this hour · " + errMsg},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 2 of 3 this hour · " + errMsg},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 3 of 3 this hour · " + errMsg},
		{"demo", "stop", "P17: session keeps exiting", "exited 4 times within an hour (last code -3): " + errMsg},
	})

	st, err := h.d.Status("demo")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "state", st.State, "waiting")
	testhelp.Equal(t, "phase", st.Phase, "P17")
	testhelp.Equal(t, "session", st.SessionID, "id-7")
	testhelp.Equal(t, "limits unknown", st.LimitsUnknown, true)
	if st.Stop == nil {
		t.Fatal("status stop nil")
	}
	testhelp.Equal(t, "stop kind", st.Stop.Kind, "crash")
	testhelp.Equal(t, "stop reason", st.Stop.Reason, "exited 4 times within an hour (last code -3): "+errMsg)
	if st.LastExit == nil {
		t.Fatal("status last exit nil")
	}
	le := *st.LastExit
	if le.At.IsZero() {
		t.Fatal("last exit at is zero")
	}
	le.At = time.Time{}
	testhelp.Equal(t, "last exit", le, control.Exit{Code: -3, Stderr: errMsg, Restarts: 3})

	stop, err := h.d.StopCheck("demo")
	if err != nil {
		t.Fatal(err)
	}
	if stop == nil {
		t.Fatal("stopcheck nil")
	}
	gotStop := *stop
	if gotStop.At.IsZero() {
		t.Fatal("stop at is zero")
	}
	gotStop.At = time.Time{}
	testhelp.Equal(t, "stop", gotStop, control.Stop{Kind: "crash", Phase: "P17", Reason: "exited 4 times within an hour (last code -3): " + errMsg})

	ev, err := h.d.Events("demo", 0, 10)
	if err != nil {
		t.Fatal(err)
	}
	evJSON, err := json.Marshal(ev)
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "events json", string(evJSON), `{"entries":[],"last":0}`)

	stateData, err := os.ReadFile(filepath.Join(h.tmp, "state", "demo", "state.json"))
	if err != nil {
		t.Fatal(err)
	}
	var state State
	if err := json.Unmarshal(stateData, &state); err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "loop state", state.Loop.State, "waiting")
	testhelp.Equal(t, "restarts", len(state.Loop.Restarts), 3)

	if err := h.d.Continue("demo"); err != nil {
		t.Fatal(err)
	}
	spawns2 := dcSpawns(h)
	if !testhelp.Equal(t, "spawn count after continue", len(spawns2), 8) {
		t.FailNow()
	}
	testhelp.Equal(t, "session ids after continue", []string{
		spawns2[4].SessionID, spawns2[5].SessionID, spawns2[6].SessionID, spawns2[7].SessionID,
	}, []string{"id-9", "id-11", "id-13", "id-15"})
	st2, err := h.d.Status("demo")
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "state after continue", st2.State, "waiting")
	if st2.Stop == nil {
		t.Fatal("status stop nil after continue")
	}
	testhelp.Equal(t, "stop kind after continue", st2.Stop.Kind, "crash")
	testhelp.Equal(t, "posts count", len(dcPosts(h)), 8)
}
