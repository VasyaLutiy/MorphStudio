package supervisor

import (
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"testing"
	"time"

	"morphstudio/control"
	"morphstudio/gitrules"
	"morphstudio/internal/testhelp"
	"morphstudio/queue"
)

func plTime(h, m, s int) time.Time {
	return time.Date(2026, 10, 8, h, m, s, 0, time.UTC)
}

func plQueue(t *testing.T) queue.Queue {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var plan queue.Plan
	if err := json.Unmarshal(data, &plan); err != nil {
		t.Fatal(err)
	}
	q, err := queue.Load(plan, queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5})
	if err != nil {
		t.Fatal(err)
	}
	return q
}

func plIDs() func() string {
	n := 0
	return func() string {
		n++
		return fmt.Sprintf("id-%d", n)
	}
}

func plStartingP17(t *testing.T) *Loop {
	t.Helper()
	l := New("demo", plQueue(t))
	if _, err := l.Begin(plTime(15, 0, 0)); err != nil {
		t.Fatal(err)
	}
	return l
}

func plRunningP17(t *testing.T) *Loop {
	t.Helper()
	l := plStartingP17(t)
	l.Restarts = []time.Time{plTime(14, 50, 0)}
	l.LastExit = &control.Exit{Code: 1, At: plTime(14, 50, 0), Stderr: "boom", Restarts: 1}
	l.StartChecked(gitrules.Start{Mode: "fresh", Local: "aaa111", Remote: "aaa111"}, plIDs(), plTime(15, 0, 1))
	return l
}

func TestPhaseLoopExample1(t *testing.T) {
	l := New("demo", plQueue(t))
	l.Restarts = []time.Time{plTime(14, 0, 0)}
	l.LastExit = &control.Exit{Code: 7, At: plTime(14, 0, 0), Stderr: "old", Restarts: 1}

	actions, err := l.Begin(plTime(15, 0, 0))
	if err != nil {
		t.Fatal(err)
	}
	want := []Action{{Kind: "start_check", Phase: "P17"}}
	testhelp.Equal(t, "Begin actions", actions, want)
	testhelp.Equal(t, "State", l.State, "starting")
	testhelp.Equal(t, "Phase", l.Phase, "P17")
	testhelp.Equal(t, "Queue.State", l.Queue.State, "running")
	testhelp.Equal(t, "Restarts", l.Restarts, []time.Time(nil))
	if l.LastExit != nil {
		t.Fatalf("LastExit = %v, want nil", l.LastExit)
	}

	actions2, err2 := l.Begin(plTime(15, 0, 0))
	testhelp.Equal(t, "second Begin actions", actions2, []Action(nil))
	if err2 == nil {
		t.Fatal("second Begin err = nil, want error")
	}
	testhelp.Equal(t, "second Begin err text", err2.Error(), "queue: cannot start in state \"running\"")
	testhelp.Equal(t, "State after second Begin", l.State, "starting")
	testhelp.Equal(t, "Phase after second Begin", l.Phase, "P17")
}

func TestPhaseLoopExample2(t *testing.T) {

func TestPhaseLoopExample3(t *testing.T) {
	q := plQueue(t)
	q.Index = 1
	q.State = "running"
	l := New("demo", q)
	l.State = "starting"
	l.Phase = "P18"

	actions := l.StartChecked(gitrules.Start{Mode: "resume", Reason: "local ahead of origin/main"}, func() string { return "id-7" }, plTime(15, 0, 0))
	want := []Action{
		{Kind: "spawn", Phase: "P18", SessionID: "id-7", Resume: true, BudgetUSD: 12},
		{Kind: "post", Milestone: control.Milestone{Kind: "start", Headline: "P18 resumed", Numbers: "session id-7 · local ahead of origin/main"}},
	}
	testhelp.Equal(t, "StartChecked actions", actions, want)
	testhelp.Equal(t, "State", l.State, "running")
	testhelp.Equal(t, "SessionID", l.SessionID, "id-7")

	q2 := plQueue(t)
	q2.Index = 1
	q2.State = "running"
	l2 := New("demo", q2)
	l2.State = "starting"
	l2.Phase = "P18"
	l2.SessionID = "id-3"

	actions2 := l2.StartChecked(gitrules.Start{Mode: "resume", Reason: "local ahead of origin/main"}, func() string { return "unused" }, plTime(15, 0, 0))
	want2 := []Action{
		{Kind: "spawn", Phase: "P18", SessionID: "id-3", Resume: true, BudgetUSD: 12},
		{Kind: "post", Milestone: control.Milestone{Kind: "start", Headline: "P18 resumed", Numbers: "session id-3 · local ahead of origin/main"}},
	}
	testhelp.Equal(t, "StartChecked actions 2", actions2, want2)
	testhelp.Equal(t, "SessionID 2", l2.SessionID, "id-3")
}

func TestPhaseLoopExample5(t *testing.T) {

	testhelp.Equal(t, "Queue.State", l.Queue.State, "running")

func TestPhaseLoopExample6(t *testing.T) {
	l := plRunningP17(t)

	actions, err := l.PhaseDone("P16", "P17")
	testhelp.Equal(t, "PhaseDone actions", actions, []Action(nil))
	if err == nil {
		t.Fatal("PhaseDone err = nil, want ErrPhaseMismatch")
	}
	if !errors.Is(err, ErrPhaseMismatch) {
		t.Fatalf("PhaseDone err = %v, want ErrPhaseMismatch", err)
	}
	testhelp.Equal(t, "PhaseDone err text", err.Error(), "phase mismatch: phase_done \"P16\" while running \"P17\"")

	l2 := plRunningP17(t)
	l2.State = "waiting"

	actions2, err2 := l2.PhaseDone("P17", "P18")
	testhelp.Equal(t, "PhaseDone actions 2", actions2, []Action(nil))
	if err2 == nil {
		t.Fatal("PhaseDone err2 = nil, want ErrPhaseMismatch")
	}
	if !errors.Is(err2, ErrPhaseMismatch) {
		t.Fatalf("PhaseDone err2 = %v, want ErrPhaseMismatch", err2)
	}
	testhelp.Equal(t, "PhaseDone err2 text", err2.Error(), "phase mismatch: phase_done \"P17\" while waiting \"P17\"")
}

func TestPhaseLoopExample7(t *testing.T) {
	l := plRunningP17(t)
	l.SessionUSD = 1.92
	const reason = "not pushed: HEAD bbb222, origin/main aaa111"

	endActions := l.EndChecked(gitrules.End{OK: false, Local: "bbb222", Remote: "aaa111", Problems: []string{reason}}, plTime(16, 0, 0))
	wantEnd := []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17 not pushed", Numbers: reason}},
	}
	testhelp.Equal(t, "EndChecked actions", endActions, wantEnd)
	testhelp.Equal(t, "State", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("Stop = nil, want not_pushed stop")
	}
	testhelp.Equal(t, "Stop", *l.Stop, control.Stop{Kind: "not_pushed", Phase: "P17", Reason: reason, At: plTime(16, 0, 0)})
	testhelp.Equal(t, "StretchUSD", l.StretchUSD, 0.0)
	testhelp.Equal(t, "Queue.State", l.Queue.State, "stopped")

	contActions, err := l.Continue(plTime(16, 5, 0))
	if err != nil {
		t.Fatal(err)
	}
	wantCont := []Action{{Kind: "end_check", Phase: "P17"}}
	testhelp.Equal(t, "Continue actions", contActions, wantCont)
	testhelp.Equal(t, "State after Continue", l.State, "running")
	if l.Stop != nil {
		t.Fatalf("Stop after Continue = %v, want nil", l.Stop)
	}
	testhelp.Equal(t, "Queue.State after Continue", l.Queue.State, "running")
	testhelp.Equal(t, "Queue.Index after Continue", l.Queue.Index, 0)

	finalActions := l.EndChecked(gitrules.End{OK: true, Local: "bbb222", Remote: "bbb222", Problems: []string{}}, plTime(16, 10, 0))
	wantFinal := []Action{
		{Kind: "kill"},
		{Kind: "start_check", Phase: "P18"},
	}
	testhelp.Equal(t, "EndChecked actions 2", finalActions, wantFinal)
	testhelp.Equal(t, "StretchUSD 2", l.StretchUSD, 1.92)
	testhelp.Equal(t, "SessionUSD 2", l.SessionUSD, 0.0)
}

func TestPhaseLoopExample8(t *testing.T) {
	q := plQueue(t)
	q.Index = 1
	q.State = "running"
	l := New("demo", q)
	l.State = "running"
	l.Phase = "P18"
	l.SessionID = "sid"
	l.SessionUSD = 0.5
	l.StretchUSD = 1.0

	actions := l.EndChecked(gitrules.End{OK: true, Local: "x", Remote: "x", Problems: []string{}}, plTime(17, 0, 0))
	want := []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P18 done: stop after smoke", Numbers: "next P19 · waiting for continue"}},
	}
	testhelp.Equal(t, "EndChecked actions", actions, want)
	testhelp.Equal(t, "State", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("Stop = nil, want smoke stop")
	}
	testhelp.Equal(t, "Stop", *l.Stop, control.Stop{Kind: "smoke", Phase: "P18", Reason: "stop after smoke", At: plTime(17, 0, 0)})
	testhelp.Equal(t, "StretchUSD", l.StretchUSD, 1.5)

	contActions, err := l.Continue(plTime(17, 30, 0))
	if err != nil {
		t.Fatal(err)
	}
	wantCont := []Action{{Kind: "start_check", Phase: "P19"}}
	testhelp.Equal(t, "Continue actions", contActions, wantCont)
	testhelp.Equal(t, "State after Continue", l.State, "starting")
	testhelp.Equal(t, "Phase after Continue", l.Phase, "P19")
	if l.Stop != nil {
		t.Fatalf("Stop after Continue = %v, want nil", l.Stop)
	}

	l.StartChecked(gitrules.Start{Mode: "fresh", Local: "x", Remote: "x"}, plIDs(), plTime(17, 31, 0))

	finalActions := l.EndChecked(gitrules.End{OK: true, Local: "x", Remote: "x", Problems: []string{}}, plTime(17, 32, 0))
	wantFinal := []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{Kind: "end", Headline: "P19 done: plan complete", Numbers: "stretch $1.5000"}},
	}
	testhelp.Equal(t, "EndChecked actions 2", finalActions, wantFinal)
	testhelp.Equal(t, "State 2", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("Stop 2 = nil, want end stop")
	}
	testhelp.Equal(t, "Stop 2", *l.Stop, control.Stop{Kind: "end", Phase: "P19", Reason: "plan complete", At: plTime(17, 32, 0)})
	testhelp.Equal(t, "Queue.State 2", l.Queue.State, "done")

	contActions2, err2 := l.Continue(plTime(18, 0, 0))
	testhelp.Equal(t, "Continue actions 2", contActions2, []Action(nil))
	if err2 == nil {
		t.Fatal("Continue err2 = nil, want error")
	}
	testhelp.Equal(t, "Continue err2 text", err2.Error(), "queue: nothing to continue (state \"done\")")
}

func TestPhaseLoopExample9(t *testing.T) {
	l := plRunningP17(t)

	actions := l.WaitOperator("gate", "forecast $1.4 over the cap", "", plTime(15, 10, 0))
	want := []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: wait_operator gate", Numbers: "forecast $1.4 over the cap"}},
	}
	testhelp.Equal(t, "WaitOperator actions", actions, want)
	testhelp.Equal(t, "State", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("Stop = nil, want gate stop")
	}
	testhelp.Equal(t, "Stop", *l.Stop, control.Stop{Kind: "gate", Phase: "P17", Reason: "forecast $1.4 over the cap", IssueURL: "", At: plTime(15, 10, 0)})

	l2 := plRunningP17(t)
	actions2 := l2.WaitOperator("emergency", "card x red after fix", "https://github.com/o/r/issues/12", plTime(15, 11, 0))
	want2 := []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: wait_operator emergency", Numbers: "card x red after fix · https://github.com/o/r/issues/12"}},
	}
	testhelp.Equal(t, "WaitOperator 2 actions", actions2, want2)
	if l2.Stop == nil {
		t.Fatal("Stop 2 = nil, want emergency stop")
	}
	testhelp.Equal(t, "Stop 2", *l2.Stop, control.Stop{Kind: "emergency", Phase: "P17", Reason: "card x red after fix", IssueURL: "https://github.com/o/r/issues/12", At: plTime(15, 11, 0)})

	restartActions, err := l2.Restart(plTime(15, 12, 0))
	testhelp.Equal(t, "Restart waiting actions", restartActions, []Action(nil))
	if err == nil {
		t.Fatal("Restart err = nil, want ErrNoSession")
	}
	if !errors.Is(err, control.ErrNoSession) {
		t.Fatalf("Restart err = %v, want ErrNoSession", err)
	}

	l3 := plRunningP17(t)
	l3.Restarts = []time.Time{plTime(15, 5, 0)}
	l3.LastExit = &control.Exit{Code: 3, At: plTime(15, 5, 0), Stderr: "", Restarts: 1}

	restartActions2, err2 := l3.Restart(plTime(15, 13, 0))
	if err2 != nil {
		t.Fatal(err2)
	}
	want3 := []Action{
		{Kind: "kill"},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "Restart running actions", restartActions2, want3)
	testhelp.Equal(t, "State after Restart", l3.State, "starting")
	testhelp.Equal(t, "SessionID after Restart", l3.SessionID, "")
	testhelp.Equal(t, "Restarts after Restart", l3.Restarts, []time.Time(nil))
	if l3.LastExit == nil {
		t.Fatal("LastExit = nil, want kept exit")
	}
	testhelp.Equal(t, "LastExit after Restart", *l3.LastExit, control.Exit{Code: 3, At: plTime(15, 5, 0), Stderr: "", Restarts: 1})
}
