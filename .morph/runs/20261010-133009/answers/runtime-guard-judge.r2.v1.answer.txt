package supervisor

import (
	"encoding/json"
	"os"
	"strings"
	"testing"
	"time"

	"morphstudio/control"
	"morphstudio/gitrules"
	"morphstudio/internal/testhelp"
	"morphstudio/queue"
	"morphstudio/stream"
)

// gdCfg is the Config of every example.
func gdCfg() Config {
	return Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}
}

// gdAt is a time on the day of every example.
func gdAt(h, m, s int) time.Time {
	return time.Date(2026, 10, 8, h, m, s, 0, time.UTC)
}

// gdQueue loads the three-phase fixture with the example caps.
func gdQueue(t testing.TB) queue.Queue {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var plan queue.Plan
	if err := json.Unmarshal(data, &plan); err != nil {
		t.Fatal(err)
	}
	q, err := queue.Load(plan, queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5})
	if err != nil {
		t.Fatal(err)
	}
	return q
}

// gdRunning builds the running loop every example starts from.
func gdRunning(t testing.TB) *Loop {
	t.Helper()
	l := New("demo", gdQueue(t))
	if _, err := l.Begin(gdAt(15, 0, 0)); err != nil {
		t.Fatal(err)
	}
	l.StartChecked(gitrules.Start{Mode: "fresh"}, func() string { return "id-1" }, gdAt(15, 0, 1))
	return l
}

func TestRuntimeGuardExample1(t *testing.T) {
	cfg := gdCfg()
	l := gdRunning(t)
	testhelp.Equal(t, "initial State", l.State, "running")
	testhelp.Equal(t, "initial Phase", l.Phase, "P17")
	testhelp.Equal(t, "initial SessionID", l.SessionID, "id-1")
	testhelp.Equal(t, "initial PhaseStartedAt", l.PhaseStartedAt, gdAt(15, 0, 1))
	testhelp.Equal(t, "initial LastEventAt", l.LastEventAt, gdAt(15, 0, 1))

	acts := l.Exited(4, "", gdAt(15, 20, 0), cfg)
	testhelp.Equal(t, "actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 4), restarting", Numbers: "restart 1 of 3 this hour"}},
		{Kind: "start_check", Phase: "P17"},
	})
	testhelp.Equal(t, "State", l.State, "starting")
	testhelp.Equal(t, "Restarts", l.Restarts, []time.Time{gdAt(15, 20, 0)})
	testhelp.Equal(t, "SessionID", l.SessionID, "id-1")
	if l.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "LastExit", *l.LastExit, control.Exit{Code: 4, At: gdAt(15, 20, 0), Stderr: "", Restarts: 1})
}

func TestRuntimeGuardExample2(t *testing.T) {
	cfg := gdCfg()
	ids := func() string { return "id-1" }
	stderr := "Error: Invalid session ID. Must be a valid UUID.\n"

	l := gdRunning(t)
	l.Exited(4, "", gdAt(15, 20, 0), cfg)
	l.StartChecked(gitrules.Start{Mode: "fresh"}, ids, gdAt(15, 20, 5))
	testhelp.Equal(t, "Restarts after example 1", l.Restarts, []time.Time{gdAt(15, 20, 0)})

	acts := l.Exited(1, stderr, gdAt(15, 30, 0), cfg)
	testhelp.Equal(t, "second exit actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 1), restarting", Numbers: "restart 2 of 3 this hour · Error: Invalid session ID. Must be a valid UUID."}},
		{Kind: "start_check", Phase: "P17"},
	})
	l.StartChecked(gitrules.Start{Mode: "resume", Reason: "dirty tree (1 paths)"}, ids, gdAt(15, 30, 5))

	acts = l.Exited(137, "", gdAt(15, 40, 0), cfg)
	testhelp.Equal(t, "third exit actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 137), restarting", Numbers: "restart 3 of 3 this hour"}},
		{Kind: "start_check", Phase: "P17"},
	})
	l.StartChecked(gitrules.Start{Mode: "fresh"}, ids, gdAt(15, 40, 5))

	acts = l.Exited(1, stderr, gdAt(15, 50, 0), cfg)
	testhelp.Equal(t, "fourth exit actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: session keeps exiting", Numbers: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID."}},
	})
	testhelp.Equal(t, "State", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("Stop is nil")
	}
	testhelp.Equal(t, "Stop", *l.Stop, control.Stop{
		Kind:   "crash",
		Phase:  "P17",
		Reason: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.",
		At:     gdAt(15, 50, 0),
	})
	testhelp.Equal(t, "Queue.State", l.Queue.State, "stopped")
	testhelp.Equal(t, "Restarts", l.Restarts, []time.Time{gdAt(15, 20, 0), gdAt(15, 30, 0), gdAt(15, 40, 0)})
	if l.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "LastExit", *l.LastExit, control.Exit{
		Code:     1,
		At:       gdAt(15, 50, 0),
		Stderr:   "Error: Invalid session ID. Must be a valid UUID.",
		Restarts: 3,
	})

	// The alternate: the fourth exit arrives at 16:25:00Z, more than an hour
	// after the 15:20 restart, so the older stamp is dropped and a restart
	// happens instead of a crash stop.
	l2 := gdRunning(t)
	l2.Exited(4, "", gdAt(15, 20, 0), cfg)
	l2.StartChecked(gitrules.Start{Mode: "fresh"}, ids, gdAt(15, 20, 5))
	l2.Exited(1, stderr, gdAt(15, 30, 0), cfg)
	l2.StartChecked(gitrules.Start{Mode: "resume", Reason: "dirty tree (1 paths)"}, ids, gdAt(15, 30, 5))
	l2.Exited(137, "", gdAt(15, 40, 0), cfg)
	l2.StartChecked(gitrules.Start{Mode: "fresh"}, ids, gdAt(15, 40, 5))

	acts = l2.Exited(1, stderr, gdAt(16, 25, 0), cfg)
	testhelp.Equal(t, "aged-out actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 1), restarting", Numbers: "restart 3 of 3 this hour · Error: Invalid session ID. Must be a valid UUID."}},
		{Kind: "start_check", Phase: "P17"},
	})
	testhelp.Equal(t, "aged-out Restarts", l2.Restarts, []time.Time{gdAt(15, 30, 0), gdAt(15, 40, 0), gdAt(16, 25, 0)})
	testhelp.Equal(t, "aged-out State", l2.State, "starting")
}

func TestRuntimeGuardExample3(t *testing.T) {
	cfg := gdCfg()
	l := gdRunning(t)
	l.StretchUSD = 28
	l.SessionUSD = 0

	acts := l.Turn(stream.Result{TotalCostUSD: 1.5}, gdAt(15, 10, 0), cfg)
	testhelp.Equal(t, "first Turn actions", acts, []Action(nil))
	testhelp.Equal(t, "SessionUSD after first Turn", l.SessionUSD, 1.5)

	acts = l.Turn(stream.Result{TotalCostUSD: 2.25}, gdAt(16, 0, 0), cfg)
	testhelp.Equal(t, "second Turn actions", acts, []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: stretch cap", Numbers: "stretch cap $30.00 reached ($30.2500)"}},
	})
	testhelp.Equal(t, "State", l.State, "waiting")
	if l.Stop == nil {
		t.Fatal("Stop is nil")
	}
	testhelp.Equal(t, "Stop.Kind", l.Stop.Kind, "cap")

	// With cfg.StretchUSD 0 there is no cap to reach, so Turn returns nil.
	l2 := gdRunning(t)
	l2.StretchUSD = 28
	l2.SessionUSD = 0
	l2.Turn(stream.Result{TotalCostUSD: 1.5}, gdAt(15, 10, 0), cfg)
	zero := Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 0, UsageAlertPercent: 50}
	acts = l2.Turn(stream.Result{TotalCostUSD: 2.25}, gdAt(16, 0, 0), zero)
	testhelp.Equal(t, "no-cap Turn actions", acts, []Action(nil))
}

func TestRuntimeGuardExample4(t *testing.T) {
	cfg := gdCfg()
	now := gdAt(15, 10, 0)
	limits := stream.Limits{
		FiveHour: stream.Window{Utilization: 0.22, ResetsAt: 1791469200},
		SevenDay: stream.Window{Utilization: 0.6, ResetsAt: 1791853200},
	}

	l := gdRunning(t)
	acts := l.Limits(limits, now, cfg)
	testhelp.Equal(t, "Limits actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "info", Headline: "P17: weekly usage 60%", Numbers: "resets 2026-10-13T01:00:00Z"}},
	})
	testhelp.Equal(t, "Alerted", l.Alerted, true)

	acts = l.Limits(limits, now, cfg)
	testhelp.Equal(t, "second Limits actions", acts, []Action(nil))

	zero := Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 0}
	l2 := gdRunning(t)
	acts = l2.Limits(limits, now, zero)
	testhelp.Equal(t, "zero-percent Limits actions", acts, []Action(nil))
	testhelp.Equal(t, "zero-percent Alerted", l2.Alerted, false)
}

func TestRuntimeGuardExample8(t *testing.T) {
	cfg := gdCfg()
	now := gdAt(15, 5, 0)

	l := New("demo", gdQueue(t))
	l.State = "waiting"

	testhelp.Equal(t, "Exited while waiting", l.Exited(0, "bye\n", now, cfg), []Action(nil))
	testhelp.Equal(t, "Tick while waiting", l.Tick(now, cfg), []Action(nil))
	testhelp.Equal(t, "Turn while waiting", l.Turn(stream.Result{TotalCostUSD: 99}, now, cfg), []Action(nil))

	idle := New("demo", gdQueue(t))
	testhelp.Equal(t, "Limits while idle", idle.Limits(stream.Limits{
		FiveHour: stream.Window{Utilization: 1, ResetsAt: 1791469200},
	}, now, cfg), []Action(nil))

	testhelp.Equal(t, "waiting State", l.State, "waiting")
	testhelp.Equal(t, "waiting Restarts", l.Restarts, []time.Time(nil))
	testhelp.Equal(t, "waiting LastExit", l.LastExit, (*control.Exit)(nil))
	testhelp.Equal(t, "idle State", idle.State, "idle")
}

func TestRuntimeGuardExample9(t *testing.T) {
	cfg := gdCfg()

	l := gdRunning(t)
	acts := l.Exited(2, "\n\n node: bad option --foo \n(use node --help for usage)\n", gdAt(15, 5, 0), cfg)
	testhelp.Equal(t, "first actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 2), restarting", Numbers: "restart 1 of 3 this hour · node: bad option --foo"}},
		{Kind: "start_check", Phase: "P17"},
	})
	if l.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "first LastExit.Stderr", l.LastExit.Stderr, "node: bad option --foo")

	l2 := gdRunning(t)
	acts = l2.Exited(3, strings.Repeat("x", 250)+"\nsecond\n", gdAt(15, 6, 0), cfg)
	testhelp.Equal(t, "second actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 3), restarting", Numbers: "restart 1 of 3 this hour · " + strings.Repeat("x", 200)}},
		{Kind: "start_check", Phase: "P17"},
	})

	l3 := gdRunning(t)
	zero := Config{MaxResumesPerHour: 0, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}
	acts = l3.Exited(-3, "claude: start fork/exec /opt/claude/bin/claude: no such file or directory", gdAt(15, 7, 0), zero)
	testhelp.Equal(t, "third actions", acts, []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: session keeps exiting", Numbers: "exited 1 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory"}},
	})
	testhelp.Equal(t, "third State", l3.State, "waiting")
	if l3.Stop == nil {
		t.Fatal("Stop is nil")
	}
	testhelp.Equal(t, "third Stop.Kind", l3.Stop.Kind, "crash")
	testhelp.Equal(t, "third Restarts", l3.Restarts, []time.Time(nil))
	if l3.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "third LastExit", *l3.LastExit, control.Exit{
		Code:     -3,
		At:       gdAt(15, 7, 0),
		Stderr:   "claude: start fork/exec /opt/claude/bin/claude: no such file or directory",
		Restarts: 0,
	})
}
