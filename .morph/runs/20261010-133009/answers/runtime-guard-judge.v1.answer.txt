package supervisor

import (
	"encoding/json"
	"os"
	"strings"
	"testing"
	"time"

	"morphstudio/control"
	"morphstudio/gitrules"
	"morphstudio/internal/testhelp"
	"morphstudio/queue"
	"morphstudio/stream"
)

func gdPlan(t testing.TB) queue.Queue {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var p queue.Plan
	if err := json.Unmarshal(data, &p); err != nil {
		t.Fatal(err)
	}
	q, err := queue.Load(p, queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5})
	if err != nil {
		t.Fatal(err)
	}
	return q
}

func gdID() string { return "id-1" }

func gdRunning(t testing.TB) *Loop {
	t.Helper()
	l := New("demo", gdPlan(t))
	if _, err := l.Begin(time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC)); err != nil {
		t.Fatal(err)
	}
	l.StartChecked(gitrules.Start{Mode: "fresh"}, gdID, time.Date(2026, 10, 8, 15, 0, 1, 0, time.UTC))
	return l
}

func TestRuntimeGuardExample1(t *testing.T) {
	cfg := Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}
	l := gdRunning(t)
	at := time.Date(2026, 10, 8, 15, 20, 0, 0, time.UTC)
	got := l.Exited(4, "", at, cfg)
	want := []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 4), restarting", Numbers: "restart 1 of 3 this hour"}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "Exited actions", got, want)
	testhelp.Equal(t, "State", l.State, "starting")
	testhelp.Equal(t, "Restarts", l.Restarts, []time.Time{at})
	testhelp.Equal(t, "SessionID", l.SessionID, "id-1")
	if l.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "LastExit", *l.LastExit, control.Exit{Code: 4, At: at, Stderr: "", Restarts: 1})
}

func TestRuntimeGuardExample2(t *testing.T) {
	cfg := Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}
	l := gdRunning(t)
	t1 := time.Date(2026, 10, 8, 15, 20, 0, 0, time.UTC)
	l.Exited(4, "", t1, cfg)
	l.StartChecked(gitrules.Start{Mode: "fresh"}, gdID, time.Date(2026, 10, 8, 15, 20, 5, 0, time.UTC))

	t2 := time.Date(2026, 10, 8, 15, 30, 0, 0, time.UTC)
	got := l.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", t2, cfg)
	want := []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 1), restarting", Numbers: "restart 2 of 3 this hour · Error: Invalid session ID. Must be a valid UUID."}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "second exit actions", got, want)

	l.StartChecked(gitrules.Start{Mode: "resume", Reason: "dirty tree (1 paths)"}, gdID, time.Date(2026, 10, 8, 15, 30, 5, 0, time.UTC))
	t3 := time.Date(2026, 10, 8, 15, 40, 0, 0, time.UTC)
	got = l.Exited(137, "", t3, cfg)
	want = []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 137), restarting", Numbers: "restart 3 of 3 this hour"}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "third exit actions", got, want)

	l.StartChecked(gitrules.Start{Mode: "fresh"}, gdID, time.Date(2026, 10, 8, 15, 40, 5, 0, time.UTC))
	t4 := time.Date(2026, 10, 8, 15, 50, 0, 0, time.UTC)
	got = l.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", t4, cfg)
	want = []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: session keeps exiting", Numbers: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID."}},
	}
	testhelp.Equal(t, "fourth exit actions", got, want)
	testhelp.Equal(t, "State", l.State, "waiting")
	testhelp.Equal(t, "Stop", l.Stop, &control.Stop{Kind: "crash", Phase: "P17", Reason: "exited 4 times within an hour (last code 1): Error: Invalid session ID. Must be a valid UUID.", At: t4})
	testhelp.Equal(t, "Queue.State", l.Queue.State, "stopped")
	testhelp.Equal(t, "Restarts", l.Restarts, []time.Time{t1, t2, t3})
	if l.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "LastExit", *l.LastExit, control.Exit{Code: 1, At: t4, Stderr: "Error: Invalid session ID. Must be a valid UUID.", Restarts: 3})

	l2 := gdRunning(t)
	l2.Exited(4, "", t1, cfg)
	l2.StartChecked(gitrules.Start{Mode: "fresh"}, gdID, time.Date(2026, 10, 8, 15, 20, 5, 0, time.UTC))
	l2.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", t2, cfg)
	l2.StartChecked(gitrules.Start{Mode: "resume", Reason: "dirty tree (1 paths)"}, gdID, time.Date(2026, 10, 8, 15, 30, 5, 0, time.UTC))
	l2.Exited(137, "", t3, cfg)
	l2.StartChecked(gitrules.Start{Mode: "fresh"}, gdID, time.Date(2026, 10, 8, 15, 40, 5, 0, time.UTC))
	t5 := time.Date(2026, 10, 8, 16, 25, 0, 0, time.UTC)
	got = l2.Exited(1, "Error: Invalid session ID. Must be a valid UUID.\n", t5, cfg)
	want = []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 1), restarting", Numbers: "restart 3 of 3 this hour · Error: Invalid session ID. Must be a valid UUID."}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "late fourth exit actions", got, want)
	testhelp.Equal(t, "late Restarts", l2.Restarts, []time.Time{t2, t3, t5})
	testhelp.Equal(t, "late State", l2.State, "starting")
}

func TestRuntimeGuardExample3(t *testing.T) {
	cfg := Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}
	l := gdRunning(t)
	l.StretchUSD = 28
	l.SessionUSD = 0
	got := l.Turn(stream.Result{TotalCostUSD: 1.5}, time.Date(2026, 10, 8, 15, 30, 0, 0, time.UTC), cfg)
	testhelp.Equal(t, "first Turn", got, []Action(nil))
	testhelp.Equal(t, "SessionUSD", l.SessionUSD, 1.5)
	l.Turn(stream.Result{TotalCostUSD: 2.25}, time.Date(2026, 10, 8, 16, 0, 0, 0, time.UTC), cfg)
	want := []Action{
		{Kind: "kill"},
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: stretch cap", Numbers: "stretch cap $30.00 reached ($30.2500)"}},
	}
	testhelp.Equal(t, "second Turn actions", got, want)
	testhelp.Equal(t, "State", l.State, "waiting")
	testhelp.Equal(t, "Stop.Kind", l.Stop.Kind, "cap")

	l2 := gdRunning(t)
	l2.StretchUSD = 28
	zero := Config{StretchUSD: 0}
	l2.Turn(stream.Result{TotalCostUSD: 1.5}, time.Date(2026, 10, 8, 16, 0, 0, 0, time.UTC), zero)
	got = l2.Turn(stream.Result{TotalCostUSD: 2.25}, time.Date(2026, 10, 8, 16, 0, 0, 0, time.UTC), zero)
	testhelp.Equal(t, "zero stretch actions", got, []Action(nil))
}

func TestRuntimeGuardExample7(t *testing.T) {

	got := l.Tick(time.Date(2026, 10, 8, 15, 29, 0, 0, time.UTC), cfg)

	}
	testhelp.Equal(t, "15:31 actions", got, want)
	testhelp.Equal(t, "Nudged", l.Nudged, true)

	got = l.Tick(time.Date(2026, 10, 8, 16, 10, 0, 0, time.UTC), cfg)
	testhelp.Equal(t, "16:10 actions", got, []Action(nil))

	l.Event(time.Date(2026, 10, 8, 16, 20, 0, 0, time.UTC))
	testhelp.Equal(t, "LastEventAt", l.LastEventAt, time.Date(2026, 10, 8, 16, 20, 0, 0, time.UTC))

	want = []Action{

func TestRuntimeGuardExample8(t *testing.T) {

	testhelp.Equal(t, "waiting State", l.State, "waiting")
	testhelp.Equal(t, "waiting Restarts", l.Restarts, []time.Time(nil))
	testhelp.Equal(t, "waiting LastExit", l.LastExit, (*control.Exit)(nil))
	testhelp.Equal(t, "idle State", idle.State, "idle")
}

func TestRuntimeGuardExample9(t *testing.T) {
	cfg := Config{MaxResumesPerHour: 3, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50}

	l := gdRunning(t)
	at := time.Date(2026, 10, 8, 15, 5, 0, 0, time.UTC)
	got := l.Exited(2, "\n\n node: bad option --foo \n(use node --help for usage)\n", at, cfg)
	want := []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 2), restarting", Numbers: "restart 1 of 3 this hour · node: bad option --foo"}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "first actions", got, want)
	if l.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "first LastExit.Stderr", l.LastExit.Stderr, "node: bad option --foo")

	l2 := gdRunning(t)
	got = l2.Exited(3, strings.Repeat("x", 250)+"\nsecond\n", time.Date(2026, 10, 8, 15, 6, 0, 0, time.UTC), cfg)
	want = []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "watchdog", Headline: "P17: session exited (code 3), restarting", Numbers: "restart 1 of 3 this hour · " + strings.Repeat("x", 200)}},
		{Kind: "start_check", Phase: "P17"},
	}
	testhelp.Equal(t, "second actions", got, want)

	l3 := gdRunning(t)
	at3 := time.Date(2026, 10, 8, 15, 7, 0, 0, time.UTC)
	got = l3.Exited(-3, "claude: start fork/exec /opt/claude/bin/claude: no such file or directory", at3, Config{MaxResumesPerHour: 0, StallMinutes: 30, StretchUSD: 30, UsageAlertPercent: 50})
	want = []Action{
		{Kind: "post", Milestone: control.Milestone{Kind: "stop", Headline: "P17: session keeps exiting", Numbers: "exited 1 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory"}},
	}
	testhelp.Equal(t, "third actions", got, want)
	testhelp.Equal(t, "third State", l3.State, "waiting")
	testhelp.Equal(t, "third Stop.Kind", l3.Stop.Kind, "crash")
	testhelp.Equal(t, "third Restarts", l3.Restarts, []time.Time(nil))
	if l3.LastExit == nil {
		t.Fatal("LastExit is nil")
	}
	testhelp.Equal(t, "third LastExit", *l3.LastExit, control.Exit{Code: -3, At: at3, Stderr: "claude: start fork/exec /opt/claude/bin/claude: no such file or directory", Restarts: 0})
}
