func (f *acFake) AdoptProject(ctx context.Context, name, dir, repoURL, language string) (ProjectView, error) {
	f.calls = append(f.calls, [4]string{name, dir, repoURL, language})
	var view ProjectView
	var err error
	if f.callIndex < len(f.views) {
		view = f.views[f.callIndex]
	}
	if f.callIndex < len(f.errs) {
		err = f.errs[f.callIndex]
	}
	f.callIndex++
	return view, err
}

func TestAdoptContractExample1(t *testing.T) {
	c := &acFake{
		views: []ProjectView{{
			Name:      "studio",
			Language:  "go",
			RepoURL:   "https://github.com/VasyaLutiy/MorphStudio",
			State:     "idle",
			CreatedAt: time.Date(2026, 10, 10, 18, 0, 0, 0, time.UTC),
		}},
		errs: []error{nil},
	}
	want := ProjectView{
		Name:      "studio",
		Language:  "go",
		RepoURL:   "https://github.com/VasyaLutiy/MorphStudio",
		State:     "idle",
		CreatedAt: time.Date(2026, 10, 10, 18, 0, 0, 0, time.UTC),
	}

	got, err := Adopt(context.Background(), c, "studio", "/home/morph/MorphStudio", "https://github.com/VasyaLutiy/MorphStudio", "go")

	testhelp.Equal(t, "view", got, want)
	testhelp.Equal(t, "err", err, nil)
	testhelp.Equal(t, "calls", c.calls, [][4]string{
		{"studio", "/home/morph/MorphStudio", "https://github.com/VasyaLutiy/MorphStudio", "go"},
	})
	var _ Adopter = c
}

func TestAdoptContractExample2(t *testing.T) {
	p := &acPlain{}

	got, err := Adopt(context.Background(), p, "studio", "/x", "https://github.com/o/r", "go")

	testhelp.Equal(t, "view", got, ProjectView{})
	if !errors.Is(err, ErrUnsupported) {
		t.Errorf("errors.Is ErrUnsupported: got %v", err)
	}
	testhelp.Equal(t, "err.Error()", err.Error(), "adopt not supported")
	code, name := Code(ErrUnsupported)
	testhelp.Equal(t, "Code", []any{code, name}, []any{500, "internal"})
}

func TestAdoptContractExample3(t *testing.T) {
	c := &acFake{
		views: []ProjectView{{}, {}},
		errs: []error{
			fmt.Errorf("%w: bootstrap: refused: not a git repository: /home/morph/nowhere", ErrBadInput),
			ErrExists,
		},
	}

	_, err1 := Adopt(context.Background(), c, "gamma", "/home/morph/nowhere", "https://github.com/o/gamma", "go")
	_, err2 := Adopt(context.Background(), c, "demo", "/home/morph/demo", "https://github.com/o/demo", "python")

	if !errors.Is(err1, ErrBadInput) {
		t.Errorf("errors.Is ErrBadInput: got %v", err1)
	}
	testhelp.Equal(t, "err1.Error()", err1.Error(), "bad input: bootstrap: refused: not a git repository: /home/morph/nowhere")
	code1, name1 := Code(err1)
	testhelp.Equal(t, "Code1", []any{code1, name1}, []any{400, "bad_input"})

	if !errors.Is(err2, ErrExists) {
		t.Errorf("errors.Is ErrExists: got %v", err2)
	}
	code2, name2 := Code(err2)
	testhelp.Equal(t, "Code2", []any{code2, name2}, []any{409, "exists"})

	testhelp.Equal(t, "calls", c.calls, [][4]string{
		{"gamma", "/home/morph/nowhere", "https://github.com/o/gamma", "go"},
		{"demo", "/home/morph/demo", "https://github.com/o/demo", "python"},
	})
}
