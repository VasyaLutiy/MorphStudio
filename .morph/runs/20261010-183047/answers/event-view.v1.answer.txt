package control

import (
	"encoding/json"
	"fmt"

	"morphstudio/eventlog"
	"morphstudio/stream"
)

// EventExit is the exit payload nested in an EventView.
type EventExit struct {
	Code   int    `json:"code"`
	Stderr string `json:"stderr"`
}

// EventView is the compact view of one event log entry.
type EventView struct {
	Seq            int64      `json:"seq"`
	Dir            string     `json:"dir"`
	Type           string     `json:"type"`
	Subtype        string     `json:"subtype,omitempty"`
	Text           string     `json:"text,omitempty"`
	Tools          []string   `json:"tools,omitempty"`
	CostUSD        float64    `json:"cost_usd,omitempty"`
	TerminalReason string     `json:"terminal_reason,omitempty"`
	Exit           *EventExit `json:"exit,omitempty"`
}

// EventsView is a window of EventViews.
type EventsView struct {
	Events []EventView `json:"events"`
	Last   int64       `json:"last"`
}

// Cut returns s when it holds at most max runes, and otherwise its first max
// runes followed by one U+2026. A max <= 0 means 200.
func Cut(s string, max int) string {
	if max <= 0 {
		max = 200
	}
	count := 0
	for i := range s {
		if count == max {
			return s[:i] + "…"
		}
		count++
	}
	return s
}

// View turns one event log entry into its compact view.
func View(e eventlog.Entry, max int) EventView {
	v := EventView{Seq: e.Seq, Dir: e.Dir}

	if e.Dir == "exit" {
		v.Type = "exit"
		var payload EventExit
		if err := json.Unmarshal(e.Msg, &payload); err == nil {
			v.Exit = &EventExit{Code: payload.Code, Stderr: Cut(payload.Stderr, max)}
		}
		return v
	}

	ev, err := stream.Parse(e.Msg)
	if err != nil {
		v.Type = "raw"
		var s string
		if json.Unmarshal(e.Msg, &s) == nil {
			v.Text = Cut(s, max)
		} else {
			v.Text = Cut(string(e.Msg), max)
		}
		return v
	}

	v.Type = ev.Type
	v.Subtype = ev.Subtype

	switch ev.Type {
	case "assistant":
		v.Text = Cut(ev.Text, max)
		v.Tools = assistantTools(ev)
	case "user":
		decodeUser(ev, &v, max)
	case "result":
		v.Text = Cut(ev.Text, max)
		if ev.Result != nil {
			v.CostUSD = ev.Result.TotalCostUSD
			v.TerminalReason = ev.Result.TerminalReason
		}
	case "rate_limit_event":
		if ev.Limits != nil {
			v.Text = fmt.Sprintf("five_hour %d%% · seven_day %d%%",
				Percent(ev.Limits.FiveHour.Utilization),
				Percent(ev.Limits.SevenDay.Utilization))
		}
	case "control_request":
		if ev.Subtype == "can_use_tool" {
			v.Tools = []string{ev.Tool}
			if ev.Question != nil {
				v.Text = Cut(ev.Question.Text, max)
			}
		}
	}
	return v
}

// Views maps a window of entries to their views. It returns a non-nil, empty
// Events slice when there are none.
func Views(e Events, max int) EventsView {
	out := EventsView{Events: []EventView{}, Last: e.Last}
	for _, entry := range e.Entries {
		out.Events = append(out.Events, View(entry, max))
	}
	return out
}

// assistantTools returns the names of message.content elements of type
// tool_use, or nil when there are none.
func assistantTools(ev stream.Event) []string {
	var top map[string]json.RawMessage
	if json.Unmarshal(ev.Raw, &top) != nil {
		return nil
	}
	msgRaw, ok := top["message"]
	if !ok {
		return nil
	}
	var msg map[string]json.RawMessage
	if json.Unmarshal(msgRaw, &msg) != nil {
		return nil
	}
	cRaw, ok := msg["content"]
	if !ok {
		return nil
	}
	var parts []json.RawMessage
	if json.Unmarshal(cRaw, &parts) != nil {
		return nil
	}
	var tools []string
	for _, p := range parts {
		var pm map[string]json.RawMessage
		if json.Unmarshal(p, &pm) != nil {
			continue
		}
		var pt string
		if json.Unmarshal(pm["type"], &pt) != nil || pt != "tool_use" {
			continue
		}
		var name string
		if json.Unmarshal(pm["name"], &name) == nil {
			tools = append(tools, name)
		}
	}
	if len(tools) == 0 {
		return nil
	}
	return tools
}

// decodeUser fills Text and Subtype from a "user" event's message.content:
// a JSON string is used as-is; an array takes the first element's type as
// Subtype and, when that type is "text", its text as Text.
func decodeUser(ev stream.Event, v *EventView, max int) {
	var top map[string]json.RawMessage
	if json.Unmarshal(ev.Raw, &top) != nil {
		return
	}
	msgRaw, ok := top["message"]
	if !ok {
		return
	}
	var msg map[string]json.RawMessage
	if json.Unmarshal(msgRaw, &msg) != nil {
		return
	}
	cRaw, ok := msg["content"]
	if !ok {
		return
	}
	var s string
	if json.Unmarshal(cRaw, &s) == nil {
		v.Text = Cut(s, max)
		return
	}
	var parts []json.RawMessage
	if json.Unmarshal(cRaw, &parts) != nil || len(parts) == 0 {
		return
	}
	var pm map[string]json.RawMessage
	if json.Unmarshal(parts[0], &pm) != nil {
		return
	}
	var pt string
	if json.Unmarshal(pm["type"], &pt) != nil {
		return
	}
	v.Subtype = pt
	if pt == "text" {
		var tx string
		if json.Unmarshal(pm["text"], &tx) == nil {
			v.Text = Cut(tx, max)
		}
	}
}
