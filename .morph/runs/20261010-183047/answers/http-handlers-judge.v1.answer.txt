type hhWrap struct {
	msg string
	err error
}

func (e hhWrap) Error() string { return e.msg }
func (e hhWrap) Unwrap() error { return e.err }

func TestHTTPHandlersExample1(t *testing.T) {
	fake := &hhFake{
		statusFn: func(project string) (control.Status, error) {
			if project == "demo" {
				return control.Status{
					State:     "busy",
					Phase:     "P18",
					Minutes:   37,
					CostUSD:   1.92,
					FiveHour:  22,
					SevenDay:  60,
					SessionID: "s-1",
					Queue: control.QueueView{
						State:    "running",
						Approved: "7b31dfe",
						Index:    1,
						Phases:   3,
						Current:  "P18",
						Reason:   "",
					},
					Repo: github.Access{
						Checked:   true,
						Reachable: true,
						Push:      true,
						Reason:    "",
						CheckedAt: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC),
					},
				}, nil
			}
			return control.Status{}, control.ErrUnknownProject
		},
	}
	h := Handlers{Control: fake}

	req := httptest.NewRequest("GET", "/projects/demo/status", nil)
	req.SetPathValue("project", "demo")
	rec := httptest.NewRecorder()
	h.Status(rec, req)

	testhelp.Equal(t, "status", rec.Code, http.StatusOK)
	testhelp.Equal(t, "content-type", rec.Header().Get("Content-Type"), "application/json")
	want := `{"state":"busy","phase":"P18","minutes":37,"cost_usd":1.92,"five_hour":22,"seven_day":60,"session_id":"s-1","queue":{"state":"running","approved":"7b31dfe","index":1,"phases":3,"current":"P18","reason":""},"repo":{"checked":true,"reachable":true,"push":true,"reason":"","checked_at":"2026-10-08T15:00:00Z"}}` + "\n"
	testhelp.Equal(t, "body", rec.Body.String(), want)

	req2 := httptest.NewRequest("GET", "/projects/nope/status", nil)
	req2.SetPathValue("project", "nope")
	rec2 := httptest.NewRecorder()
	h.Status(rec2, req2)

	testhelp.Equal(t, "status", rec2.Code, http.StatusNotFound)
	testhelp.Equal(t, "body", rec2.Body.String(), `{"error":{"code":"unknown_project","message":"unknown project"}}`+"\n")
}

func TestHTTPHandlersExample4(t *testing.T) {
	badBodies := []string{
		`{"option":`,
		`{"option":1,"x":2}`,
		`{"option":1} {}`,
		``,
	}
	fake := &hhFake{}
	h := Handlers{Control: fake}
	for _, body := range badBodies {
		var req *http.Request
		if body != "" {
			req = httptest.NewRequest("POST", "/projects/demo/answer", strings.NewReader(body))
		} else {
			req = httptest.NewRequest("POST", "/projects/demo/answer", nil)
		}
		req.Header.Set("Content-Type", "application/json")
		req.SetPathValue("project", "demo")
		rec := httptest.NewRecorder()
		h.Answer(rec, req)
		testhelp.Equal(t, "status", rec.Code, http.StatusBadRequest)
		testhelp.Equal(t, "body", rec.Body.String(), `{"error":{"code":"bad_json","message":"bad json"}}`+"\n")
	}
	testhelp.Equal(t, "calls", len(fake.calls), 0)

	fake.answerErr = nil
	req := httptest.NewRequest("POST", "/projects/demo/answer", strings.NewReader(`{"option":2}`))
	req.Header.Set("Content-Type", "application/json")
	req.SetPathValue("project", "demo")
	rec := httptest.NewRecorder()
	h.Answer(rec, req)
	testhelp.Equal(t, "status", rec.Code, http.StatusOK)
	testhelp.Equal(t, "body", rec.Body.String(), `{"ok":true}`+"\n")
	testhelp.Equal(t, "calls", fake.calls, []string{`Answer("demo", 2)`})

	fake.answerErr = hhWrap{msg: "option out of range: 3 of 1..2", err: control.ErrBadOption}
	req2 := httptest.NewRequest("POST", "/projects/demo/answer", strings.NewReader(`{"option":3}`))
	req2.Header.Set("Content-Type", "application/json")
	req2.SetPathValue("project", "demo")
	rec2 := httptest.NewRecorder()
	h.Answer(rec2, req2)
	testhelp.Equal(t, "status", rec2.Code, http.StatusBadRequest)
	testhelp.Equal(t, "body", rec2.Body.String(), `{"error":{"code":"bad_option","message":"option out of range: 3 of 1..2"}}`+"\n")
}

func TestHTTPHandlersExample9(t *testing.T) {
	a := &hhAdopter{
		hhFake: &hhFake{},
		adoptView: control.ProjectView{
			Name:      "studio",
			Language:  "go",
			RepoURL:   "https://github.com/VasyaLutiy/MorphStudio",
			State:     "idle",
			CreatedAt: time.Date(2026, 10, 10, 18, 0, 0, 0, time.UTC),
		},
	}
	h := Handlers{Control: a}

	body1 := `{"name":"studio","dir":"/home/morph/MorphStudio","repo_url":"https://github.com/VasyaLutiy/MorphStudio","language":"go"}`
	req := httptest.NewRequest("POST", "/projects/adopt", strings.NewReader(body1))
	req.Header.Set("Content-Type", "application/json")
	rec := httptest.NewRecorder()
	h.AdoptProject(rec, req)
	testhelp.Equal(t, "status", rec.Code, http.StatusCreated)
	want := `{"name":"studio","language":"go","repo_url":"https://github.com/VasyaLutiy/MorphStudio","state":"idle","created_at":"2026-10-10T18:00:00Z"}` + "\n"
	testhelp.Equal(t, "body", rec.Body.String(), want)
	testhelp.Equal(t, "args", a.adoptArgs, []string{"studio", "/home/morph/MorphStudio", "https://github.com/VasyaLutiy/MorphStudio", "go"})

	a.adoptErr = hhWrap{msg: "bad input: bootstrap: refused: not a git repository: /home/morph/nowhere", err: control.ErrBadInput}
	req2 := httptest.NewRequest("POST", "/projects/adopt", strings.NewReader(body1))
	req2.Header.Set("Content-Type", "application/json")
	rec2 := httptest.NewRecorder()
	h.AdoptProject(rec2, req2)
	testhelp.Equal(t, "status", rec2.Code, http.StatusBadRequest)
	testhelp.Equal(t, "body", rec2.Body.String(), `{"error":{"code":"bad_input","message":"bad input: bootstrap: refused: not a git repository: /home/morph/nowhere"}}`+"\n")

	a.adoptErr = control.ErrExists
	req3 := httptest.NewRequest("POST", "/projects/adopt", strings.NewReader(body1))
	req3.Header.Set("Content-Type", "application/json")
	rec3 := httptest.NewRecorder()
	h.AdoptProject(rec3, req3)
	testhelp.Equal(t, "status", rec3.Code, http.StatusConflict)
	testhelp.Equal(t, "body", rec3.Body.String(), `{"error":{"code":"exists","message":"project exists"}}`+"\n")

	a.adoptErr = hhWrap{msg: "bad input: bootstrap: refused: github: not a github repository url: ", err: control.ErrBadInput}
	body4 := `{"name":"studio","dir":"/x"}`
	req4 := httptest.NewRequest("POST", "/projects/adopt", strings.NewReader(body4))
	req4.Header.Set("Content-Type", "application/json")
	rec4 := httptest.NewRecorder()
	h.AdoptProject(rec4, req4)
	testhelp.Equal(t, "status", rec4.Code, http.StatusBadRequest)
	testhelp.Equal(t, "body", rec4.Body.String(), `{"error":{"code":"bad_input","message":"bad input: bootstrap: refused: github: not a github repository url: "}}`+"\n")
	testhelp.Equal(t, "args", a.adoptArgs, []string{"studio", "/x", "", ""})

	a.adoptErr = nil
	body5 := `{"name":"studio","extra":1}`
	req5 := httptest.NewRequest("POST", "/projects/adopt", strings.NewReader(body5))
	req5.Header.Set("Content-Type", "application/json")
	rec5 := httptest.NewRecorder()
	h.AdoptProject(rec5, req5)
	testhelp.Equal(t, "status", rec5.Code, http.StatusBadRequest)
	testhelp.Equal(t, "body", rec5.Body.String(), `{"error":{"code":"bad_json","message":"bad json"}}`+"\n")
	testhelp.Equal(t, "args", a.adoptArgs, []string{"studio", "/x", "", ""})

	fakeNoAdopt := &hhFake{}
	h2 := Handlers{Control: fakeNoAdopt}
	req6 := httptest.NewRequest("POST", "/projects/adopt", strings.NewReader(body1))
	req6.Header.Set("Content-Type", "application/json")
	rec6 := httptest.NewRecorder()
	h2.AdoptProject(rec6, req6)
	testhelp.Equal(t, "status", rec6.Code, http.StatusInternalServerError)
	testhelp.Equal(t, "body", rec6.Body.String(), `{"error":{"code":"internal","message":"internal"}}`+"\n")
	testhelp.Equal(t, "calls", len(fakeNoAdopt.calls), 0)
}
