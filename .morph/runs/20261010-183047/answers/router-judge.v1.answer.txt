// rtRecord is one recorded Control call: the method name, the project it
// named, and any further string arguments.
type rtRecord struct {
	Method  string
	Project string
	Args    []string
}

// rtFake implements control.Control and records every call.
type rtFake struct {
	records []rtRecord
}

// rec appends one recorded call.
func (f *rtFake) rec(method, project string, args ...string) {
	f.records = append(f.records, rtRecord{Method: method, Project: project, Args: args})
}

func (f *rtFake) Projects() []control.ProjectView {
	f.rec("Projects", "")
	return nil
}

func (f *rtFake) CreateProject(ctx context.Context, name, language, repoURL string) (control.ProjectView, error) {
	f.rec("CreateProject", name, language, repoURL)
	return control.ProjectView{}, nil
}

	f.rec("Status", project)
	if project != "demo" {
		return control.Status{}, control.ErrUnknownProject

// rtServe drives h through httptest and returns the recorder.
func rtServe(t *testing.T, h http.Handler, method, path, auth, body string) *httptest.ResponseRecorder {
	t.Helper()
	var rdr io.Reader
	if body != "" {
		rdr = strings.NewReader(body)
	}
	req := httptest.NewRequest(method, path, rdr)
	if body != "" {
		req.Header.Set("Content-Type", "application/json")
	}
	if auth != "" {
		req.Header.Set("Authorization", auth)
	}
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	return rec
}

// TestRouterExample1: GET /healthz needs no bearer and answers {"status":"ok"}.
func TestRouterExample1(t *testing.T) {
	f := &rtFake{}
	r := NewRouter(Handlers{Control: f}, "tok-1", nil)

	rec := rtServe(t, r, http.MethodGet, "/healthz", "", "")
	testhelp.Equal(t, "healthz status", rec.Code, http.StatusOK)
	testhelp.Equal(t, "healthz content type", rec.Header().Get("Content-Type"), "application/json")
	testhelp.Equal(t, "healthz body", rec.Body.String(), `{"status":"ok"}`+"\n")
}

// TestRouterExample2: an unacceptable Authorization is refused with 401.
func TestRouterExample2(t *testing.T) {
	f := &rtFake{}
	r := NewRouter(Handlers{Control: f}, "tok-1", nil)

	const unauthorized = `{"error":{"code":"unauthorized","message":"bearer token required"}}`
	for _, auth := range []string{"", "Bearer wrong", "Basic dG9rLTE=", "Bearer tok-1 "} {
		rec := rtServe(t, r, http.MethodGet, "/projects/demo/status", auth, "")
		testhelp.Equal(t, auth+" status", rec.Code, http.StatusUnauthorized)
		testhelp.Equal(t, auth+" body", rec.Body.String(), unauthorized+"\n")
	}
}

// TestRouterExample3: a correct bearer reaches the handler; an unknown project is 404.
func TestRouterExample3(t *testing.T) {
	f := &rtFake{}
	r := NewRouter(Handlers{Control: f}, "tok-1", nil)

	const body = `{"state":"busy","phase":"P18","minutes":37,"cost_usd":0,"five_hour":0,"seven_day":0,"session_id":"s-1","queue":{"state":"","approved":"","index":0,"phases":0,"current":"","reason":""},"repo":{"checked":false,"reachable":false,"push":false,"reason":"","checked_at":"0001-01-01T00:00:00Z"}}`
	for _, auth := range []string{"Bearer tok-1", "bearer tok-1"} {
		rec := rtServe(t, r, http.MethodGet, "/projects/demo/status", auth, "")
		testhelp.Equal(t, auth+" status", rec.Code, http.StatusOK)
		testhelp.Equal(t, auth+" body", rec.Body.String(), body+"\n")
	}

	rec := rtServe(t, r, http.MethodGet, "/projects/nope/status", "Bearer tok-1", "")
	testhelp.Equal(t, "unknown project status", rec.Code, http.StatusNotFound)
}

// TestRouterExample5: the remaining project routes reach the control in order.
func TestRouterExample5(t *testing.T) {
	f := &rtFake{}
	r := NewRouter(Handlers{Control: f}, "tok-1", nil)
	const auth = "Bearer tok-1"

	cases := []struct {
		method string
		path   string
		body   string
	}{
		{http.MethodPost, "/projects/demo/messages", `{"text":"hi"}`},
		{http.MethodGet, "/projects/demo/pending", ""},
		{http.MethodPost, "/projects/demo/interrupt", ""},
		{http.MethodGet, "/projects/demo/usage", ""},
		{http.MethodPost, "/projects/demo/continue", ""},
		{http.MethodGet, "/projects/demo/stop", ""},
		{http.MethodPut, "/projects/demo/github-token", `{"token":"t"}`},
	}
	for _, c := range cases {
		rec := rtServe(t, r, c.method, c.path, auth, c.body)
		testhelp.Equal(t, c.method+" "+c.path+" status", rec.Code, http.StatusOK)
	}

	want := []rtRecord{
		{Method: "Order", Project: "demo", Args: []string{"hi"}},
		{Method: "Pending", Project: "demo"},
		{Method: "Interrupt", Project: "demo"},
		{Method: "Usage", Project: "demo"},
		{Method: "Continue", Project: "demo"},
		{Method: "StopCheck", Project: "demo"},
		{Method: "PutGithubToken", Project: "demo", Args: []string{"t"}},
	}
	testhelp.Equal(t, "records", f.records, want)
}

// TestRouterExample7: the MCP mount is not behind Bearer; a nil mount is not there.
func TestRouterExample7(t *testing.T) {
	f := &rtFake{}
	mcp := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(299)
		_, _ = io.WriteString(w, "mcp "+r.URL.Path)
	})
	r := NewRouter(Handlers{Control: f}, "tok-1", mcp)

	paths := []struct {
		method string
		path   string
	}{
		{http.MethodGet, "/mcp"},
		{http.MethodPost, "/mcp/demo"},
		{http.MethodPost, "/mcp/demo/session"},
	}
	for _, c := range paths {
		rec := rtServe(t, r, c.method, c.path, "", "")
		testhelp.Equal(t, c.path+" status", rec.Code, 299)
		testhelp.Equal(t, c.path+" body", rec.Body.String(), "mcp "+c.path)
	}

	f2 := &rtFake{}
	r2 := NewRouter(Handlers{Control: f2}, "tok-1", nil)
	for _, c := range paths {
		rec := rtServe(t, r2, c.method, c.path, "", "")
		testhelp.Equal(t, c.path+" nil mount status", rec.Code, http.StatusNotFound)
	}
}

// TestRouterExample8: adopt reaches the adopter, not CreateProject.
func TestRouterExample8(t *testing.T) {
	f := &rtFake{}
	r := NewRouter(Handlers{Control: rtAdopter{rtFake: f}}, "tok-1", nil)

	const adoptBody = `{"name":"studio","dir":"/home/morph/MorphStudio","repo_url":"https://github.com/VasyaLutiy/MorphStudio","language":"go"}`

	rec := rtServe(t, r, http.MethodPost, "/projects/adopt", "Bearer tok-1", adoptBody)
	testhelp.Equal(t, "adopt status", rec.Code, http.StatusCreated)
	testhelp.Equal(t, "adopt body", rec.Body.String(),
		`{"name":"studio","language":"go","repo_url":"https://github.com/VasyaLutiy/MorphStudio","state":"idle","created_at":"2026-10-10T18:00:00Z"}`+"\n")

	rec = rtServe(t, r, http.MethodPost, "/projects/adopt", "", adoptBody)
	testhelp.Equal(t, "adopt unauthorized status", rec.Code, http.StatusUnauthorized)
	testhelp.Equal(t, "adopt unauthorized body", rec.Body.String(),
		`{"error":{"code":"unauthorized","message":"bearer token required"}}`+"\n")

	rec = rtServe(t, r, http.MethodGet, "/projects/adopt", "Bearer tok-1", "")
	testhelp.Equal(t, "adopt wrong method status", rec.Code, http.StatusMethodNotAllowed)

	want := []rtRecord{
		{Method: "AdoptProject", Project: "studio", Args: []string{
			"/home/morph/MorphStudio", "https://github.com/VasyaLutiy/MorphStudio", "go"}},
	}
	testhelp.Equal(t, "adopt records", f.records, want)
}
