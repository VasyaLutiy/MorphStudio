// dcClock is the harness clock: Now returns base + n seconds and bumps n.
type dcClock struct {
	mu   sync.Mutex
	base time.Time
	n    int
}

func (c *dcClock) Now() time.Time {
	c.mu.Lock()
	defer c.mu.Unlock()
	t := c.base.Add(time.Duration(c.n) * time.Second)
	c.n++
	return t
}

func (c *dcClock) Add(d time.Duration) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.base = c.base.Add(d)
}

// dcHarness is the one test harness of this file.
type dcHarness struct {
	t      *testing.T
	tmp    string
	reg    *registry.Registry
	runner *runner.Fake
	clock  *dcClock

	mu       sync.Mutex
	launches []claude.Launch
	fakes    []*claude.Fake
	posts    [][4]string
	nextID   int
	spawnErr error

	do  func(*http.Request) (*http.Response, error)
	cfg Config
}

func dcNew(t *testing.T) *dcHarness {
	t.Helper()
	tmp := t.TempDir()
	reg, err := registry.Open(filepath.Join(tmp, "state"))
	if err != nil {
		t.Fatal(err)
	}
	h := &dcHarness{
		t:     t,
		tmp:   tmp,
		reg:   reg,
		clock: &dcClock{base: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC)},
		runner: &runner.Fake{Script: map[string]runner.Result{
			"git rev-parse --abbrev-ref HEAD": {Stdout: "main\n"},
			"git rev-parse HEAD":              {Stdout: "aaa111\n"},
			"git rev-parse origin/main":       {Stdout: "aaa111\n"},
			"git status --porcelain":          {Stdout: ""},
		}},
		do: dcHTTP(200, `{"permissions":{"push":true}}`),
		cfg: Config{
			ClaudeBin:   "/opt/claude/bin/claude",
			MorphBin:    "/usr/local/bin/morph",
			ProjectsDir: filepath.Join(tmp, "projects"),
			MCPBaseURL:  "http://127.0.0.1:7181",
			Defaults:    queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4},
			MaxParallel: 1,
			Loop: supervisor.Config{
				MaxResumesPerHour: 3,
				StallMinutes:      30,
				StretchUSD:        30,
				UsageAlertPercent: 50,
			},
		},
	}
	return h
}

func dcHTTP(code int, body string) func(*http.Request) (*http.Response, error) {
	return func(*http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: code,
			Body:       io.NopCloser(strings.NewReader(body)),
			Header:     http.Header{},
		}, nil
	}
}

func dcPlan(t *testing.T) queue.Plan {

func (h *dcHarness) newID() string {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.nextID++
	return fmt.Sprintf("id-%d", h.nextID)
}

func (h *dcHarness) snapLaunches() []claude.Launch {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([]claude.Launch, len(h.launches))
	copy(out, h.launches)
	return out
}

func (h *dcHarness) snapFakes() []*claude.Fake {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([]*claude.Fake, len(h.fakes))
	copy(out, h.fakes)
	return out
}

func (h *dcHarness) snapPosts() [][4]string {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([][4]string, len(h.posts))
	copy(out, h.posts)
	return out
}

func (h *dcHarness) open() *Daemon {
	h.t.Helper()
	spawn := func(ctx context.Context, l claude.Launch) (claude.Process, error) {
		h.mu.Lock()
		defer h.mu.Unlock()
		h.launches = append(h.launches, l)
		if h.spawnErr != nil {
			return nil, h.spawnErr
		}
		fp := claude.NewFake()
		h.fakes = append(h.fakes, fp)
		return fp, nil
	}
	post := func(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
		h.mu.Lock()
		defer h.mu.Unlock()
		h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
		return telegram.Status{}, nil
	}
	d, err := Open(context.Background(), Deps{
		Registry: h.reg,
		Runner:   h.runner,
		Spawn:    spawn,
		Post:     post,
		GitHubDo: h.do,
		Now:      h.clock.Now,
		NewID:    h.newID,
		Config:   h.cfg,
	})
	if err != nil {
		h.t.Fatal(err)
	}
	return d
}

func dcReadLog(t *testing.T, path string) []eventlog.Entry {
	t.Helper()
	b, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	var out []eventlog.Entry
	for _, line := range strings.Split(strings.TrimRight(string(b), "\n"), "\n") {
		if line == "" {
			continue
		}
		var e eventlog.Entry
		if err := json.Unmarshal([]byte(line), &e); err != nil {
			t.Fatal(err)
		}
		out = append(out, e)
	}
	return out
}

func dcEqArgs(got, want []string) bool {
	if len(got) != len(want) {
		return false
	}
	for i := range got {
		if got[i] != want[i] {
			return false
		}
	}
	return true
}

// Example 6: PhaseDone when HEAD is not origin/main stops the loop with
// not_pushed; Continue resumes it.
func TestDaemonCoreExample6(t *testing.T) {
	h := dcNew(t)
	h.addProject("demo", "go", "https://github.com/acme/demo")
	d := h.open()
	plan := dcPlan(t)

	_, err := d.PlanLoad("demo", plan)
	testhelp.Equal(t, "planload err", err, nil)
	fp := h.snapFakes()[0]

	h.runner.Script["git rev-parse HEAD"] = runner.Result{Stdout: "bbb222\n"}
	err = d.PhaseDone("demo", "id-2", "P17", "P18")
	testhelp.Equal(t, "phase done err", err, nil)

	h.runner.Script["git rev-parse HEAD"] = runner.Result{Stdout: "aaa111\n"}

	testhelp.Equal(t, "spawn count", len(h.snapLaunches()), 1)
	testhelp.Equal(t, "not killed", fp.Killed(), false)

	posts := h.snapPosts()
	testhelp.Equal(t, "last post", posts[len(posts)-1], [4]string{"demo", "stop", "P17 not pushed", "not pushed: HEAD bbb222, origin/main aaa111"})

	st, err := d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "state", st.State, "waiting")

	stop, err := d.StopCheck("demo")
	testhelp.Equal(t, "stopcheck err", err, nil)
	testhelp.Equal(t, "stop not nil", stop != nil, true)
	s := *stop
	testhelp.Equal(t, "stop at", s.At.IsZero(), false)
	s.At = time.Time{}
	testhelp.Equal(t, "stop", s, control.Stop{Kind: "not_pushed", Phase: "P17", Reason: "not pushed: HEAD bbb222, origin/main aaa111"})

	err = d.Continue("demo")
	testhelp.Equal(t, "continue err", err, nil)
	testhelp.Equal(t, "killed after continue", fp.Killed(), true)
	launches := h.snapLaunches()
	testhelp.Equal(t, "launches", len(launches), 2)
	testhelp.Equal(t, "second session", launches[1].SessionID, "id-3")
	testhelp.Equal(t, "second budget", launches[1].BudgetUSD, float64(12))

	st, err = d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "phase after continue", st.Phase, "P18")
}

// Example 8: PutGithubToken stores secrets, checks access, pushes on first
// success and reports failure otherwise.
func TestDaemonCoreExample8(t *testing.T) {
	h := dcNew(t)
	h.runner.Script["git rev-parse --verify -q origin/main"] = runner.Result{Code: 1}
	h.runner.Script["git push -q -u origin main"] = runner.Result{Code: 0}
	h.addProject("demo", "go", "https://github.com/acme/demo")
	d := h.open()

	res, err := d.PutGithubToken(context.Background(), "demo", "ghp_abc")
	testhelp.Equal(t, "put err", err, nil)
	testhelp.Equal(t, "checked", res.Access.Checked, true)
	testhelp.Equal(t, "at", res.Access.CheckedAt.IsZero(), false)
	res.Access.CheckedAt = time.Time{}
	testhelp.Equal(t, "result", control.TokenResult{Access: res.Access, Pushed: res.Pushed, PushError: res.PushError}, control.TokenResult{Access: github.Access{Checked: true, Reachable: true, Push: true}, Pushed: true, PushError: ""})

	tok, err := h.reg.ReadSecret("demo", "github-token")
	testhelp.Equal(t, "token err", err, nil)
	testhelp.Equal(t, "token", tok, "ghp_abc")

	cred, err := h.reg.ReadSecret("demo", "git-credentials")
	testhelp.Equal(t, "cred err", err, nil)
	testhelp.Equal(t, "cred", cred, "https://x-access-token:ghp_abc@github.com\n")

	fi, err := os.Stat(h.reg.SecretPath("demo", "github-token"))
	testhelp.Equal(t, "mode err", err, nil)
	testhelp.Equal(t, "token mode", fi.Mode().Perm(), os.FileMode(0o600))
	fi, err = os.Stat(h.reg.SecretPath("demo", "git-credentials"))
	testhelp.Equal(t, "mode err", err, nil)
	testhelp.Equal(t, "cred mode", fi.Mode().Perm(), os.FileMode(0o600))

	st, err := d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	rep := st.Repo
	testhelp.Equal(t, "status repo at", rep.CheckedAt.IsZero(), false)
	rep.CheckedAt = time.Time{}
	testhelp.Equal(t, "status repo", rep, github.Access{Checked: true, Reachable: true, Push: true})

	dir := filepath.Join(h.tmp, "projects", "demo")
	testhelp.Equal(t, "calls", h.runner.Calls, []runner.Call{
		{Dir: dir, Name: "git", Args: []string{"rev-parse", "--verify", "-q", "origin/main"}},
		{Dir: dir, Name: "git", Args: []string{"push", "-q", "-u", "origin", "main"}},
	})

	h.do = dcHTTP(401, `{"message":"Bad credentials"}`)
	res, err = d.PutGithubToken(context.Background(), "demo", "ghp_abc")
	testhelp.Equal(t, "second err", err, nil)
	testhelp.Equal(t, "second at", res.Access.CheckedAt.IsZero(), false)

// Example 11: every spawn fails; the loop restarts until it crashes, then
// Continue begins a fresh batch.
func TestDaemonCoreExample11(t *testing.T) {
	h := dcNew(t)
	h.spawnErr = errors.New("claude: start fork/exec /opt/claude/bin/claude: no such file or directory")
	h.addProject("demo", "go", "https://github.com/acme/demo")
	d := h.open()
	plan := dcPlan(t)

	qv, err := d.PlanLoad("demo", plan)
	testhelp.Equal(t, "planload err", err, nil)
	testhelp.Equal(t, "qv", qv, control.QueueView{State: "stopped", Approved: "7b31dfe", Index: 0, Phases: 3, Current: "P17", Reason: "crash"})

	launches := h.snapLaunches()
	testhelp.Equal(t, "launches", len(launches), 4)
	testhelp.Equal(t, "s0", launches[0].SessionID, "id-1")
	testhelp.Equal(t, "s1", launches[1].SessionID, "id-3")
	testhelp.Equal(t, "s2", launches[2].SessionID, "id-5")
	testhelp.Equal(t, "s3", launches[3].SessionID, "id-7")
	for i, l := range launches {
		testhelp.Equal(t, fmt.Sprintf("resume %d", i), l.Resume, false)
	}

	tok, err := h.reg.ReadSecret("demo", "session-token")
	testhelp.Equal(t, "token err", err, nil)
	testhelp.Equal(t, "token", tok, "id-8")

	testhelp.Equal(t, "posts", h.snapPosts(), [][4]string{
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 1 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 2 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 3 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "stop", "P17: session keeps exiting", "exited 4 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
	})

	st, err := d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "state", st.State, "waiting")

	testhelp.Equal(t, "LastExit", le, control.Exit{Code: -3, Stderr: "claude: start fork/exec /opt/claude/bin/claude: no such file or directory", Restarts: 3})

	stop, err := d.StopCheck("demo")

// Example 12: DoneWait 30s holds the end batch until the clock passes the
// deadline.
func TestDaemonCoreExample12(t *testing.T) {
	h := dcNew(t)
	h.cfg.DoneWait = 30 * time.Second
	h.addProject("demo", "go", "https://github.com/acme/demo")
	d := h.open()
	plan := dcPlan(t)

	_, err := d.PlanLoad("demo", plan)
	testhelp.Equal(t, "planload err", err, nil)
	fp := h.snapFakes()[0]

	err = d.PhaseDone("demo", "id-2", "P17", "P18")
	testhelp.Equal(t, "phase done err", err, nil)

	d.Tick(context.Background())
	testhelp.Equal(t, "not killed", fp.Killed(), false)
	testhelp.Equal(t, "spawns", len(h.snapLaunches()), 1)
	testhelp.Equal(t, "posts len", len(h.snapPosts()), 1)

	st, err := d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "phase inside window", st.Phase, "P17")
	testhelp.Equal(t, "state inside window", st.State, "busy")

	for _, c := range h.runner.Calls {
		if c.Name == "git" && dcEqArgs(c.Args, []string{"branch", "--list", "morph/*", "--no-merged", "main"}) {
			t.Errorf("unexpected branch --list call")
		}
	}

	h.clock.Add(60 * time.Second)
	d.Tick(context.Background())

	testhelp.Equal(t, "killed", fp.Killed(), true)
	launches := h.snapLaunches()
	testhelp.Equal(t, "spawns after window", len(launches), 2)
	testhelp.Equal(t, "second session", launches[1].SessionID, "id-3")
	testhelp.Equal(t, "second resume", launches[1].Resume, false)
	testhelp.Equal(t, "second budget", launches[1].BudgetUSD, float64(12))

	tok, err := h.reg.ReadSecret("demo", "session-token")
	testhelp.Equal(t, "token err", err, nil)
	testhelp.Equal(t, "token", tok, "id-4")

	posts := h.snapPosts()
	testhelp.Equal(t, "last post", posts[len(posts)-1], [4]string{"demo", "start", "P18 started", "session id-3 · cap $12 · 3 h"})

	st, err = d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "phase after window", st.Phase, "P18")
	testhelp.Equal(t, "index after window", st.Queue.Index, 1)
}

	testhelp.Equal(t, "created at zero", view.CreatedAt.IsZero(), false)

		{Dir: dir, Name: "git", Args: []string{"rev-parse", "--is-inside-work-tree"}},
		{Dir: dir, Name: "git", Args: []string{"rev-parse", "--abbrev-ref", "HEAD"}},
		{Dir: dir, Name: "git", Args: []string{"remote", "get-url", "origin"}},
