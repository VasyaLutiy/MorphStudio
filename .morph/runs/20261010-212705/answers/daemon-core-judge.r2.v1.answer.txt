var dcCreated = time.Date(2026, 10, 8, 14, 0, 0, 0, time.UTC)

// dcClock is the harness clock: Now returns base + n seconds and bumps n;
// Add moves the base forward under the same mutex.
type dcClock struct {
	mu   sync.Mutex
	base time.Time
	n    int
}

func (c *dcClock) Now() time.Time {
	c.mu.Lock()
	defer c.mu.Unlock()
	t := c.base.Add(time.Duration(c.n) * time.Second)
	c.n++
	return t
}

func (c *dcClock) Add(d time.Duration) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.base = c.base.Add(d)
}

// dcHarness is the one test harness of this file.
type dcHarness struct {
	t      *testing.T
	tmp    string
	reg    *registry.Registry
	runner *runner.Fake
	clock  *dcClock

	mu       sync.Mutex
	launches []claude.Launch
	fakes    []*claude.Fake
	posts    [][4]string
	nextID   int
	spawnErr error

	do  func(*http.Request) (*http.Response, error)
	cfg Config
}

func dcNew(t *testing.T) *dcHarness {
	t.Helper()
	tmp := t.TempDir()
	reg, err := registry.Open(filepath.Join(tmp, "state"))
	if err != nil {
		t.Fatal(err)
	}
	h := &dcHarness{
		t:      t,
		tmp:    tmp,
		reg:    reg,
		clock:  &dcClock{base: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC)},
		nextID: 1,
		do:     dcHTTP(200, `{"permissions":{"push":true}}`),
		cfg:    dcConfig(tmp),
	}
	h.runner = &runner.Fake{Script: map[string]runner.Result{
		"git rev-parse --abbrev-ref HEAD": {Stdout: "main\n"},
		"git rev-parse HEAD":              {Stdout: "aaa111\n"},
		"git rev-parse origin/main":       {Stdout: "aaa111\n"},
		"git status --porcelain":          {Stdout: ""},
	}}
	return h
}

func dcConfig(tmp string) Config {
	return Config{
		ClaudeBin:   "/opt/claude/bin/claude",
		MorphBin:    "/usr/local/bin/morph",
		ProjectsDir: filepath.Join(tmp, "projects"),
		MCPBaseURL:  "http://127.0.0.1:7181",
		Defaults:    queue.Caps{ClaudeUSD: 25, Hours: 2, ExecutorUSD: 4},
		MaxParallel: 1,
		Loop: supervisor.Config{
			MaxResumesPerHour: 3,
			StallMinutes:      30,
			StretchUSD:        30,
			UsageAlertPercent: 50,
		},
	}
}

func (h *dcHarness) addProject(t *testing.T, name string) {
	t.Helper()
	dir := filepath.Join(h.tmp, "projects", name)
	if err := os.MkdirAll(filepath.Join(dir, "docs"), 0o755); err != nil {
		t.Fatal(err)
	}
	data, err := os.ReadFile("../tests/fixtures/git/measure.md")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(dir, "docs", "MEASURE.md"), data, 0o644); err != nil {
		t.Fatal(err)
	}
	if err := h.reg.Add(registry.Project{
		Name:      name,
		Language:  "go",
		RepoURL:   "https://github.com/acme/" + name,
		Dir:       dir,
		Caps:      queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5},
		CreatedAt: dcCreated,
	}); err != nil {
		t.Fatal(err)
	}
}

func (h *dcHarness) spawn(ctx context.Context, l claude.Launch) (claude.Process, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.launches = append(h.launches, l)
	if h.spawnErr != nil {
		return nil, h.spawnErr
	}
	f := claude.NewFake()
	h.fakes = append(h.fakes, f)
	return f, nil
}

func (h *dcHarness) post(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
	return telegram.Status{}, nil
}

func (h *dcHarness) newID() string {
	h.mu.Lock()
	defer h.mu.Unlock()
	id := fmt.Sprintf("id-%d", h.nextID)
	h.nextID++
	return id
}

func (h *dcHarness) launchesCopy() []claude.Launch {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([]claude.Launch, len(h.launches))
	copy(out, h.launches)
	return out
}

func (h *dcHarness) postsCopy() [][4]string {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([][4]string, len(h.posts))
	copy(out, h.posts)
	return out
}

func (h *dcHarness) fakeAt(t *testing.T, i int) *claude.Fake {
	t.Helper()
	h.mu.Lock()
	defer h.mu.Unlock()
	if i >= len(h.fakes) {
		t.Fatalf("fake index %d out of %d", i, len(h.fakes))
	}
	return h.fakes[i]
}

func (h *dcHarness) open(t *testing.T) *Daemon {
	t.Helper()
	d, err := Open(context.Background(), Deps{
		Registry: h.reg,
		Runner:   h.runner,
		Spawn:    h.spawn,
		Post:     h.post,
		GitHubDo: func(r *http.Request) (*http.Response, error) { return h.do(r) },
		Now:      h.clock.Now,
		NewID:    h.newID,
		Config:   h.cfg,
	})
	if err != nil {
		t.Fatal(err)
	}
	return d
}

func dcHTTP(code int, body string) func(*http.Request) (*http.Response, error) {
	return func(*http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: code,
			Body:       io.NopCloser(strings.NewReader(body)),
			Header:     http.Header{},
		}, nil
	}
}

func dcPlan(t *testing.T) queue.Plan {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/plan/queue-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var p queue.Plan
	if err := json.Unmarshal(data, &p); err != nil {
		t.Fatal(err)
	}
	return p
}

func TestDaemonCoreExample8(t *testing.T) {
	h := dcNew(t)
	h.addProject(t, "demo")
	h.runner.Script["git rev-parse --verify -q origin/main"] = runner.Result{Code: 1}
	h.runner.Script["git push -q -u origin main"] = runner.Result{Code: 0}
	h.do = dcHTTP(200, `{"permissions":{"push":true}}`)
	d := h.open(t)

	res, err := d.PutGithubToken(context.Background(), "demo", "ghp_abc")
	testhelp.Equal(t, "err", err, nil)
	testhelp.Equal(t, "checked at", res.Access.CheckedAt.IsZero(), false)
	res.Access.CheckedAt = time.Time{}
	testhelp.Equal(t, "result", res, control.TokenResult{
		Access: github.Access{Checked: true, Reachable: true, Push: true},
		Pushed: true,
	})

	tok, err := h.reg.ReadSecret("demo", "github-token")
	testhelp.Equal(t, "tok err", err, nil)
	testhelp.Equal(t, "tok", tok, "ghp_abc")
	cred, err := h.reg.ReadSecret("demo", "git-credentials")
	testhelp.Equal(t, "cred err", err, nil)
	testhelp.Equal(t, "cred", cred, "https://x-access-token:ghp_abc@github.com\n")

	info, err := os.Stat(h.reg.SecretPath("demo", "github-token"))
	testhelp.Equal(t, "tok stat err", err, nil)
	testhelp.Equal(t, "tok mode", info.Mode().Perm(), os.FileMode(0o600))
	info, err = os.Stat(h.reg.SecretPath("demo", "git-credentials"))
	testhelp.Equal(t, "cred stat err", err, nil)
	testhelp.Equal(t, "cred mode", info.Mode().Perm(), os.FileMode(0o600))

	st, err := d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	gotAccess := st.Repo
	testhelp.Equal(t, "repo checked", gotAccess.CheckedAt.IsZero(), false)
	gotAccess.CheckedAt = time.Time{}
	testhelp.Equal(t, "repo", gotAccess, github.Access{Checked: true, Reachable: true, Push: true})

	testhelp.Equal(t, "calls", h.runner.Calls, []runner.Call{
		{Dir: filepath.Join(h.tmp, "projects", "demo"), Name: "git", Args: []string{"rev-parse", "--verify", "-q", "origin/main"}},
		{Dir: filepath.Join(h.tmp, "projects", "demo"), Name: "git", Args: []string{"push", "-q", "-u", "origin", "main"}},
	})

	h.do = dcHTTP(401, `{"message":"Bad credentials"}`)
	res2, err := d.PutGithubToken(context.Background(), "demo", "ghp_abc")
	testhelp.Equal(t, "err2", err, nil)
	testhelp.Equal(t, "checked at2", res2.Access.CheckedAt.IsZero(), false)
	res2.Access.CheckedAt = time.Time{}
	testhelp.Equal(t, "result2", res2, control.TokenResult{
		Access: github.Access{Checked: true, Reason: "401 bad credentials"},
	})
	testhelp.Equal(t, "calls still 2", len(h.runner.Calls), 2)

	_, err = d.PutGithubToken(context.Background(), "demo", "")
	testhelp.Equal(t, "empty", errors.Is(err, control.ErrBadInput), true)

	_, err = d.PutGithubToken(context.Background(), "nope", "x")
	testhelp.Equal(t, "unknown", errors.Is(err, control.ErrUnknownProject), true)
}

	testhelp.Equal(t, "last exit", le, control.Exit{

	})

	testhelp.Equal(t, "stop err", err, nil)
	stop.At = time.Time{}
	testhelp.Equal(t, "stop", *stop, control.Stop{
		Kind:   "crash",
		Phase:  "P17",
		Reason: "exited 4 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory",
	})

	testhelp.Equal(t, "events err", err, nil)
	testhelp.Equal(t, "events", ev, control.Events{Entries: []eventlog.Entry{}, Last: 0})

		t.Fatal(err)

	testhelp.Equal(t, "tok", tok, "id-8")

		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 1 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 2 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 3 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
