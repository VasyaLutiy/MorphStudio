type dcClock struct {
	mu   sync.Mutex
	base time.Time
	n    int
}

func (c *dcClock) Now() time.Time {
	c.mu.Lock()
	defer c.mu.Unlock()
	t := c.base.Add(time.Duration(c.n) * time.Second)
	c.n++
	return t
}

func (c *dcClock) Add(d time.Duration) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.base = c.base.Add(d)
}

type dcHarness struct {
	t      *testing.T
	tmp    string
	reg    *registry.Registry
	runner *runner.Fake
	clock  *dcClock
	deps   Deps
	d      *Daemon

	mu       sync.Mutex
	launches []claude.Launch
	procs    []*claude.Fake
	posts    [][4]string
	id       int
	spawnErr error
	doFunc   func(*http.Request) (*http.Response, error)
}

func (h *dcHarness) nextID() string {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.id++
	return fmt.Sprintf("id-%d", h.id)
}

func (h *dcHarness) spawn(ctx context.Context, l claude.Launch) (claude.Process, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.launches = append(h.launches, l)
	if h.spawnErr != nil {
		return nil, h.spawnErr
	}
	p := claude.NewFake()
	h.procs = append(h.procs, p)
	return p, nil
}

func (h *dcHarness) post(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
	h.mu.Lock()
	h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
	h.mu.Unlock()
	return telegram.Status{}, nil
}

func (h *dcHarness) do(req *http.Request) (*http.Response, error) {
	h.mu.Lock()
	fn := h.doFunc
	h.mu.Unlock()
	if fn != nil {
		return fn(req)
	}
	return &http.Response{
		StatusCode: http.StatusOK,
		Body:       io.NopCloser(strings.NewReader("{}")),
		Header:     http.Header{},
	}, nil
}

func (h *dcHarness) setDo(fn func(*http.Request) (*http.Response, error)) {
	h.mu.Lock()
	h.doFunc = fn
	h.mu.Unlock()
}

func (h *dcHarness) setSpawnErr(err error) {
	h.mu.Lock()
	h.spawnErr = err
	h.mu.Unlock()
}

func (h *dcHarness) snapLaunches() []claude.Launch {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([]claude.Launch, len(h.launches))
	copy(out, h.launches)
	return out
}

func (h *dcHarness) snapPosts() [][4]string {
	h.mu.Lock()
	defer h.mu.Unlock()
	out := make([][4]string, len(h.posts))
	copy(out, h.posts)
	return out
}

func (h *dcHarness) proc(i int) *claude.Fake {
	h.mu.Lock()
	defer h.mu.Unlock()
	if i >= len(h.procs) {
		h.t.Fatalf("no proc %d", i)
	}
	return h.procs[i]
}

func dcWriteMeasure(t *testing.T, dir string) {
	t.Helper()
	data, err := os.ReadFile("../tests/fixtures/git/measure.md")
	if err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(dir, "docs", "MEASURE.md")
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, data, 0o644); err != nil {
		t.Fatal(err)
	}
}

func dcNew(t *testing.T, names []string, scripts map[string]runner.Result, cfg Config) *dcHarness {
	t.Helper()
	tmp := t.TempDir()
	reg, err := registry.Open(filepath.Join(tmp, "state"))
	if err != nil {
		t.Fatal(err)
	}
	for _, name := range names {
		dir := filepath.Join(tmp, "projects", name)
		dcWriteMeasure(t, dir)
		p := registry.Project{
			Name:      name,
			Language:  "go",
			RepoURL:   "https://github.com/acme/" + name,
			Dir:       dir,
			Caps:      queue.Caps{ClaudeUSD: 30, Hours: 3, ExecutorUSD: 5},
			CreatedAt: time.Date(2026, 10, 8, 14, 0, 0, 0, time.UTC),
		}
		if err := reg.Add(p); err != nil {
			t.Fatal(err)
		}
	}
	if scripts == nil {
		scripts = dcBaseScript()
	}
	h := &dcHarness{
		t:      t,
		tmp:    tmp,
		reg:    reg,
		runner: &runner.Fake{Script: scripts},
		clock:  &dcClock{base: time.Date(2026, 10, 8, 15, 0, 0, 0, time.UTC)},
	}
	cfg.ProjectsDir = filepath.Join(tmp, "projects")
	h.deps = Deps{
		Registry: reg,
		Runner:   h.runner,
		Spawn:    h.spawn,
		Post:     h.post,
		GitHubDo: h.do,
		Now:      h.clock.Now,
		NewID:    h.nextID,
		Config:   cfg,
	}
	d, err := Open(context.Background(), h.deps)
	if err != nil {
		t.Fatal(err)
	}
	h.d = d
	return h
}

func dcZeroAccess(a github.Access) github.Access {
	a.CheckedAt = time.Time{}
	return a
}

func dcZeroStop(s *control.Stop) *control.Stop {
	if s == nil {
		return nil
	}
	cp := *s
	cp.At = time.Time{}
	return &cp
}

func dcZeroExit(e *control.Exit) *control.Exit {
	if e == nil {
		return nil
	}
	cp := *e
	cp.At = time.Time{}
	return &cp
}

func dcCleanStatus(s control.Status) control.Status {
	s.Stop = dcZeroStop(s.Stop)
	s.LastExit = dcZeroExit(s.LastExit)
	return s
}

func TestDaemonCoreExample8(t *testing.T) {
	scripts := dcBaseScript()
	scripts["git rev-parse --verify -q origin/main"] = runner.Result{Code: 1}
	scripts["git push -q -u origin main"] = runner.Result{Code: 0}
	h := dcNew(t, []string{"demo"}, scripts, dcConfig())

	h.setDo(func(req *http.Request) (*http.Response, error) {
		return &http.Response{
			StatusCode: http.StatusOK,
			Body:       io.NopCloser(strings.NewReader(`{"permissions":{"push":true}}`)),
			Header:     http.Header{},
		}, nil
	})

	res, err := h.d.PutGithubToken(context.Background(), "demo", "ghp_abc")
	testhelp.Equal(t, "put err", err, nil)
	access := res.Access
	if access.CheckedAt.IsZero() {
		t.Errorf("access CheckedAt is zero")
	}
	access.CheckedAt = time.Time{}
	testhelp.Equal(t, "access", access, github.Access{Checked: true, Reachable: true, Push: true})
	testhelp.Equal(t, "pushed", res.Pushed, true)
	testhelp.Equal(t, "push error", res.PushError, "")

	testhelp.Equal(t, "token err", err, nil)
	testhelp.Equal(t, "token", token, "ghp_abc")
	cred, err := h.reg.ReadSecret("demo", "git-credentials")
	testhelp.Equal(t, "cred err", err, nil)
	testhelp.Equal(t, "cred", cred, github.CredentialLine("ghp_abc"))

	testhelp.Equal(t, "token stat err", err, nil)

		return &http.Response{
			StatusCode: http.StatusUnauthorized,

	access = res.Access
	if access.CheckedAt.IsZero() {
		t.Errorf("access2 CheckedAt is zero")

func TestDaemonCoreExample11(t *testing.T) {
	h := dcNew(t, []string{"demo"}, nil, dcConfig())
	h.setSpawnErr(errors.New("claude: start fork/exec /opt/claude/bin/claude: no such file or directory"))
	plan := dcPlan(t)

	qv, err := h.d.PlanLoad("demo", plan)
	testhelp.Equal(t, "planload err", err, nil)
	testhelp.Equal(t, "queue", qv, control.QueueView{State: "stopped", Approved: "7b31dfe", Index: 0, Phases: 3, Current: "P17", Reason: "crash"})

	launches := h.snapLaunches()
	testhelp.Equal(t, "launches", len(launches), 4)
	testhelp.Equal(t, "ids", []string{launches[0].SessionID, launches[1].SessionID, launches[2].SessionID, launches[3].SessionID}, []string{"id-1", "id-3", "id-5", "id-7"})
	for i, l := range launches {
		testhelp.Equal(t, fmt.Sprintf("resume %d", i), l.Resume, false)
	}

	secret, err := h.reg.ReadSecret("demo", "session-token")
	testhelp.Equal(t, "secret err", err, nil)
	testhelp.Equal(t, "secret", secret, "id-8")

	posts := h.snapPosts()
	expectedPosts := [][4]string{
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 1 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 2 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "watchdog", "P17: session exited (code -3), restarting", "restart 3 of 3 this hour · claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
		{"demo", "stop", "P17: session keeps exiting", "exited 4 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory"},
	}
	testhelp.Equal(t, "posts", posts, expectedPosts)

	st, err := h.d.Status("demo")
	testhelp.Equal(t, "status err", err, nil)
	testhelp.Equal(t, "state", st.State, "waiting")
	testhelp.Equal(t, "phase", st.Phase, "P17")
	testhelp.Equal(t, "session", st.SessionID, "id-7")
	testhelp.Equal(t, "limits unknown", st.LimitsUnknown, true)

	if st.Stop == nil {
		t.Fatalf("nil stop")
	}
	if st.Stop.At.IsZero() {
		t.Errorf("stop At is zero")
	}
	stop := *st.Stop
	stop.At = time.Time{}
	testhelp.Equal(t, "stop", stop, control.Stop{Kind: "crash", Phase: "P17", Reason: "exited 4 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory"})

	if st.LastExit == nil {
		t.Fatalf("nil last exit")
	}
	if st.LastExit.At.IsZero() {
		t.Errorf("last exit At is zero")
	}
	ex := *st.LastExit
	ex.At = time.Time{}
	testhelp.Equal(t, "last exit", ex, control.Exit{Code: -3, Stderr: "claude: start fork/exec /opt/claude/bin/claude: no such file or directory", Restarts: 3})

	sp, err := h.d.StopCheck("demo")
	testhelp.Equal(t, "stopcheck err", err, nil)
	if sp == nil {
		t.Fatalf("nil stopcheck")
	}
	cp := *sp
	cp.At = time.Time{}
	testhelp.Equal(t, "stopcheck", cp, control.Stop{Kind: "crash", Phase: "P17", Reason: "exited 4 times within an hour (last code -3): claude: start fork/exec /opt/claude/bin/claude: no such file or directory"})

	ev, err := h.d.Events("demo", 0, 10)
	testhelp.Equal(t, "events err", err, nil)
	testhelp.Equal(t, "events", ev, control.Events{Entries: []eventlog.Entry{}, Last: 0})

	data, err := os.ReadFile(filepath.Join(h.tmp, "state", "demo", "state.json"))
	testhelp.Equal(t, "state read err", err, nil)
	var state State
	if err := json.Unmarshal(data, &state); err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "state loop", state.Loop.State, "waiting")
	testhelp.Equal(t, "state restarts", len(state.Loop.Restarts), 3)

	err = h.d.Continue("demo")
	testhelp.Equal(t, "continue", err, nil)

	launches = h.snapLaunches()
	testhelp.Equal(t, "launches2", len(launches), 8)
	testhelp.Equal(t, "ids2", []string{launches[4].SessionID, launches[5].SessionID, launches[6].SessionID, launches[7].SessionID}, []string{"id-9", "id-11", "id-13", "id-15"})

	st, err = h.d.Status("demo")
	testhelp.Equal(t, "status err2", err, nil)
	testhelp.Equal(t, "state2", st.State, "waiting")
	if st.Stop != nil {
		testhelp.Equal(t, "stop kind2", st.Stop.Kind, "crash")
	}
	testhelp.Equal(t, "posts count", len(h.snapPosts()), 8)
}

func TestDaemonCoreExample12(t *testing.T) {

	_, err := h.d.PlanLoad("demo", plan)
	if err != nil {
		t.Fatal(err)
	}

	testhelp.Equal(t, "phase done", err, nil)

	testhelp.Equal(t, "phase", st.Phase, "P17")
	testhelp.Equal(t, "state", st.State, "busy")

	for _, call := range h.runner.Calls {
		if len(call.Args) == 5 && call.Args[0] == "branch" && call.Args[1] == "--list" && call.Args[2] == "morph/*" && call.Args[3] == "--no-merged" && call.Args[4] == "main" {
			t.Errorf("branch call before DoneWait: %v", call.Args)
		}
	}

	h.d.Tick(context.Background())
	testhelp.Equal(t, "killed false2", fp.Killed(), false)
	testhelp.Equal(t, "spawn still one", len(h.snapLaunches()), 1)

	h.clock.Add(60 * time.Second)
	h.d.Tick(context.Background())

	testhelp.Equal(t, "killed true", fp.Killed(), true)

	testhelp.Equal(t, "spawn two", len(launches), 2)
	testhelp.Equal(t, "session", launches[1].SessionID, "id-3")

	testhelp.Equal(t, "secret", secret, "id-4")

	posts := h.snapPosts()
	testhelp.Equal(t, "last post", posts[len(posts)-1], [4]string{"demo", "start", "P18 started", "session id-3 · cap $12 · 3 h"})

	st, err = h.d.Status("demo")
	testhelp.Equal(t, "status err2", err, nil)
	testhelp.Equal(t, "phase2", st.Phase, "P18")

}
