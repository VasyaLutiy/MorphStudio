// mmStatus is the status the fake returns in every example.

// mmHandler builds the mount over f with the example's tokens and known set.
func mmHandler(f *mmFake) http.Handler {
	return Handler(f, func(p string) bool { return p == "demo" }, Tokens{
		API: "tok-1",
		Session: func(project string) (string, bool) {
			if project == "demo" {
				return "id-2", true
			}
			return "", false
		},
	})
}

// mmServe serves one request through h and returns the recorder.
func mmServe(h http.Handler, method, target, auth, body string) *httptest.ResponseRecorder {
	req := httptest.NewRequest(method, target, strings.NewReader(body))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Accept", "application/json, text/event-stream")
	if auth != "" {
		req.Header.Set("Authorization", auth)
	}
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	return rec
}

// mmJSONRPC is the decoded shape of a JSON-RPC response.
type mmJSONRPC struct {
	ID     int `json:"id"`
	Result struct {
		Tools []struct {
			Name string `json:"name"`
		} `json:"tools"`
		Content []struct {
			Type string `json:"type"`
			Text string `json:"text"`
		} `json:"content"`
		IsError bool `json:"isError"`
	} `json:"result"`
}

// mmDecode decodes a JSON-RPC response body.
func mmDecode(t *testing.T, rec *httptest.ResponseRecorder) mmJSONRPC {
	t.Helper()
	var r mmJSONRPC
	if err := json.Unmarshal(rec.Body.Bytes(), &r); err != nil {
		t.Fatalf("decode: %v: %s", err, rec.Body.String())
	}
	return r
}

// mmToolNames returns the sorted tool names of a decoded tools/list response.
func mmToolNames(got mmJSONRPC) []string {
	names := make([]string, 0, len(got.Result.Tools))
	for _, tool := range got.Result.Tools {
		names = append(names, tool.Name)
	}
	sort.Strings(names)
	return names
}

func TestMCPMountExample1(t *testing.T) {
	f := &mmFake{status: mmStatus()}
	h := mmHandler(f)

	rec := mmServe(h, "POST", "/mcp/demo", "Bearer tok-1", mmListBody)
	testhelp.Equal(t, "status", rec.Code, 200)
	if ct := rec.Header().Get("Content-Type"); !strings.HasPrefix(ct, "application/json") {
		t.Errorf("content type: got %q", ct)
	}

	got := mmDecode(t, rec)
	testhelp.Equal(t, "id", got.ID, 1)
	testhelp.Equal(t, "tool names", mmToolNames(got), []string{"answer", "continue", "events", "interrupt", "order", "pending", "plan_load", "restart", "status", "stop_check", "usage"})
}

func TestMCPMountExample2(t *testing.T) {
	f := &mmFake{status: mmStatus()}

	rec := mmServe(h, "POST", "/mcp/demo", "Bearer tok-1", mmStatusBody)

	testhelp.Equal(t, "isError", got.Result.IsError, false)
	testhelp.Equal(t, "calls", f.calls, []mmCall{{Method: "Status", Args: []any{"demo"}}})
}

func TestMCPMountExample4(t *testing.T) {
	f := &mmFake{status: mmStatus()}
	h := mmHandler(f)

	cases := []struct {
		name   string
		target string
		auth   string
	}{
		{"no token", "/mcp/demo", ""},
		{"wrong token", "/mcp/demo", "Bearer wrong"},
		{"api token for session", "/mcp/demo/session", "Bearer tok-1"},
		{"other project session", "/mcp/other/session", "Bearer id-2"},
	}
	for _, c := range cases {
		rec := mmServe(h, "POST", c.target, c.auth, mmListBody)
		testhelp.Equal(t, c.name+" status", rec.Code, 401)
		testhelp.Equal(t, c.name+" content type", rec.Header().Get("Content-Type"), "application/json")
		testhelp.Equal(t, c.name+" body", rec.Body.String(), mmUnauthorized)
	}
	testhelp.Equal(t, "calls", f.calls, []mmCall(nil))
}

}
