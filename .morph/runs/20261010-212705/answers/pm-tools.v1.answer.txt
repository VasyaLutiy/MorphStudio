```go
package mcpserver

import (
	"context"
	"encoding/json"

	"github.com/modelcontextprotocol/go-sdk/mcp"

	"morphstudio/control"
	"morphstudio/queue"
)

// ServerName is the MCP implementation name shared by both servers.
const ServerName = "morphd"

// ServerVersion is the MCP implementation version shared by both servers.
const ServerVersion = "0.1.0"

// emptyInput is the input of a tool that takes no arguments.
type emptyInput struct{}

// jsonResult renders v as the tool's one text content.
func jsonResult(v any) (*mcp.CallToolResult, any, error) {
	data, err := json.Marshal(v)
	if err != nil {
		return errorResult(err)
	}
	return &mcp.CallToolResult{Content: []mcp.Content{&mcp.TextContent{Text: string(data)}}}, nil, nil
}

// errorResult renders err as an isError result carrying its stable code.
func errorResult(err error) (*mcp.CallToolResult, any, error) {
	_, code := control.Code(err)
	return &mcp.CallToolResult{
		Content: []mcp.Content{&mcp.TextContent{Text: code + ": " + err.Error()}},
		IsError: true,
	}, nil, nil
}

// okView is the {"ok":true} view returned by mutating tools.
type okView struct {
	OK bool `json:"ok"`
}

// createInput is the input of project_create.
type createInput struct {
	Name     string `json:"name"`
	Language string `json:"language"`
	RepoURL  string `json:"repo_url"`
}

// adoptInput is the input of project_adopt.
type adoptInput struct {
	Name     string `json:"name"`
	Dir      string `json:"dir"`
	RepoURL  string `json:"repo_url"`
	Language string `json:"language"`
}

// UserServer builds the user-level MCP server over c.
func UserServer(c control.Control) *mcp.Server {
	s := mcp.NewServer(&mcp.Implementation{Name: ServerName, Version: ServerVersion}, nil)

	mcp.AddTool(s, &mcp.Tool{Name: "projects_list", Description: "List all projects."},
		func(ctx context.Context, req *mcp.CallToolRequest, in emptyInput) (*mcp.CallToolResult, any, error) {
			p := c.Projects()
			if p == nil {
				p = []control.ProjectView{}
			}
			return jsonResult(map[string]any{"projects": p})
		})

	mcp.AddTool(s, &mcp.Tool{Name: "project_create", Description: "Create a project."},
		func(ctx context.Context, req *mcp.CallToolRequest, in createInput) (*mcp.CallToolResult, any, error) {
			v, err := c.CreateProject(ctx, in.Name, in.Language, in.RepoURL)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(v)
		})

	mcp.AddTool(s, &mcp.Tool{Name: "project_adopt", Description: "Register a repository that already exists as a project."},
		func(ctx context.Context, req *mcp.CallToolRequest, in adoptInput) (*mcp.CallToolResult, any, error) {
			v, err := control.Adopt(ctx, c, in.Name, in.Dir, in.RepoURL, in.Language)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(v)
		})

	return s
}

// orderInput is the input of order.
type orderInput struct {
	Text string `json:"text"`
}

// answerInput is the input of answer.
type answerInput struct {
	Option int `json:"option"`
}

// restartInput is the input of restart.
type restartInput struct {
	Force bool `json:"force,omitempty"`
}

// eventsInput is the input of events.
type eventsInput struct {
	Since int64 `json:"since,omitempty"`
	Limit int   `json:"limit,omitempty"`
}

// capsInput is the input of a plan phase's caps.
type capsInput struct {
	ClaudeUSD   float64 `json:"claude_usd,omitempty"`
	Hours       float64 `json:"hours,omitempty"`
	ExecutorUSD float64 `json:"executor_usd,omitempty"`
}

// phaseInput is the input of one plan phase.
type phaseInput struct {
	ID        string     `json:"id"`
	StopAfter string     `json:"stop_after,omitempty"`
	Caps      *capsInput `json:"caps,omitempty"`
}

// planInput is the input of plan_load.
type planInput struct {
	Approved string       `json:"approved"`
	Phases   []phaseInput `json:"phases"`
}

// questionView is the view of pending: the question or null.
type questionView struct {
	Question *control.Question `json:"question"`
}

// stopView is the view of stop_check: the stop or null.
type stopView struct {
	Stop *control.Stop `json:"stop"`
}

// PMServer builds the per-project MCP server over c for project.
func PMServer(c control.Control, project string) *mcp.Server {
	s := mcp.NewServer(&mcp.Implementation{Name: ServerName, Version: ServerVersion}, nil)

	mcp.AddTool(s, &mcp.Tool{Name: "status", Description: "Report the project's status."},
		func(ctx context.Context, req *mcp.CallToolRequest, in emptyInput) (*mcp.CallToolResult, any, error) {
			v, err := c.Status(project)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(v)
		})

	mcp.AddTool(s, &mcp.Tool{Name: "order", Description: "Give the running session an order."},
		func(ctx context.Context, req *mcp.CallToolRequest, in orderInput) (*mcp.CallToolResult, any, error) {
			v, err := c.Order(project, in.Text)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(v)
		})

	mcp.AddTool(s, &mcp.Tool{Name: "pending", Description: "Return the project's pending question, if any."},
		func(ctx context.Context, req *mcp.CallToolRequest, in emptyInput) (*mcp.CallToolResult, any, error) {
			q, err := c.Pending(project)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(questionView{Question: q})
		})

	mcp.AddTool(s, &mcp.Tool{Name: "answer", Description: "Answer the pending question with the given option."},
		func(ctx context.Context, req *mcp.CallToolRequest, in answerInput) (*mcp.CallToolResult, any, error) {
			if err := c.Answer(project, in.Option); err != nil {
				return errorResult(err)
			}
			return jsonResult(okView{OK: true})
		})

	mcp.AddTool(s, &mcp.Tool{Name: "interrupt", Description: "Interrupt the running session."},
		func(ctx context.Context, req *mcp.CallToolRequest, in emptyInput) (*mcp.CallToolResult, any, error) {
			if err := c.Interrupt(project); err != nil {
				return errorResult(err)
			}
			return jsonResult(okView{OK: true})
		})

	mcp.AddTool(s, &mcp.Tool{Name: "usage", Description: "Report rate-limit utilization and session spend."},
		func(ctx context.Context, req *mcp.CallToolRequest, in emptyInput) (*mcp.CallToolResult, any, error) {
			v, err := c.Usage(project)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(v)
		})

	mcp.AddTool(s, &mcp.Tool{Name: "restart", Description: "Restart the project's session."},
		func(ctx context.Context, req *mcp.CallToolRequest, in restartInput) (*mcp.CallToolResult, any, error) {
			if err := c.Restart(project, in.Force); err != nil {
				return errorResult(err)
			}
			return jsonResult(okView{OK: true})
		})

	mcp.AddTool(s, &mcp.Tool{Name: "plan_load", Description: "Load an approved plan into the project's queue."},
		func(ctx context.Context, req *mcp.CallToolRequest, in planInput) (*mcp.CallToolResult, any, error) {
			p := queue.Plan{Approved: in.Approved}
			for _, ph := range in.Phases {
				phase := queue.Phase{ID: ph.ID, StopAfter: ph.StopAfter}
				if ph.Caps != nil {
					phase.Caps = queue.Caps{
						ClaudeUSD:   ph.Caps.ClaudeUSD,
						Hours:       ph.Caps.Hours,
						ExecutorUSD: ph.Caps.ExecutorUSD,
					}
				}
				p.Phases = append(p.Phases, phase)
			}
			v, err := c.PlanLoad(project, p)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(v)
		})

	mcp.AddTool(s, &mcp.Tool{Name: "continue", Description: "Continue the queue after a stop."},
		func(ctx context.Context, req *mcp.CallToolRequest, in emptyInput) (*mcp.CallToolResult, any, error) {
			if err := c.Continue(project); err != nil {
				return errorResult(err)
			}
			return jsonResult(okView{OK: true})
		})

	mcp.AddTool(s, &mcp.Tool{Name: "stop_check", Description: "Return the reason the queue stopped, if any."},
		func(ctx context.Context, req *mcp.CallToolRequest, in emptyInput) (*mcp.CallToolResult, any, error) {
			st, err := c.StopCheck(project)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(stopView{Stop: st})
		})

	mcp.AddTool(s, &mcp.Tool{Name: "events", Description: "Return the project's recent event log entries."},
		func(ctx context.Context, req *mcp.CallToolRequest, in eventsInput) (*mcp.CallToolResult, any, error) {
			limit := in.Limit
			if limit == 0 {
				limit = 50
			}
			ev, err := c.Events(project, in.Since, limit)
			if err != nil {
				return errorResult(err)
			}
			return jsonResult(control.Views(ev, 200))
		})

	return s
}
```
