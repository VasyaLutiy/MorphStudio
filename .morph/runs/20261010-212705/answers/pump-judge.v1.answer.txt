func puOpen(t *testing.T, h *puHarness) {
	t.Helper()
	d, err := Open(context.Background(), h.deps)
	if err != nil {
		t.Fatal(err)
	}
	h.d = d
}

func puNew(t *testing.T) *puHarness {
	t.Helper()
	h := puBare(t)
	puOpen(t, h)
	return h
}

func (h *puHarness) spawn(ctx context.Context, l claude.Launch) (claude.Process, error) {
	h.mu.Lock()
	h.launches = append(h.launches, l)
	f := claude.NewFake()
	h.procs = append(h.procs, f)
	on := h.onSpawn
	h.mu.Unlock()
	if on != nil {
		return on(ctx, l)
	}
	return f, nil
}

func (h *puHarness) post(ctx context.Context, project, kind, headline, numbers string) (telegram.Status, error) {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.posts = append(h.posts, [4]string{project, kind, headline, numbers})
	return telegram.Status{}, nil
}

func (h *puHarness) clock() time.Time {
	h.mu.Lock()
	defer h.mu.Unlock()
	now := h.now
	h.now = h.now.Add(time.Second)
	return now
}

func (h *puHarness) newID() string {
	h.mu.Lock()
	defer h.mu.Unlock()
	h.nextID++
	return fmt.Sprintf("id-%d", h.nextID)
}

func (h *puHarness) procAt(i int) *claude.Fake {
	h.mu.Lock()
	defer h.mu.Unlock()

// puPoll retries get every 10 ms for within and fails with the last value.
func puPoll[T any](t *testing.T, what string, within time.Duration, get func() T, done func(T) bool) T {
	t.Helper()
	deadline := time.Now().Add(within)
	var last T
	for {
		last = get()
		if done(last) {
			return last
		}
		if time.Now().After(deadline) {
			t.Fatalf("pu: %s: timed out, last %#v", what, last)
		}
		time.Sleep(10 * time.Millisecond)
	}
}

// puPong emits the four probe lines of the pong turn and waits for the cost.
func puPong(t *testing.T, h *puHarness) {
	t.Helper()
	fp := h.procAt(0)
	for _, n := range []int{4, 5, 6, 7} {
		_, msg := testhelp.ProbeLine(t, puProbe, n)
		fp.Emit(msg)
	}
	puPoll(t, "cost", 2*time.Second, func() float64 {
		s, err := h.d.Status("demo")
		if err != nil {
			t.Fatal(err)
		}
		return s.CostUSD
	}, func(v float64) bool { return v > 0 })
}

func puLog(t *testing.T, h *puHarness, id string) []eventlog.Entry {
	t.Helper()
	b, err := os.ReadFile(filepath.Join(h.tmp, "state", "demo", "sessions", id+".jsonl"))
	if err != nil {
		t.Fatal(err)
	}
	text := strings.TrimSuffix(string(b), "\n")
	if text == "" {
		return nil
	}
	lines := strings.Split(text, "\n")
	out := make([]eventlog.Entry, 0, len(lines))
	for _, line := range lines {
		var e eventlog.Entry
		if err := json.Unmarshal([]byte(line), &e); err != nil {
			t.Fatalf("pu: log %s: %v", id, err)
		}
		out = append(out, e)
	}
	return out
}

func puCompact(t *testing.T, raw []byte) string {
	t.Helper()
	var buf bytes.Buffer
	if err := json.Compact(&buf, raw); err != nil {
		t.Fatal(err)
	}
	return buf.String()
}

func puSeqs(es []eventlog.Entry) []int64 {
	out := make([]int64, len(es))
	for i, e := range es {
		out[i] = e.Seq
	}
	return out
}

func puDirs(es []eventlog.Entry) []string {
	out := make([]string, len(es))
	for i, e := range es {
		out[i] = e.Dir
	}
	return out
}

func puMsgs(es []eventlog.Entry) []string {
	out := make([]string, len(es))
	for i, e := range es {
		out[i] = string(e.Msg)
	}
	return out
}

		t.Fatal(err)
	}
