package users

import (
	"encoding/json"
	"os"
	"path/filepath"
	"testing"
	"time"

	"morphstudio/internal/testhelp"
)

func tlBase(t *testing.T) string {
	t.Helper()
	base := filepath.Join(t.TempDir(), "ustate")
	b, err := os.ReadFile("../tests/fixtures/users/users-3.json")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.MkdirAll(base, 0o700); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(base, "users.json"), b, 0o600); err != nil {
		t.Fatal(err)
	}
	return base
}

func tlOpen(t *testing.T, admin string) *Users {
	t.Helper()
	u, err := Open(tlBase(t), admin)
	if err != nil {
		t.Fatal(err)
	}
	return u
}

func tlFixtureUsers(t *testing.T) map[string]User {
	t.Helper()
	b, err := os.ReadFile("../tests/fixtures/users/users-3.json")
	if err != nil {
		t.Fatal(err)
	}
	var list []User
	if err := json.Unmarshal(b, &list); err != nil {
		t.Fatal(err)
	}
	m := make(map[string]User, len(list))
	for _, v := range list {
		m[v.Name] = v
	}
	return m
}

func TestTokenLookupExample1(t *testing.T) {
	u := tlOpen(t, "admin-op-9")
	want := tlFixtureUsers(t)

	// The fixture holds the three users with these exact token hashes.
	testhelp.Equal(t, "fixture alice TokenHash", want["alice"].TokenHash,
		"61fdf299956e0522e0a49b4ae572f446b7f811dd73234bc6ddc67aac81d9dcf2")
	testhelp.Equal(t, "fixture bob TokenHash", want["bob"].TokenHash,
		"a0a996e6da7d3784ee7348bee3e43d60f64b72a3ea18196d9f0fbc99a40a8dbd")
	testhelp.Equal(t, "fixture carol TokenHash", want["carol"].TokenHash,
		"b9a24afb8ff6092a52e6232b5b6b76b476503d5322dcdcdf73e75989201448f7")
	testhelp.Equal(t, "fixture alice CreatedAt", want["alice"].CreatedAt,
		time.Date(2026, 10, 12, 10, 1, 0, 0, time.UTC))

	got, ok := u.LookupToken("tok-bob-2")
	testhelp.Equal(t, "LookupToken(tok-bob-2) ok", ok, true)
	exp := want["bob"]
	testhelp.Equal(t, "bob Name", got.Name, exp.Name)
	testhelp.Equal(t, "bob LinuxUser", got.LinuxUser, exp.LinuxUser)
	testhelp.Equal(t, "bob ChatID", got.ChatID, exp.ChatID)
	testhelp.Equal(t, "bob TokenHash", got.TokenHash, exp.TokenHash)
	testhelp.Equal(t, "bob CreatedAt", got.CreatedAt, exp.CreatedAt)

	got, ok = u.LookupToken("tok-alice-1")
	testhelp.Equal(t, "LookupToken(tok-alice-1) ok", ok, true)
	exp = want["alice"]
	testhelp.Equal(t, "alice Name", got.Name, exp.Name)
	testhelp.Equal(t, "alice LinuxUser", got.LinuxUser, exp.LinuxUser)
	testhelp.Equal(t, "alice ChatID", got.ChatID, exp.ChatID)
	testhelp.Equal(t, "alice TokenHash", got.TokenHash, exp.TokenHash)
	testhelp.Equal(t, "alice CreatedAt", got.CreatedAt, exp.CreatedAt)

	got, ok = u.LookupToken("tok-carol-3")
	testhelp.Equal(t, "LookupToken(tok-carol-3) ok", ok, true)
	exp = want["carol"]
	testhelp.Equal(t, "carol Name", got.Name, exp.Name)
	testhelp.Equal(t, "carol LinuxUser", got.LinuxUser, exp.LinuxUser)
	testhelp.Equal(t, "carol ChatID", got.ChatID, exp.ChatID)
	testhelp.Equal(t, "carol TokenHash", got.TokenHash, exp.TokenHash)
	testhelp.Equal(t, "carol CreatedAt", got.CreatedAt, exp.CreatedAt)
}

func TestTokenLookupExample2(t *testing.T) {
	u := tlOpen(t, "admin-op-9")
	before := u.List()

	got, ok := u.LookupToken("wrong-token")
	testhelp.Equal(t, "LookupToken(wrong-token) user", got, User{})
	testhelp.Equal(t, "LookupToken(wrong-token) ok", ok, false)

	got, ok = u.LookupToken("")
	testhelp.Equal(t, "LookupToken(empty) user", got, User{})
	testhelp.Equal(t, "LookupToken(empty) ok", ok, false)

	after := u.List()
	testhelp.Equal(t, "List unchanged", after, before)
}

func TestTokenLookupExample3(t *testing.T) {
	u := tlOpen(t, "admin-op-9")

	testhelp.Equal(t, "IsAdmin(admin-op-9)", u.IsAdmin("admin-op-9"), true)
	testhelp.Equal(t, "IsAdmin(tok-bob-2)", u.IsAdmin("tok-bob-2"), false)
	testhelp.Equal(t, "IsAdmin(empty)", u.IsAdmin(""), false)
}

func TestTokenLookupExample4(t *testing.T) {
	u := tlOpen(t, "")

	testhelp.Equal(t, "IsAdmin(admin-op-9)", u.IsAdmin("admin-op-9"), false)
	testhelp.Equal(t, "IsAdmin(empty)", u.IsAdmin(""), false)
}
