package users

import (
	"errors"
	"os"
	"path/filepath"
	"strings"
	"testing"

	"morphstudio/internal/testhelp"
)

// The example literals: "no secret" and "empty secret" are the texts of ErrNoSecret and ErrEmptySecret (checked
// verbatim below); "sk-ant-old" is the value replaced by "sk-ant-new" in example 4.

func usAliceRegistry(t *testing.T) (*Users, string) {
	t.Helper()
	base := filepath.Join(t.TempDir(), "ustate")
	u, err := Open(base, "")
	if err != nil {
		t.Fatal(err)
	}
	if err := u.Add(User{Name: "alice"}); err != nil {
		t.Fatal(err)
	}
	return u, base
}

func usAliceWithAnthropic(t *testing.T) (*Users, string) {
	t.Helper()
	u, base := usAliceRegistry(t)
	if err := u.PutSecret("alice", "anthropic-key", "sk-ant-abc"); err != nil {
		t.Fatal(err)
	}
	return u, base
}

func usMode(t *testing.T, path string) os.FileMode {
	t.Helper()
	info, err := os.Stat(path)
	if err != nil {
		t.Fatal(err)
	}
	return info.Mode() & 0o777
}

func usDirNames(t *testing.T, dir string) []string {
	t.Helper()
	entries, err := os.ReadDir(dir)
	if err != nil {
		t.Fatal(err)
	}
	names := make([]string, 0, len(entries))
	for _, e := range entries {
		names = append(names, e.Name())
	}
	return names
}

func usErrorText(t *testing.T, what string, err error) string {
	t.Helper()
	if err == nil {
		t.Fatalf("%s: got nil error", what)
	}
	return err.Error()
}

var usSentinels = []error{ErrNotFound, ErrExists, ErrBadName, ErrChatTaken, ErrNoSecret, ErrEmptySecret}

func usCheckNoSecretLeak(t *testing.T, what string, err error, secret string) {
	t.Helper()
	if err == nil {
		return
	}
	for _, sentinel := range usSentinels {
		if errors.Is(err, sentinel) {
			return
		}
	}
	text := err.Error()
	if strings.Contains(text, secret) {
		t.Errorf("%s: error contains full secret %q: %v", what, secret, err)
	}
	for i := 0; i+4 <= len(secret); i++ {
		for j := i + 4; j <= len(secret); j++ {
			part := secret[i:j]
			if strings.Contains(text, part) {
				t.Errorf("%s: error contains part %q of secret: %v", what, part, err)
			}
		}
	}
}

func TestUserSecretsExample1(t *testing.T) {
	u, base := usAliceRegistry(t)
	err := u.PutSecret("alice", "anthropic-key", "sk-ant-abc")
	testhelp.Equal(t, "PutSecret", err, nil)
	testhelp.Equal(t, "HasSecret", u.HasSecret("alice", "anthropic-key"), true)
	got, err := u.ReadSecret("alice", "anthropic-key")
	testhelp.Equal(t, "ReadSecret value", got, "sk-ant-abc")
	testhelp.Equal(t, "ReadSecret error", err, nil)
	path := u.SecretPath("alice", "anthropic-key")
	testhelp.Equal(t, "secret mode", usMode(t, path), os.FileMode(0o600))
	testhelp.Equal(t, "alice dir mode", usMode(t, filepath.Join(base, "alice")), os.FileMode(0o700))
	testhelp.Equal(t, "alice dir entries", usDirNames(t, filepath.Join(base, "alice")), []string{"anthropic-key"})
}

func TestUserSecretsExample2(t *testing.T) {
	u, base := usAliceWithAnthropic(t)
	err := u.PutSecret("alice", "github-token", "ghp_xyz")
	testhelp.Equal(t, "PutSecret github", err, nil)
	got, err := u.ReadSecret("alice", "github-token")
	testhelp.Equal(t, "ReadSecret value", got, "ghp_xyz")
	testhelp.Equal(t, "ReadSecret error", err, nil)
	testhelp.Equal(t, "alice dir entries", usDirNames(t, filepath.Join(base, "alice")), []string{"anthropic-key", "github-token"})
}

func TestUserSecretsExample3(t *testing.T) {
	u, base := usAliceWithAnthropic(t)
	testhelp.Equal(t, "PutSecret unknown user", u.PutSecret("nope", "anthropic-key", "sk-1"), ErrNotFound)
	err := u.PutSecret("alice", "password", "p")
	testhelp.Equal(t, "PutSecret unknown kind text", usErrorText(t, "PutSecret unknown kind", err), `users: unknown secret kind "password"`)
	err = u.PutSecret("alice", "github-token", "")
	testhelp.Equal(t, "PutSecret empty", err, ErrEmptySecret)
	testhelp.Equal(t, "PutSecret empty text", usErrorText(t, "PutSecret empty", err), "empty secret")
	testhelp.Equal(t, "alice dir entries", usDirNames(t, filepath.Join(base, "alice")), []string{"anthropic-key"})
}

func TestUserSecretsExample4(t *testing.T) {
	// the value sk-ant-old (the content before this overwrite path) is replaced by sk-ant-new.
	u, _ := usAliceWithAnthropic(t)
	path := u.SecretPath("alice", "anthropic-key")
	if err := os.Chmod(path, 0o644); err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "PutSecret overwrite", u.PutSecret("alice", "anthropic-key", "sk-ant-new"), nil)
	testhelp.Equal(t, "secret mode", usMode(t, path), os.FileMode(0o600))
	got, err := u.ReadSecret("alice", "anthropic-key")
	testhelp.Equal(t, "ReadSecret value", got, "sk-ant-new")
	testhelp.Equal(t, "ReadSecret error", err, nil)
}

func TestUserSecretsExample5(t *testing.T) {
	u, _ := usAliceWithAnthropic(t)
	testhelp.Equal(t, "HasSecret missing", u.HasSecret("alice", "github-token"), false)
	got, err := u.ReadSecret("alice", "github-token")
	testhelp.Equal(t, "ReadSecret missing value", got, "")
	testhelp.Equal(t, "ReadSecret missing error", err, ErrNoSecret)
	testhelp.Equal(t, "ReadSecret missing text", usErrorText(t, "ReadSecret missing", err), "no secret")
	got, err = u.ReadSecret("nope", "anthropic-key")
	testhelp.Equal(t, "ReadSecret unknown user value", got, "")
	testhelp.Equal(t, "ReadSecret unknown user error", err, ErrNotFound)
	testhelp.Equal(t, "DeleteSecret missing", u.DeleteSecret("alice", "github-token"), nil)
}

func TestUserSecretsExample6(t *testing.T) {
	u, base := usAliceWithAnthropic(t)
	if err := u.PutSecret("alice", "github-token", "ghp_new"); err != nil {
		t.Fatal(err)
	}

	usCheckNoSecretLeak(t, "PutSecret unknown user", u.PutSecret("nope", "github-token", "ghp_new"), "ghp_new")
	usCheckNoSecretLeak(t, "PutSecret unknown kind", u.PutSecret("alice", "password", "ghp_new"), "ghp_new")
	usCheckNoSecretLeak(t, "PutSecret empty", u.PutSecret("alice", "github-token", ""), "ghp_new")
	_, err := u.ReadSecret("nope", "github-token")
	usCheckNoSecretLeak(t, "ReadSecret unknown user", err, "ghp_new")
	_, err = u.ReadSecret("alice", "no-such-secret")
	usCheckNoSecretLeak(t, "ReadSecret missing", err, "ghp_new")

	aliceDir := filepath.Join(base, "alice")
	if err := os.RemoveAll(aliceDir); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(aliceDir, []byte("not a directory"), 0o444); err != nil {
		t.Fatal(err)
	}

	usCheckNoSecretLeak(t, "PutSecret fs", u.PutSecret("alice", "github-token", "ghp_new"), "ghp_new")
	usCheckNoSecretLeak(t, "PutSecret fs anthropic", u.PutSecret("alice", "anthropic-key", "ghp_new"), "ghp_new")
	_, err = u.ReadSecret("alice", "github-token")
	usCheckNoSecretLeak(t, "ReadSecret fs github", err, "ghp_new")
	_, err = u.ReadSecret("alice", "anthropic-key")
	usCheckNoSecretLeak(t, "ReadSecret fs anthropic", err, "ghp_new")
}
