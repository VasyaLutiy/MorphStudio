package users

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
)

var (
	ErrNoSecret    = errors.New("no secret")
	ErrEmptySecret = errors.New("empty secret")
)

// SecretKinds lists the kinds of per-user secret a user may hold.
var SecretKinds = []string{"anthropic-key", "github-token"}

func knownSecretKind(kind string) bool {
	for _, k := range SecretKinds {
		if k == kind {
			return true
		}
	}
	return false
}

// SecretPath is the path of a user's secret of the given kind.
func (u *Users) SecretPath(name, kind string) string {
	return filepath.Join(u.base, name, kind)
}

// PutSecret writes a user's secret of the given kind. The value is written
// through a temporary file of mode 0600 in the user's directory (mode 0700)
// and renamed onto SecretPath, so the file always ends with mode 0600. No
// error text returned here ever contains the value.
func (u *Users) PutSecret(name, kind, value string) error {
	if _, ok := u.byName[name]; !ok {
		return ErrNotFound
	}
	if !knownSecretKind(kind) {
		return fmt.Errorf("users: unknown secret kind %q", kind)
	}
	if value == "" {
		return ErrEmptySecret
	}
	dir := filepath.Join(u.base, name)
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return err
	}
	tmp, err := os.CreateTemp(dir, ".secret-")
	if err != nil {
		return err
	}
	tmpName := tmp.Name()
	if err := tmp.Chmod(0o600); err != nil {
		tmp.Close()
		os.Remove(tmpName)
		return err
	}
	if _, err := tmp.WriteString(value); err != nil {
		tmp.Close()
		os.Remove(tmpName)
		return err
	}
	if err := tmp.Close(); err != nil {
		os.Remove(tmpName)
		return err
	}
	if err := os.Rename(tmpName, filepath.Join(dir, kind)); err != nil {
		os.Remove(tmpName)
		return err
	}
	return nil
}

// HasSecret reports whether a secret file of the given kind exists.
func (u *Users) HasSecret(name, kind string) bool {
	info, err := os.Stat(u.SecretPath(name, kind))
	if err != nil {
		return false
	}
	return !info.IsDir()
}

// ReadSecret returns the content of a user's secret verbatim. The value is
// for morphd's own use only: it is never returned to a client, never logged
// and never appears in any error text.
func (u *Users) ReadSecret(name, kind string) (string, error) {
	if _, ok := u.byName[name]; !ok {
		return "", ErrNotFound
	}
	data, err := os.ReadFile(u.SecretPath(name, kind))
	if err != nil {
		if os.IsNotExist(err) {
			return "", ErrNoSecret
		}
		return "", err
	}
	return string(data), nil
}

// DeleteSecret removes a user's secret file. A missing file is not an error.
func (u *Users) DeleteSecret(name, kind string) error {
	if _, ok := u.byName[name]; !ok {
		return ErrNotFound
	}
	if err := os.Remove(u.SecretPath(name, kind)); err != nil {
		if os.IsNotExist(err) {
			return nil
		}
		return err
	}
	return nil
}
