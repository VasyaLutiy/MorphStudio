```go
package users

import (
	"path/filepath"
	"testing"
	"time"

	"morphstudio/internal/testhelp"
)

// tlStoreWithAdmin opens a fresh store under the test's temporary directory and
// adds the two users ada and zoe; adminToken is the store's admin token.
func tlStoreWithAdmin(t *testing.T, adminToken string) *Store {
	t.Helper()
	s, err := Open(filepath.Join(t.TempDir(), "users"), adminToken)
	if err != nil {
		t.Fatal(err)
	}
	if err := s.Add(User{
		Name:      "ada",
		LinuxUser: "mu-ada",
		ChatID:    111,
		CreatedAt: time.Date(2026, 10, 11, 9, 0, 0, 0, time.UTC),
	}, "tok-ada"); err != nil {
		t.Fatal(err)
	}
	if err := s.Add(User{
		Name:      "zoe",
		LinuxUser: "mu-zoe",
		ChatID:    999,
		CreatedAt: time.Date(2026, 10, 11, 9, 1, 0, 0, time.UTC),
	}, "tok-zoe"); err != nil {
		t.Fatal(err)
	}
	return s
}

// tlStore is the store of example 1: ada and zoe, opened with admin token "admin-0".
func tlStore(t *testing.T) *Store {
	t.Helper()
	return tlStoreWithAdmin(t, "admin-0")
}

// tlAda is the stored ada user, token "tok-ada".
func tlAda() User {
	return User{
		Name:      "ada",
		LinuxUser: "mu-ada",
		ChatID:    111,
		TokenHash: hashToken("tok-ada"),
		CreatedAt: time.Date(2026, 10, 11, 9, 0, 0, 0, time.UTC),
	}
}

// tlZoe is the stored zoe user, token "tok-zoe".
func tlZoe() User {
	return User{
		Name:      "zoe",
		LinuxUser: "mu-zoe",
		ChatID:    999,
		TokenHash: hashToken("tok-zoe"),
		CreatedAt: time.Date(2026, 10, 11, 9, 1, 0, 0, time.UTC),
	}
}

// Token Lookup example 1: Lookup finds ada by "tok-ada" and zoe by "tok-zoe",
// and finds nothing for an unknown token.
func TestTokenLookupExample1(t *testing.T) {
	s := tlStore(t)

	u, ok := s.Lookup("tok-ada")
	testhelp.Equal(t, "Lookup(tok-ada) user", u, tlAda())
	testhelp.Equal(t, "Lookup(tok-ada) ok", ok, true)

	u, ok = s.Lookup("tok-zoe")
	testhelp.Equal(t, "Lookup(tok-zoe) user", u, tlZoe())
	testhelp.Equal(t, "Lookup(tok-zoe) ok", ok, true)

	u, ok = s.Lookup("tok-mid")
	testhelp.Equal(t, "Lookup(tok-mid) user", u, User{})
	testhelp.Equal(t, "Lookup(tok-mid) ok", ok, false)
}

// Token Lookup example 2: the admin token is the admin token, not a user's;
// a user's token is not the admin token; the empty token is neither.
func TestTokenLookupExample2(t *testing.T) {
	s := tlStore(t)

	testhelp.Equal(t, "IsAdminToken(admin-0)", s.IsAdminToken("admin-0"), true)
	testhelp.Equal(t, "IsAdminToken(tok-ada)", s.IsAdminToken("tok-ada"), false)
	testhelp.Equal(t, "IsAdminToken(empty)", s.IsAdminToken(""), false)

	u, ok := s.Lookup("admin-0")
	testhelp.Equal(t, "Lookup(admin-0) user", u, User{})
	testhelp.Equal(t, "Lookup(admin-0) ok", ok, false)
}

// Token Lookup example 3: a store opened with no admin token has no admin token,
// not even for "admin-0" or the empty token.
func TestTokenLookupExample3(t *testing.T) {
	s := tlStoreWithAdmin(t, "")

	testhelp.Equal(t, "IsAdminToken(admin-0)", s.IsAdminToken("admin-0"), false)
	testhelp.Equal(t, "IsAdminToken(empty)", s.IsAdminToken(""), false)
}

// Token Lookup example 4: the user returned by Lookup is a copy, so changing it
// does not change the stored user.
func TestTokenLookupExample4(t *testing.T) {
	s := tlStore(t)

	u, ok := s.Lookup("tok-ada")
	testhelp.Equal(t, "Lookup(tok-ada) ok", ok, true)
	testhelp.Equal(t, "Lookup(tok-ada) user", u, tlAda())

	u.Name = "changed"

	got, ok := s.Lookup("tok-ada")
	testhelp.Equal(t, "Lookup(tok-ada) user after copy", got, tlAda())
	testhelp.Equal(t, "Lookup(tok-ada) ok after copy", ok, true)
}
```
