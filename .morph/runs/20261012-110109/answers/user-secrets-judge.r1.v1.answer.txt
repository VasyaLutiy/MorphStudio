package users

import (
	"errors"

	"morphstudio/internal/testhelp"

func TestUserSecretsExample1(t *testing.T) {
	base := filepath.Join(t.TempDir(), "users")
	s, err := Open(base, "admin-0")
	if err != nil {
		t.Fatal(err)
	}
	err = s.Add(User{
		Name:      "ada",
		LinuxUser: "mu-ada",
		ChatID:    111,
		CreatedAt: time.Date(2026, 10, 11, 9, 0, 0, 0, time.UTC),
	}, "tok-ada")
	if err != nil {
		t.Fatal(err)
	}

	err = s.PutSecret("ada", "anthropic-key", "sk-ant-xyz1")
	testhelp.Equal(t, "PutSecret err", err, nil)
	testhelp.Equal(t, "HasSecret", s.HasSecret("ada", "anthropic-key"), true)

	value, err := s.ReadSecret("ada", "anthropic-key")
	testhelp.Equal(t, "ReadSecret value", value, "sk-ant-xyz1")
	testhelp.Equal(t, "ReadSecret err", err, nil)

	info, err := os.Stat(s.SecretPath("ada", "anthropic-key"))
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "secret mode", info.Mode()&0o777, os.FileMode(0o600))

	dirInfo, err := os.Stat(filepath.Join(base, "ada"))
	if err != nil {
		t.Fatal(err)
	}
	testhelp.Equal(t, "user dir mode", dirInfo.Mode()&0o777, os.FileMode(0o700))

	entries, err := os.ReadDir(filepath.Join(base, "ada"))
	if err != nil {
		t.Fatal(err)
	}
	names := []string{}
	for _, e := range entries {
		names = append(names, e.Name())
	}
	testhelp.Equal(t, "user dir holds", names, []string{"anthropic-key"})
}

	err = s.Add(User{
		Name:      "ada",
		LinuxUser: "mu-ada",
		ChatID:    111,
		CreatedAt: time.Date(2026, 10, 11, 9, 0, 0, 0, time.UTC),
	}, "tok-ada")
	if err != nil {
		t.Fatal(err)
	}
	if err := s.PutSecret("ada", "anthropic-key", "sk-ant-xyz1"); err != nil {
		t.Fatal(err)
	}

	err = s.PutSecret("nope", "anthropic-key", "x-1")
	testhelp.Equal(t, "PutSecret unknown user is ErrNotFound", errors.Is(err, ErrNotFound), true)

	err = s.PutSecret("ada", "password", "x-1")
	if err == nil {
		t.Fatal("PutSecret with unknown kind returned nil error")
	}
	testhelp.Equal(t, "unknown kind error text", err.Error(), "users: unknown secret kind \"password\"")

	err = s.PutSecret("ada", "github-token", "")
	testhelp.Equal(t, "PutSecret empty value is ErrEmptySecret", errors.Is(err, ErrEmptySecret), true)

	entries, err := os.ReadDir(filepath.Join(base, "ada"))
	if err != nil {
		t.Fatal(err)
	}
	names := []string{}
	for _, e := range entries {
		names = append(names, e.Name())
	}
	testhelp.Equal(t, "user dir holds", names, []string{"anthropic-key"})
}

		t.Fatal(err)
	}
	if err := s.PutSecret("ada", "anthropic-key", "sk-ant-xyz1"); err != nil {
		t.Fatal(err)
	}

	err1 := s.PutSecret("nope", "anthropic-key", "sk-very-secret-2")
	if err1 == nil {
		t.Fatal("PutSecret for unknown user returned nil error")
	}
	testhelp.Equal(t, "anthropic-key err is ErrNotFound", errors.Is(err1, ErrNotFound), true)
	testhelp.Equal(t, "anthropic-key err leaks value", strings.Contains(err1.Error(), "sk-very-secret-2"), false)

	err2 := s.PutSecret("nope", "github-token", "ghp-very-secret-2")
	if err2 == nil {
		t.Fatal("PutSecret for unknown user returned nil error")
	}
	testhelp.Equal(t, "github-token err is ErrNotFound", errors.Is(err2, ErrNotFound), true)
	testhelp.Equal(t, "github-token err leaks value", strings.Contains(err2.Error(), "ghp-very-secret-2"), false)
}

	testhelp.Equal(t, "ReadSecret value", value, "sk-ant-new")
	testhelp.Equal(t, "ReadSecret err", err, nil)
}

	testhelp.Equal(t, "ReadSecret missing value", value, "")
	testhelp.Equal(t, "ReadSecret missing err is ErrNoSecret", errors.Is(err, ErrNoSecret), true)

	testhelp.Equal(t, "ReadSecret unknown user value", value, "")
	testhelp.Equal(t, "ReadSecret unknown user err is ErrNotFound", errors.Is(err, ErrNotFound), true)

	testhelp.Equal(t, "DeleteSecret missing", s.DeleteSecret("ada", "github-token"), nil)
}
