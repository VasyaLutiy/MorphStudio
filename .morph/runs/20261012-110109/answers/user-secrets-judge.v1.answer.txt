```go
package users

import (
	"errors"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"

	"morphstudio/internal/testhelp"
)

func usAda(t *testing.T) *Store {
	t.Helper()
	s, err := Open(filepath.Join(t.TempDir(), "users"), "admin-0")
	if err != nil {
		t.Fatal(err)
	}
	err = s.Add(User{
		Name:      "ada",
		LinuxUser: "mu-ada",
		ChatID:    111,
		CreatedAt: time.Date(2026, 10, 11, 9, 0, 0, 0, time.UTC),
	}, "tok-ada")
	if err != nil {
		t.Fatal(err)
	}
	return s
}

func usAdaWithAnthropic(t *testing.T) *Store {
	t.Helper()
	s := usAda(t)
	if err := s.PutSecret("ada", "anthropic-key", "sk-ant-xyz1"); err != nil {
		t.Fatal(err)
	}
	return s
}

func usMode(t *testing.T, path string) os.FileMode {
	t.Helper()
	info, err := os.Stat(path)
	if err != nil {
		t.Fatal(err)
	}
	return info.Mode() & 0o777
}

func usDirNames(t *testing.T, dir string) []string {
	t.Helper()
	entries, err := os.ReadDir(dir)
	if err != nil {
		t.Fatal(err)
	}
	names := make([]string, 0, len(entries))
	for _, e := range entries {
		names = append(names, e.Name())
	}
	return names
}

func TestUserSecretsExample1(t *testing.T) {
	s := usAda(t)

	testhelp.Equal(t, "PutSecret err", s.PutSecret("ada", "anthropic-key", "sk-ant-xyz1"), nil)
	testhelp.Equal(t, "HasSecret", s.HasSecret("ada", "anthropic-key"), true)

	val, err := s.ReadSecret("ada", "anthropic-key")
	testhelp.Equal(t, "ReadSecret value", val, "sk-ant-xyz1")
	testhelp.Equal(t, "ReadSecret err", err, nil)

	path := s.SecretPath("ada", "anthropic-key")
	testhelp.Equal(t, "secret mode", usMode(t, path), os.FileMode(0o600))
	testhelp.Equal(t, "user dir mode", usMode(t, filepath.Join(s.Base(), "ada")), os.FileMode(0o700))
	testhelp.Equal(t, "user dir holds", usDirNames(t, filepath.Join(s.Base(), "ada")), []string{"anthropic-key"})
}

func TestUserSecretsExample2(t *testing.T) {
	s := usAdaWithAnthropic(t)

	testhelp.Equal(t, "PutSecret err", s.PutSecret("ada", "github-token", "ghp_abc"), nil)

	val, err := s.ReadSecret("ada", "github-token")
	testhelp.Equal(t, "ReadSecret value", val, "ghp_abc")
	testhelp.Equal(t, "ReadSecret err", err, nil)

	testhelp.Equal(t, "secret mode", usMode(t, s.SecretPath("ada", "github-token")), os.FileMode(0o600))
	testhelp.Equal(t, "user dir holds", usDirNames(t, filepath.Join(s.Base(), "ada")), []string{"anthropic-key", "github-token"})
}

func TestUserSecretsExample3(t *testing.T) {
	s := usAdaWithAnthropic(t)

	err := s.PutSecret("nope", "anthropic-key", "x-1")
	testhelp.Equal(t, "unknown user err is ErrNotFound", errors.Is(err, ErrNotFound), true)

	err = s.PutSecret("ada", "password", "x-1")
	if err == nil {
		t.Fatal("PutSecret with unknown kind returned nil")
	}
	testhelp.Equal(t, "unknown kind err text", err.Error(), "users: unknown secret kind \"password\"")

	err = s.PutSecret("ada", "github-token", "")
	testhelp.Equal(t, "empty secret err is ErrEmptySecret", errors.Is(err, ErrEmptySecret), true)

	testhelp.Equal(t, "user dir holds", usDirNames(t, filepath.Join(s.Base(), "ada")), []string{"anthropic-key"})
}

func TestUserSecretsExample4(t *testing.T) {
	s := usAdaWithAnthropic(t)

	err1 := s.PutSecret("nope", "anthropic-key", "sk-very-secret-2")
	if err1 == nil {
		t.Fatal("PutSecret for unknown user returned nil")
	}
	testhelp.Equal(t, "err1 is ErrNotFound", errors.Is(err1, ErrNotFound), true)
	testhelp.Equal(t, "err1 leaks value", strings.Contains(err1.Error(), "sk-very-secret-2"), false)

	err2 := s.PutSecret("nope", "github-token", "ghp-very-secret-2")
	if err2 == nil {
		t.Fatal("PutSecret for unknown user returned nil")
	}
	testhelp.Equal(t, "err2 is ErrNotFound", errors.Is(err2, ErrNotFound), true)
	testhelp.Equal(t, "err2 leaks value", strings.Contains(err2.Error(), "ghp-very-secret-2"), false)
}

func TestUserSecretsExample5(t *testing.T) {
	s := usAdaWithAnthropic(t)

	path := s.SecretPath("ada", "anthropic-key")
	if err := os.Chmod(path, 0o644); err != nil {
		t.Fatal(err)
	}

	testhelp.Equal(t, "PutSecret err", s.PutSecret("ada", "anthropic-key", "sk-ant-new"), nil)
	testhelp.Equal(t, "secret mode", usMode(t, path), os.FileMode(0o600))

	val, err := s.ReadSecret("ada", "anthropic-key")
	testhelp.Equal(t, "ReadSecret value", val, "sk-ant-new")
	testhelp.Equal(t, "ReadSecret err", err, nil)
}

func TestUserSecretsExample6(t *testing.T) {
	s := usAdaWithAnthropic(t)

	testhelp.Equal(t, "HasSecret github-token", s.HasSecret("ada", "github-token"), false)

	val, err := s.ReadSecret("ada", "github-token")
	testhelp.Equal(t, "ReadSecret missing value", val, "")
	testhelp.Equal(t, "ReadSecret missing err is ErrNoSecret", errors.Is(err, ErrNoSecret), true)

	val, err = s.ReadSecret("nope", "anthropic-key")
	testhelp.Equal(t, "ReadSecret unknown user value", val, "")
	testhelp.Equal(t, "ReadSecret unknown user err is ErrNotFound", errors.Is(err, ErrNotFound), true)

	testhelp.Equal(t, "DeleteSecret missing", s.DeleteSecret("ada", "github-token"), nil)
}

func TestUserSecretsExample7(t *testing.T) {
	s := usAdaWithAnthropic(t)

	testhelp.Equal(t, "PutSecret github-token", s.PutSecret("ada", "github-token", "tok-9"), nil)
	testhelp.Equal(t, "DeleteSecret", s.DeleteSecret("ada", "github-token"), nil)
	testhelp.Equal(t, "HasSecret after delete", s.HasSecret("ada", "github-token"), false)
	testhelp.Equal(t, "user dir holds", usDirNames(t, filepath.Join(s.Base(), "ada")), []string{"anthropic-key"})
}
```
