```go
package users

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
)

// ErrNoSecret is returned by ReadSecret when the user has no stored secret of
// the requested kind.
var ErrNoSecret = errors.New("no secret")

// ErrEmptySecret is returned by PutSecret when the value to store is empty.
var ErrEmptySecret = errors.New("empty secret")

// SecretKinds lists the secret kinds a user may hold.
var SecretKinds = []string{"anthropic-key", "github-token"}

// knownSecretKind reports whether kind is one of SecretKinds.
func knownSecretKind(kind string) bool {
	for _, k := range SecretKinds {
		if k == kind {
			return true
		}
	}
	return false
}

// SecretPath is the file that holds name's secret of kind. The secret lives
// in a per-user directory under the store's base.
func (s *Store) SecretPath(name, kind string) string {
	return filepath.Join(s.base, name, kind)
}

// PutSecret stores value as name's secret of kind. The user directory is
// created with mode 0o700 and the value is written through a temporary file
// of mode 0o600 renamed onto SecretPath, so the stored file always has mode
// 0o600, even when it existed before with another mode.
func (s *Store) PutSecret(name, kind, value string) error {
	if _, err := s.Get(name); err != nil {
		return ErrNotFound
	}
	if !knownSecretKind(kind) {
		return fmt.Errorf("users: unknown secret kind %q", kind)
	}
	if value == "" {
		return ErrEmptySecret
	}

	dir := filepath.Join(s.base, name)
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return err
	}
	if err := os.Chmod(dir, 0o700); err != nil {
		return err
	}

	f, err := os.CreateTemp(dir, "secret-*")
	if err != nil {
		return err
	}
	tmp := f.Name()
	if _, err := f.WriteString(value); err != nil {
		f.Close()
		os.Remove(tmp)
		return err
	}
	if err := f.Chmod(0o600); err != nil {
		f.Close()
		os.Remove(tmp)
		return err
	}
	if err := f.Close(); err != nil {
		os.Remove(tmp)
		return err
	}
	if err := os.Rename(tmp, s.SecretPath(name, kind)); err != nil {
		os.Remove(tmp)
		return err
	}
	return nil
}

// HasSecret reports whether name has a stored secret of kind.
func (s *Store) HasSecret(name, kind string) bool {
	_, err := os.Stat(s.SecretPath(name, kind))
	return err == nil
}

// ReadSecret returns name's secret of kind verbatim. A missing secret is
// ErrNoSecret and an unknown user is ErrNotFound.
func (s *Store) ReadSecret(name, kind string) (string, error) {
	if _, err := s.Get(name); err != nil {
		return "", ErrNotFound
	}
	data, err := os.ReadFile(s.SecretPath(name, kind))
	if err != nil {
		if errors.Is(err, os.ErrNotExist) {
			return "", ErrNoSecret
		}
		return "", err
	}
	return string(data), nil
}

// DeleteSecret removes name's secret of kind. Removing a secret that does not
// exist is not an error.
func (s *Store) DeleteSecret(name, kind string) error {
	if err := os.Remove(s.SecretPath(name, kind)); err != nil {
		if errors.Is(err, os.ErrNotExist) {
			return nil
		}
		return err
	}
	return nil
}
```
