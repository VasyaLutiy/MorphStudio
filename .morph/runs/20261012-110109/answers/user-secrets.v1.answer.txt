package users

import (
	"errors"
	"fmt"
	"os"
	"path/filepath"
)

// ErrNoSecret is returned by ReadSecret when the user has no value for a kind.
var ErrNoSecret = errors.New("no secret")

// ErrEmptySecret is returned by PutSecret when the value is empty.
var ErrEmptySecret = errors.New("empty secret")

// SecretKinds lists the secret kinds a user may hold.
var SecretKinds = []string{"anthropic-key", "github-token"}

func validSecretKind(kind string) bool {
	for _, k := range SecretKinds {
		if k == kind {
			return true
		}
	}
	return false
}

// SecretPath is the file that holds name's secret of the given kind.
func (s *Store) SecretPath(name, kind string) string {
	return filepath.Join(s.base, name, kind)
}

// PutSecret stores value as name's secret of kind, writing it through a
// 0o600 temporary file renamed onto SecretPath.
func (s *Store) PutSecret(name, kind, value string) error {
	if !validSecretKind(kind) {
		return fmt.Errorf("users: unknown secret kind %q", kind)
	}
	if value == "" {
		return ErrEmptySecret
	}

	s.mu.RLock()
	_, ok := s.users[name]
	s.mu.RUnlock()
	if !ok {
		return ErrNotFound
	}

	dir := filepath.Join(s.base, name)
	if err := os.MkdirAll(dir, 0o700); err != nil {
		return err
	}
	if err := os.Chmod(dir, 0o700); err != nil {
		return err
	}

	f, err := os.CreateTemp(dir, "secret-*")
	if err != nil {
		return err
	}
	tmpName := f.Name()
	defer func() {
		if tmpName != "" {
			os.Remove(tmpName)
		}
	}()

	if _, err := f.WriteString(value); err != nil {
		f.Close()
		return err
	}
	if err := f.Chmod(0o600); err != nil {
		f.Close()
		return err
	}
	if err := f.Close(); err != nil {
		return err
	}
	if err := os.Rename(tmpName, s.SecretPath(name, kind)); err != nil {
		return err
	}
	tmpName = ""
	return nil
}

// HasSecret reports whether name has a secret of kind stored.
func (s *Store) HasSecret(name, kind string) bool {
	s.mu.RLock()
	_, ok := s.users[name]
	s.mu.RUnlock()
	if !ok {
		return false
	}
	_, err := os.Stat(s.SecretPath(name, kind))
	return err == nil
}

// ReadSecret returns name's secret of kind verbatim.
func (s *Store) ReadSecret(name, kind string) (string, error) {
	s.mu.RLock()
	_, ok := s.users[name]
	s.mu.RUnlock()
	if !ok {
		return "", ErrNotFound
	}

	data, err := os.ReadFile(s.SecretPath(name, kind))
	if err != nil {
		if errors.Is(err, os.ErrNotExist) {
			return "", ErrNoSecret
		}
		return "", err
	}
	return string(data), nil
}

// DeleteSecret removes name's secret of kind; a missing secret is not an error.
func (s *Store) DeleteSecret(name, kind string) error {
	s.mu.RLock()
	_, ok := s.users[name]
	s.mu.RUnlock()
	if !ok {
		return ErrNotFound
	}

	if err := os.Remove(s.SecretPath(name, kind)); err != nil {
		if errors.Is(err, os.ErrNotExist) {
			return nil
		}
		return err
	}
	return nil
}
